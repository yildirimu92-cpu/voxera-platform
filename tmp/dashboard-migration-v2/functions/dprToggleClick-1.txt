function dprToggleClick(e, rowEl) {
  e.stopPropagation();
  if (window._vxInboxSelectionMode && rowEl && rowEl.closest && rowEl.closest('#anrufe-list, #anrufe-split-left')) {
    return;
  }
  // Touch-Geräte: click nach touchend ignorieren
  if (e.type === 'click' && (Date.now() - _dprLastTouch) < 800) return;
  if (e.target.closest('.dpr-btn, .dpr-quick-btn, .vx-ibtn, button')) return;
  var _clickedCard = rowEl && rowEl.closest ? rowEl.closest('.dpr-card') : null;
  var _clickedId = (rowEl && rowEl.dataset && (rowEl.dataset.id || rowEl.dataset.recordId)) || (_clickedCard && _clickedCard.dataset ? _clickedCard.dataset.id : '');
  if (_clickedId && typeof vxSetActiveRequestRow === 'function') vxSetActiveRequestRow(_clickedId);
  if (_dprToggleLock) return;
  _dprToggleLock = true;
  setTimeout(function(){ _dprToggleLock = false; }, 400);
  // Als gelesen markieren — Punkt entfernen
  vxMarkRowAsRead(rowEl);
  // Anfragen Split: Klick → direkt Detail-Seite
  if (dprOpenAnfragenDetailFromRow(rowEl)) return;
  // [v3.0] Heute-Tab → immer Detail-Seite (kein Expand)
  var inHeuteList = rowEl.closest('#dash-priority-list');
  if (inHeuteList) {
    var recId = rowEl.dataset.id || rowEl.getAttribute('data-record-id');
    if (recId && vxOpenPriorityRecord(recId)) return;
  }
  _dprToggleExec(rowEl);
}
