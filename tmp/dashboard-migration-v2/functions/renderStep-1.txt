function renderStep() {
    const subEl = document.getElementById('confirm-sub');
    if (!subEl) return;
    if (step === 1) {
      subEl.innerHTML = '<div style="font-size:16px;font-weight:700;color:var(--ink);margin-bottom:14px;">Warum möchten Sie kündigen?</div>'
        + '<div id="vx-cancel-reasons" style="display:grid;gap:8px;"></div>'
        + '<div id="vx-cancel-other-wrap" style="display:none;margin-top:10px;"><textarea id="vx-cancel-other" placeholder="Bitte Grund eingeben" style="width:100%;min-height:88px;border:0.5px solid var(--line);border-radius:10px;padding:10px;font-size:13px;font-family:inherit;resize:vertical;"></textarea></div>';
      const reasonsEl = subEl.querySelector('#vx-cancel-reasons');
      reasonsEl.innerHTML = reasons.map(function(reason) {
        const active = selectedReason === reason;
        return '<button data-reason="' + reason.replace(/"/g, '&quot;') + '" style="text-align:left;background:' + (active ? '#EFF6FF' : '#fff') + ';border:0.5px solid ' + (active ? '#1A6FE8' : 'var(--line)') + ';color:var(--ink);border-radius:10px;padding:10px 12px;font-size:13px;font-family:inherit;cursor:pointer;">' + reason + '</button>';
      }).join('');
      reasonsEl.querySelectorAll('button[data-reason]').forEach(function(btn) {
        btn.onclick = function() {
          selectedReason = btn.getAttribute('data-reason') || '';
          renderStep();
        };
      });
      const isOther = selectedReason === 'Sonstiges';
      const otherWrap = subEl.querySelector('#vx-cancel-other-wrap');
      otherWrap.style.display = isOther ? 'block' : 'none';
      const otherInput = subEl.querySelector('#vx-cancel-other');
      otherInput.value = customReason;
      otherInput.oninput = function() { customReason = String(otherInput.value || ''); refreshButtons(); };
    } else {
      const reasonText = selectedReason === 'Sonstiges' ? customReason.trim() : selectedReason;
      subEl.innerHTML = ''
        + '<div style="font-size:16px;font-weight:700;color:var(--ink);margin-bottom:8px;">Kündigung bestätigen</div>'
        + '<div style="font-size:13px;color:var(--ink2);line-height:1.55;margin-bottom:8px;">Ausgewählter Grund: <strong>' + (reasonText || '–') + '</strong></div>'
        + '<div style="font-size:13px;color:var(--ink2);line-height:1.55;margin-bottom:8px;">Vertragsende: <strong>' + vxFmtDate(cancellationDate.toISOString()) + '</strong></div>'
        + '<div style="font-size:12px;color:#B45309;background:#FFFBEB;border:0.5px solid #FDE68A;border-radius:10px;padding:9px 10px;">Nach Ablauf der Kündigungsfrist wird Ihr Zugang deaktiviert.</div>';
    }
    refreshButtons();
  }
  function refreshButtons() {
    const okBtn = document.getElementById('confirm-ok-btn');
    const cancelBtn = document.querySelector('#confirm-overlay .modal-btn-s');
    if (!okBtn || !cancelBtn) return;
    const canContinue = !!selectedReason && (selectedReason !== 'Sonstiges' || !!customReason.trim());
    if (step === 1) {
      cancelBtn.style.display = 'none';
      okBtn.textContent = 'Weiter →';
      okBtn.style.background = '#0D1F3C';
      okBtn.disabled = !canContinue;
      okBtn.style.opacity = canContinue ? '1' : '.45';
      okBtn.style.cursor = canContinue ? 'pointer' : 'not-allowed';
    } else {
      cancelBtn.style.display = '';
      cancelBtn.textContent = 'Abbrechen';
      okBtn.textContent = 'Ja, Kündigung einreichen';
      okBtn.style.background = '#DC2626';
      okBtn.disabled = false;
      okBtn.style.opacity = '1';
      okBtn.style.cursor = 'pointer';
    }
  }
  openConfirmModal({
    title: '',
    text: '',
    confirmLabel: 'Weiter →',
    onConfirm: async function() {
      if (step === 1) { step = 2; renderStep(); return false; }
      const reasonText = selectedReason === 'Sonstiges' ? customReason.trim() : selectedReason;
      const cancelRes = await fetch('/.netlify/functions/customer-cancel-contract', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ action: 'cancel', termination_reason: reasonText })
      });
      if (!cancelRes.ok) throw new Error('Kündigung konnte nicht eingereicht werden.');
      await fetch('/.netlify/functions/ai-change-notify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          mail_type: 'cancellation_request',
          customer_id: customerMeta.customerId || null,
          customer_name: customerMeta.customerName || null,
          plan: customerMeta.plan || null,
          cancellation_notice: ct.cancellation_notice || '30 Tage'
        })
      });
      ct.cancellation_date = cancellationDate.toISOString();
      ct.termination_reason = reasonText;
      ct.termination_type = 'customer_request';
      if (customerContractMeta) {
        customerContractMeta.cancellation_date = ct.cancellation_date;
        customerContractMeta.termination_reason = ct.termination_reason;
        customerContractMeta.termination_type = ct.termination_type;
      }
      vxRenderContractDetails();
      vxRenderCancellationSection();
      toast('Kündigung eingereicht.');
      return true;
    }
  });
  renderStep();
}

async function vxSubmitReactivationRequest() {
  openConfirmModal({
    title: 'Möchten Sie Ihre Kündigung wirklich zurückziehen?',
    text: '',
    confirmLabel: 'Ja, zurückziehen',
    onConfirm: async function() {
      const res = await fetch('/.netlify/functions/customer-cancel-contract', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ action: 'reactivate' })
      });
      if (!res.ok) throw new Error('Kündigung konnte nicht zurückgezogen werden.');
      const ct = customerContractMeta || (customerContractState && (customerContractState.effectiveContract || customerContractState.latestContract)) || {};
      ct.cancellation_date = null;
      ct.termination_reason = null;
      ct.termination_type = null;
      if (customerContractMeta) {
        customerContractMeta.cancellation_date = null;
        customerContractMeta.termination_reason = null;
        customerContractMeta.termination_type = null;
      }
      vxRenderContractDetails();
      vxRenderCancellationSection();
      toast('Kündigung zurückgezogen. Ihr Abonnement läuft weiter.');
      return true;
    }
  });
}

async function vxRequestAddon(addonCode) {
  const addonNames = {
    sms_notify_kunde: 'SMS Benachrichtigungen (CHF 9/Mt)',
    sms_confirm_caller: 'SMS Bestätigung an Anrufer (CHF 19/Mt)',
    extra_minutes_50: 'Zusatzminuten kaufen — 50 Min. für CHF 15.00 einmalig',
    plan_upgrade: 'Plan-Upgrade anfragen — aktueller Plan: ' + (customerMeta.plan || 'unbekannt')
  };
  const name = addonNames[addonCode] || addonCode;
  try {
    const _sb = getSupabaseAuthClient();
    const { data: { session } } = await _sb.auth.getSession();
    await fetch('/.netlify/functions/customer-change-request', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${session?.access_token || ''}` },
      body: JSON.stringify({
        mail_type: 'addon_request',
        addon_code: addonCode,
        message: `Add-on Anfrage: ${name}`
      })
    });
    alert('✓ Anfrage gesendet. Wir aktivieren das Add-on und informieren Sie per E-Mail.');
  } catch(e) {
    alert('Fehler beim Senden der Anfrage.');
  }
}

  }
  function makeSecondaryCode(code, label) {
    var filled = String(code || '').replace('{NR}', nr);
    return '<div class="vx-forwarding-code-secondary">'
      + '<div><div class="vx-forwarding-code-secondary-label">' + label.replace(/\s*\(empfohlen\)/i,'') + '</div><div class="vx-forwarding-code-secondary-value">' + filled + '</div></div>'
      + '<div style="display:flex;gap:8px;flex-shrink:0;"><button type="button" class="btn btn--secondary btn--sm" onclick="vxCopyCode(this)" data-code="' + encodeAttr(filled) + '">Kopieren</button><a class="btn btn--ghost btn--sm" href="' + makeTelHref(filled) + '">Anwählen</a></div>'
    + '</div>';
  }

  var instrHtml = '';
  if (provider.steps && provider.steps.length) {
    instrHtml += '<div class="vx-forwarding-steps">';
    provider.steps.forEach(function(s, i) {
      instrHtml += '<div class="vx-forwarding-step">'
        + '<div class="vx-forwarding-step-num">' + (i+1) + '</div>'
        + '<div><div class="vx-forwarding-step-text">' + s.text + '</div>'
        + (s.sub ? '<div class="vx-forwarding-step-sub">' + s.sub + '</div>' : '')
        + '</div></div>';
    });
    instrHtml += '</div>';
  }

  if (provider.codes) {
    var recommended = provider.codes.miss || null;
    if (recommended) instrHtml += makeRecommendedCode(recommended.code, recommended.label || 'Bei Nichtannahme');

    var secondaryKeys = Object.keys(provider.codes).filter(function(k){ return k !== 'off' && k !== 'miss'; });
    if (secondaryKeys.length) {
      instrHtml += '<div class="vx-forwarding-secondary"><details class="vx-forwarding-details"><summary><span>Weitere Optionen</span><i class="ph-bold ph-caret-down" aria-hidden="true"></i></summary><div class="vx-forwarding-details-body">';
      secondaryKeys.forEach(function(k) {
        var item = provider.codes[k];
        instrHtml += makeSecondaryCode(item.code, item.label);
      });
      instrHtml += '</div></details></div>';
    }
  }

  var instrTitle = shell.querySelector('#shell-instr-title');
  var instrBody = shell.querySelector('#shell-instr-body');
  if (instrTitle) instrTitle.textContent = provider.name || 'Anleitung';
  if (instrBody) instrBody.innerHTML = instrHtml;

  var deactCard = shell.querySelector('#shell-deactivation-card');
  var deactCodes = shell.querySelector('#shell-deactivation-codes');
  if (provider.codes && provider.codes.off && deactCard && deactCodes) {
    deactCodes.innerHTML = makeSecondaryCode(provider.codes.off.code, provider.codes.off.label || 'Weiterleitung aus');
    deactCard.style.display = 'block';
  } else if (deactCard) {
    deactCard.style.display = 'none';
  }
}

function vxCopyCode(btn) {
  var code = btn ? btn.getAttribute('data-code') : '';
  if (!code) return;
  if (navigator.clipboard) {
    navigator.clipboard.writeText(code).then(function() {
      if (typeof showToast === 'function') showToast('✓ Code kopiert.');
    });
  }
}

function vxOnboardingUpdateSteps() {
  const steps = {
    1: customerMeta.forwardingSetupCompleted === true,
    2: true, // Assistent intern verwaltet — immer als erledigt
    3: customerMeta.notificationMode !== 'none' && customerMeta.notificationMode !== undefined
  };

  let doneCount = 0;
  Object.entries(steps).forEach(function([num, done]) {
    const step = document.getElementById('vx-ob-step-' + num);
    const check = document.getElementById('vx-ob-check-' + num);
    const icon = document.getElementById('vx-ob-icon-' + num);
    if (done) {
      doneCount++;
      if (step) { step.style.borderColor = '#059669'; step.style.background = 'rgba(5,150,105,.04)'; }
      if (check) { check.style.background = '#059669'; check.style.borderColor = '#059669'; check.innerHTML = '<svg width="12" height="12" viewBox="0 0 12 12" fill="none"><path d="M2 6l3 3 5-5" stroke="#fff" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>'; }
      if (icon) icon.style.background = 'rgba(5,150,105,.1)';
    } else {
      if (step) { step.style.borderColor = 'var(--line)'; step.style.background = ''; }
      if (check) { check.style.background = ''; check.style.borderColor = 'var(--line)'; check.innerHTML = ''; }
      if (icon) icon.style.background = 'var(--blue-subtle)';
    }
  });

  // Progressbar: 2 sichtbare Schritte (Step 1 + Step 3)
  const visibleDone = (steps[1] ? 1 : 0) + (steps[3] ? 1 : 0);
  const bar = document.getElementById('vx-onboarding-progress');
  if (bar) bar.style.width = Math.round(visibleDone / 2 * 100) + '%';

  // Fertig wenn Steps 1 + 3 erledigt
  const complete = document.getElementById('vx-ob-complete');
  const finishBtn = document.getElementById('vx-ob-finish-btn');
  const allDone = steps[1] && steps[3];
  if (allDone) {
    if (complete) complete.style.display = 'block';
    if (finishBtn) finishBtn.textContent = 'Fertig ✓';
  } else {
    if (complete) complete.style.display = 'none';
    if (finishBtn) finishBtn.textContent = 'Los geht\'s →';
  }
}

function buildDailyReport(records, manualTasks) {
  var name = getAssistantName();
  var hour = new Date().getHours();
  var isNightTime = hour >= 0 && hour < 6; // 00:00-05:59 — zeige gestrigen Tag
  var todayFilter = function(r) {
    var ts = getRecordTimestamp(r);
    if (!ts) return false;
    if (isActivationCallCategory((r.fields||{}).category)) return false;
    if (isNightTime) {
      // Zeige gestrigen Kalendertag
      var d = new Date(ts);
      var yesterday = new Date(); yesterday.setDate(yesterday.getDate() - 1);
      var fmt = new Intl.DateTimeFormat('en-CA', { timeZone: DISPLAY_TIMEZONE, year:'numeric', month:'2-digit', day:'2-digit' });
      return fmt.format(d) === fmt.format(yesterday);
    }
    return isToday(ts);
  };
  var today = (records || []).filter(todayFilter);
  var totalToday = today.length;
  var dayLabel = isNightTime ? 'Gestern' : 'Heute';
  var callbacks = (records || []).filter(function(r) {
    var f = r.fields || {};
    return !isClosedStatus(f.dashboard_status) && (f.callback_requested === true || f.callback_requested === 'true');
  });
  var hotLeads = today.filter(function(r) {
    return String((r.fields||{}).lead_quality||'').toLowerCase() === 'hot';
  });
  // [PATCH 2026-05-13 v2.7]
  // Aufgaben offen = offene customer_tasks; Call-Signale nicht doppelt zählen.
  var totalOpen = (typeof getOpenTaskCount === 'function') ? getOpenTaskCount() : 0;

  if (typeof vxBuildDailyReportText === 'function') return vxBuildDailyReportText(records || [], manualTasks || []);

  if (totalToday === 0 && totalOpen === 0) {
    if (isNightTime) return 'Gute Nacht. ' + name + ' ist bereit für morgen.';
    return name + ' ist bereit und wartet auf den ersten Anruf. Alle Systeme laufen.';
  }
  var parts = [];
  if (totalToday === 1) parts.push(dayLabel + ' hat ' + name + ' 1 Anruf entgegengenommen.');
  else if (totalToday > 1) parts.push(dayLabel + ' hat ' + name + ' ' + totalToday + ' Anrufe entgegengenommen.');
  if (hotLeads.length > 0) {
    var topLead = hotLeads[0];
    var leadName = (topLead.fields||{}).caller_name || (topLead.fields||{}).company_name || '';
    if (leadName) parts.push(leadName + ' ist ein Hot Lead — nicht vergessen.');
    else parts.push(hotLeads.length + ' Hot Lead' + (hotLeads.length > 1 ? 's' : '') + ' heute.');
  }
  if (callbacks.length > 0) parts.push(callbacks.length + ' offene' + (callbacks.length === 1 ? 'r' : '') + ' Rückruf' + (callbacks.length === 1 ? '' : 'e') + '.');
  else if (totalOpen > 0) parts.push(totalOpen + ' offene Aufgabe' + (totalOpen === 1 ? '' : 'n') + '.');
  if (parts.length === 0) parts.push(name + ' hat alles im Griff.');
  return parts.join(' ');
}

// [PATCH 2026-05-13 v2.2 — Tagesbericht stabil 1x täglich]
// Vorher: sessionStorage Cache (verlor bei Tab-Schließen, lief in jeder Session neu)
//         Plus _aiReportGenerated als Session-Flag → bei Reload neu generiert
// Jetzt: localStorage mit Datum als Key → 1x pro Kalendertag generiert, bleibt stabil
//        Zeitstempel zeigt: "Stand: HH:MM"
//        Refresh-Button rechts oben für manuelle Neugenerierung
var _aiReportGenerated = false;
var _aiReportForceRefresh = false; // Wird auf true gesetzt wenn manuell neu generiert wird

async function generateAiDailyReport(records, manualTasks, lastSignInAt) {
  // Wenn bereits in dieser Session generiert UND nicht explizit Force-Refresh: skippen
  if (_aiReportGenerated && !_aiReportForceRefresh) return;
  _aiReportForceRefresh = false; // Reset für nächsten Aufruf

  // [PATCH 2026-05-13 v2.2 — Stabil 1x täglich via localStorage]
  // Cache-Key enthält das Tagesdatum (en-CA = YYYY-MM-DD), pro Voxera-Kunde isoliert
  var customerId = (typeof customerContext !== 'undefined' && customerContext && customerContext.customerId) ? customerContext.customerId : 'default';
  var today = new Date().toLocaleDateString('en-CA', { timeZone: 'Europe/Zurich' });
  var currentOpenTaskCount = (typeof getOpenTaskCount === 'function') ? getOpenTaskCount() : 0;
  var cacheKey = 'voxera_ai_report_v23_' + customerId + '_' + today + '_tasks' + currentOpenTaskCount;

  var deterministicReport = (typeof vxBuildDailyReportText === 'function') ? vxBuildDailyReportText(records || [], manualTasks || allManualTaskRecords || []) : buildDailyReport(records || [], manualTasks || []);
  var deterministicTime = new Date().toISOString();
  _showAiReport(deterministicReport || 'Tagesfokus konnte nicht erstellt werden.', false, deterministicTime);
  _aiReportGenerated = true;
  return;

  // Legacy KI-Pfad bleibt im Code, wird aber aktuell bewusst nicht verwendet.
  // Grund: Tagesfokus und KPIs dürfen keine abweichenden Zählungen erzeugen.
  _showAiReport('Wird geladen…', true);

  try {
    var cachedStr = localStorage.getItem(cacheKey);
    if (cachedStr) {
      var cached = JSON.parse(cachedStr);
      if (cached && cached.text && cached.generatedAt) {
        _showAiReport(cached.text, false, cached.generatedAt);
        _aiReportGenerated = true;
        return;
      }
    }
  } catch(e) { /* Cache kaputt → einfach neu generieren */ }

  var allRecords = records || [];

  // Nur offene Rückrufe
  var openCallbacks = allRecords.filter(function(r) {
    var f = r.fields || {};
    return !isClosedStatus(f.dashboard_status) &&
           (f.callback_requested === true || f.callback_requested === 'true');
  });

  // Nur offene Aufgaben — [PATCH 2026-05-13 v2 — Konsistenz]
  // Vorher: separates Zählen von openTasks + manualOpen → driftete von Hero und KPI-Strip
  // Jetzt: zentrale getOpenTaskCount() — dieselbe Quelle wie überall sonst
  var totalOpen = currentOpenTaskCount;

  // Dringend
  var dringend = allRecords.filter(function(r) {
    var f = r.fields || {};
    return !isClosedStatus(f.dashboard_status) &&
           (String(f.priority||'').toLowerCase() === 'high' ||
            String(f.lead_quality||'').toLowerCase() === 'hot');
  });

  var lastLogin = lastSignInAt ? new Date(lastSignInAt) : null;
  var lastLoginStr = lastLogin
    ? lastLogin.toLocaleDateString('de-CH', { weekday:'long', day:'numeric', month:'long', hour:'2-digit', timeZone:'Europe/Zurich', minute:'2-digit' })
    : 'zuletzt';

  var callbacksSummary = openCallbacks.slice(0, 5).map(function(r) {
    var f = r.fields || {};
    var name = f.caller_name || f.company_name || 'Unbekannt';
    var summary = f.summary || f.call_summary || '';
    return '- ' + name + (summary ? ': ' + String(summary).slice(0, 100) : '');
  }).join('\n');

  if (openCallbacks.length === 0 && totalOpen === 0) {
    var emptyText = 'Alles erledigt — keine offenen Aufgaben oder Rückrufe.';
    var nowIso = new Date().toISOString();
    try { vxSafeJsonLocalStorageSet(cacheKey, { text: emptyText, generatedAt: nowIso }, { maxBytes: 80000 }); } catch(e) {}
    _showAiReport(emptyText, false, nowIso);
    _aiReportGenerated = true;
    return;
  }

  var prompt = 'Du bist der persönliche Assistent eines KMU-Inhabers. Schreibe 2-3 Sätze, direkt und freundlich auf Deutsch (du-Form). Keine Floskeln, keine Begrüssung. Kein Markdown, keine Sternchen, keine Hashtags. Nur Fliesstext. function _showAiReport(text, isLoading, generatedAtIso) {
  // [PATCH 2026-05-13 v2.2 — Zeitstempel + Refresh-Button]
  // Vorher: Text wurde nur angezeigt, keine Indikation wann er generiert wurde
  // Jetzt: "Stand: HH:MM" rechts oben + Klick-Option zum Neugenerieren
  var reportBlock = document.getElementById('dash-daily-report');
  var reportText = document.getElementById('dash-daily-report-text');
  if (!reportBlock || !reportText) return;
  if (!text) { reportBlock.style.display = 'none'; return; }
  // [PATCH v2.19 — Markdown-Artefakte aus KI-Text entfernen]
  // Das KI-Modell gibt manchmal Markdown-Headings (# Aktuelle Situation) oder
  // fette/kursive Formatierungen zurück, die als Rohtext unschön aussehen.
  // Hier werden die häufigsten Markdown-Artefakte bereinigt.
  var cleanText = String(text)
    .replace(/^#{1,3}\s+/gm, '')       // ## Heading → leer
    .replace(/\*\*(.*?)\*\*/g, '$1')   // **fett** → fett
    .replace(/\*(.*?)\*/g, '$1')       // *kursiv* → kursiv
    .replace(/`([^`]+)`/g, '$1')       // `code` → code
    .replace(/\n{3,}/g, '\n\n')        // 3+ Zeilenumbrüche → max 2
    .trim();
  reportText.textContent = cleanText;
  reportText.style.color = isLoading ? 'var(--slate2)' : 'var(--ink2)';
  reportBlock.style.display = '';

  // Zeitstempel und Refresh-Button — nur wenn nicht Ladezustand
  var metaEl = document.getElementById('dash-daily-report-meta');
  if (!metaEl) {
    // Meta-Container existiert noch nicht → erstellen und einfügen
    metaEl = document.createElement('div');
    metaEl.id = 'dash-daily-report-meta';
    metaEl.style.cssText = 'display:flex;justify-content:flex-end;align-items:center;gap:8px;margin-top:8px;font-size:11px;color:var(--slate2);';
    reportText.parentNode.insertBefore(metaEl, reportText.nextSibling);
  }
  if (isLoading || !generatedAtIso) {
    metaEl.style.display = 'none';
  } else {
    var d = new Date(generatedAtIso);
    var hh = String(d.getHours()).padStart(2,'0');
    var mm = String(d.getMinutes()).padStart(2,'0');
    metaEl.innerHTML = '<span>Zuletzt aktualisiert: ' + hh + ':' + mm + '</span>' +
      '<button id="dash-daily-report-refresh" type="button" ' +
      'style="background:none;border:none;color:#1A6FE8;cursor:pointer;font-family:inherit;font-size:11px;padding:2px 6px;border-radius:4px;" ' +
      'onmouseover="this.style.background=\'rgba(26,111,232,0.08)\'" ' +
      'onmouseout="this.style.background=\'none\'">Neu generieren</button>' +
      '<span id="dash-daily-report-status" style="display:none;color:#94A3B8;"></span>';
    metaEl.style.display = 'flex';
    var refreshBtn = document.getElementById('dash-daily-report-refresh');
    var statusEl = document.getElementById('dash-daily-report-status');
    if (refreshBtn) {
      refreshBtn.onclick = async function() {
        if (refreshBtn.disabled) return;
        var oldLabel = refreshBtn.textContent;
        refreshBtn.disabled = true;
        refreshBtn.textContent = 'Generiere…';
        if (statusEl) { statusEl.style.display = 'none'; statusEl.textContent = ''; }
        _aiReportGenerated = false;
        _aiReportForceRefresh = true;
        try {
          await generateAiDailyReport(window.allRecords || allRecords, window.allManualTaskRecords || allManualTaskRecords, null);
          if (statusEl) { statusEl.textContent = 'Aktualisiert'; statusEl.style.color = '#64748B'; statusEl.style.display = 'inline'; }
        } catch(e) {
          if (statusEl) { statusEl.textContent = 'Aktualisierung fehlgeschlagen'; statusEl.style.color = '#DC2626'; statusEl.style.display = 'inline'; }
        } finally {
          refreshBtn.disabled = false;
          refreshBtn.textContent = oldLabel;
        }
      };
    }
  }

  /* phosphor: no init needed */
}

function getReachabilityStatus(records) {
  var calls = (records || []).filter(function(r) {
    return !isActivationCallCategory((r.fields||{}).category);
  });
  if (calls.length === 0) return { status: 'unbekannt', label: 'Noch kein Anruf', cls: 'gray', daysSince: null };
  calls.sort(function(a,b) {
    return new Date(getRecordTimestamp(b)) - new Date(getRecordTimestamp(a));
  });
  var last = new Date(getRecordTimestamp(calls[0]));
  var now = new Date();
  var diffMs = now - last;
  var diffDays = Math.floor(diffMs / (1000*60*60*24));
  if (diffDays === 0) return { status: 'aktiv', label: 'Aktiv · Letzter Anruf heute', cls: 'green', daysSince: 0 };
  if (diffDays === 1) return { status: 'aktiv', label: 'Aktiv · Letzter Anruf gestern', cls: 'green', daysSince: 1 };
  if (diffDays <= 7) return { status: 'unklar', label: 'Letzter Anruf vor ' + diffDays + ' Tagen', cls: 'amber', daysSince: diffDays };
  return { status: 'inaktiv', label: 'Kein Anruf seit ' + diffDays + ' Tagen', cls: 'red', daysSince: diffDays };
}

function isToday(raw) {
  if (!raw) return false;
  const d = parseFollowUpDate(raw) || new Date(raw);
  if (Number.isNaN(d.getTime())) return false;
  // Kalendertag — Mitternacht als Grenze (sauber und intuitiv)
  const fmt = new Intl.DateTimeFormat('en-CA', { timeZone: DISPLAY_TIMEZONE, year: 'numeric', month: '2-digit', day: '2-digit' });
  return fmt.format(d) === fmt.format(new Date());
}

function isThisWeek(raw) {
  if (!raw) return false;
  const d = parseFollowUpDate(raw) || new Date(raw), now = new Date();
  if (Number.isNaN(d.getTime())) return false;
  const start = new Date(now);
  // Monday-first week (Swiss / ISO 8601): Sun=0 → offset 6, Mon=1 → offset 0, …, Sat=6 → offset 5
  const dayOffset = now.getDay() === 0 ? 6 : now.getDay() - 1;
  start.setDate(now.getDate() - dayOffset);
  start.setHours(0,0,0,0);
  return d >= start;
}

function fmtDateOnly(raw) {
  if (!raw) return '–';
  const localParts = parseLocalDateTimeParts(raw);
  if (localParts) return formatPartsDate(localParts);
  const d = new Date(raw);
  if (Number.isNaN(d.getTime())) return '–';
  return formatDateLongCH(d);
}

function fmtDateTimeLocal(raw) {
  const localValue = normalizeFollowUpLocalValue(raw);
  if (localValue) return localValue;
  if (!raw) return '';
  try {
    const localParts = parseLocalDateTimeParts(raw);
    if (localParts) return `${localParts.year}-${pad2(localParts.month)}-${pad2(localParts.day)}T${pad2(localParts.hour)}:${pad2(localParts.minute)}`;
    const d = parseFollowUpDate(raw) || new Date(raw);
    if (Number.isNaN(d.getTime())) return '';
    return d.getFullYear()+'-'+pad2(d.getMonth()+1)+'-'+pad2(d.getDate())+'T'+pad2(d.getHours())+':'+pad2(d.getMinutes());
  } catch(e) { return ''; }
}

function formatDurationForDisplay(durationSeconds) {
  if (!durationSeconds || durationSeconds <= 0) return '–';
  var secs = Number(durationSeconds);
  if (Number.isNaN(secs)) return '–';
  if (secs < 60) return Math.round(secs) + ' Sek.';
  return Math.round(secs / 60) + ' Min.';
}

function resolveCallEndLabel(fields) {
  // Bug-Fix (29.04.2026): Vorher zeigte das Badge "Abgeschlossen" für JEDEN
  // erfolgreichen Anruf – das war redundant, weil 95% der Anrufe erfolgreich sind.
  // Das Workflow-Status-Feld (dashboard_status: Neu / In Bearbeitung / Abgeschlossen)
  // ist davon getrennt und wird über statusChip() angezeigt.
  //
  // Neue Logik: Badge erscheint NUR in Ausnahme-Fällen (Anrufer-seitige Probleme):
  // - "Verpasst": Anrufer hat aufgelegt bevor Bot abnehmen konnte
  // - "Abgebrochen": Anrufer hat zu früh aufgelegt (< 5 Sek, kein Inhalt)
  // - "Fehler": Technischer Fehler
  // Erfolgreiche Anrufe bekommen KEIN Badge → ruhigeres UI.
  if (!fields) return null;
  const status = String(fields.status || fields.call_status || fields.call_result || '').trim().toLowerCase();
  const secs = Number(fields.duration_seconds || 0);
  const hasSummary = !!(fields.call_summary && !isDashLikePlaceholder(String(fields.call_summary)));

  // Technischer Fehler
  if (status === 'failed' || status === 'error') {
    return { label: 'Fehler', cls: 'chip-hot' };
  }

  // Anrufer hat nicht abgenommen / Leitung besetzt / kein Bot-Pickup
  if (status === 'no-answer' || status === 'no_answer' || status === 'busy') {
    return { label: 'Verpasst', cls: 'chip-meta' };
  }

  // Explizit als abgebrochen markiert ODER zu kurz für sinnvolles Gespräch
  if (status === 'abandoned') {
    return { label: 'Abgebrochen', cls: 'chip-meta' };
  }
  if (secs > 0 && secs < 5 && !hasSummary) {
    return { label: 'Abgebrochen', cls: 'chip-meta' };
  }
  // status=started ohne Duration und ohne Summary → vor Handoff abgebrochen
  if (status === 'started' && secs === 0 && !hasSummary) {
    return { label: 'Abgebrochen', cls: 'chip-meta' };
  }

  // Default: Erfolgreicher Anruf → KEIN Badge (Workflow-Status reicht)
  return null;
}

function monthsSince(raw) {
  if (!raw) return 0;
  const d = new Date(raw);
  if (Number.isNaN(d.getTime())) return 0;
  const now = new Date();
  let months = (now.getFullYear() - d.getFullYear()) * 12 + (now.getMonth() - d.getMonth());
  if (now.getDate() < d.getDate()) months -= 1;
  return Math.max(1, months + 1);
}

function getBilledMinutesForRecord(record) {

function toValidNumber(value) {

async function loadCustomerContractMeta(customerId, subscriptionId) {
  customerContractMeta = null;
  customerContractState = {
    status: 'unknown',
    resolution: 'loading',
    hasContract: false,
    hasActiveContract: false,
    latestContract: null,
    effectiveContract: null
  };
  if (!_sb || !customerId) return null;

  // Source-of-truth for customer contract state is the server-side resolver
  // (same auth/customer mapping + contract selection logic as dashboard functions).
  try {
    const json = await callDashboardFunction('customer-contract-state', {});
    const serverState = (json && json.state) || null;
    if (serverState && typeof serverState === 'object') {
      const contractRows = Array.isArray(serverState.contracts) ? serverState.contracts : [];
      if (contractRows.length) {
        customerContractState = deriveContractState(contractRows);
      } else {
        customerContractState = {
          status: normalizeContractStatus(serverState.status) || 'unknown',
          resolution: 'resolved',
          hasContract: serverState.hasContract === true,
          hasActiveContract: serverState.hasActiveContract === true,
          latestContract: serverState.latestContract || null,
          effectiveContract: serverState.effectiveContract || null
        };
      }
      customerContractState.resolution = 'resolved';
      customerContractState.hasActiveContract = resolveHasActiveContractFromState(customerContractState);
      customerContractMeta = customerContractState.effectiveContract || customerContractState.latestContract || null;
      renderVertrag();
      vxDebugInfo('[plan-card-debug] contract-state-source', {
        serverState: {
          status: serverState.status,
          hasContract: serverState.hasContract,
          hasActiveContract: serverState.hasActiveContract,
          latestContract: serverState.latestContract || null,
          effectiveContract: serverState.effectiveContract || null
        },
        finalContractState: customerContractState,
        finalCustomerContractMeta: customerContractMeta
      });
      return customerContractMeta;
    }
  } catch (e) {
    console.warn('[contract-meta] function lookup failed, fallback to direct read', e);
  }

  // Fallback for older/local environments without deployed function.
  try {
    let contractsQuery = _sb
      .from('contracts')
      .select('*')
      .order('updated_at', { ascending: false })
      .order('created_at', { ascending: false })
      .limit(50);
    const normalizedSubscriptionId = String(subscriptionId || '').trim();
    if (normalizedSubscriptionId) {
      contractsQuery = contractsQuery.or(`customer_id.eq.${customerId},subscription_id.eq.${normalizedSubscriptionId}`);
    } else {
      contractsQuery = contractsQuery.eq('customer_id', customerId);
    }
    const { data, error } = await contractsQuery;
    if (error) return null;
    customerContractState = deriveContractState(data || []);
    customerContractState.resolution = 'resolved';
    customerContractMeta = customerContractState.effectiveContract || null;
    renderVertrag();
    return customerContractMeta;
  } catch (e) {
    console.warn('[contract-meta] load failed', e);
    customerContractState = {
      status: 'unknown',
      resolution: 'error',
      hasContract: false,
      hasActiveContract: false,
      latestContract: null,
      effectiveContract: null
    };
    return null;
  }
}

var _renderUsageLastRun = 0;
var _RENDER_USAGE_MIN_INTERVAL_MS = 300;
function renderUsage(records) {
  var _ts = Date.now();
  if (_ts - _renderUsageLastRun < _RENDER_USAGE_MIN_INTERVAL_MS) return;
  _renderUsageLastRun = _ts;
  const planCfg = getPlanConfigEntryForPlan(customerMeta.plan) || PLAN_CONFIG['Business'] || {monthlyPrice:0,minutes:0,extraRate:0};
  const cfg = {
    monthlyPrice: customerMeta.contractMonthlyPrice != null ? customerMeta.contractMonthlyPrice : Number(planCfg.monthlyPrice || 0),
    minutes: (customerMeta.contractIncludedMinutes != null ? customerMeta.contractIncludedMinutes : Number(planCfg.minutes || 0)) + Number(customerMeta.extraMinutes || 0),
    extraRate: customerMeta.contractExtraRate != null ? customerMeta.contractExtraRate : Number(planCfg.extraRate || 0)
  };
  const now = new Date();

  // Compute billing period start — contract anniversary takes priority
  const _ruContractRow = (typeof selectCommercialContractRow === 'function') ? selectCommercialContractRow() : null;
  const _ruStartDate = (customerMeta.startDate && customerMeta.startDate !== '') 
    ? customerMeta.startDate 
    : (_ruContractRow && _ruContractRow.start_date) || '';
  let periodStart = null;
  if (_ruStartDate) {
    const startDate = new Date(_ruStartDate);
    if (!Number.isNaN(startDate.getTime())) {
      const dayOfMonth = startDate.getDate();
      periodStart = new Date(now.getFullYear(), now.getMonth(), dayOfMonth, 0, 0, 0, 0);
      if (periodStart > now) {
        periodStart = new Date(now.getFullYear(), now.getMonth() - 1, dayOfMonth, 0, 0, 0, 0);
      }
      if (periodStart < startDate) periodStart = startDate;
    }
  }

  const monthRecords = records.filter(r => {
    // Filter by current customer_id
    if (r.fields && r.fields.customer_id && customerMeta.customerId) {
      const recordCustomerId = String(r.fields.customer_id).trim();
      const currentCustomerId = String(customerMeta.customerId).trim();
      if (recordCustomerId !== currentCustomerId) return false;
    }
    // Filter by billing period using canonical record timestamp
    const raw = getRecordTimestamp(r);
    if (!raw) return false;
    const d = new Date(raw);
    return !Number.isNaN(d.getTime()) && d >= periodStart && d <= now;
  });
  const used = Math.round(monthRecords.reduce((sum, r) => {
    const billed = getBilledMinutesForRecord(r);
    // Wenn duration_seconds fehlt, estimateMinutesForRecord als Fallback verwenden
    return sum + (billed > 0 ? billed : estimateMinutesForRecord(r));
  }, 0));
  const included = cfg.minutes || 0;
  const over = Math.max(0, used - included);
  const pct = included > 0 ? Math.min(100, Math.round((used / included) * 100)) : 0;
  const extraCost = (over * (cfg.extraRate || 0)).toFixed(2);

  const set = (id, val) => { const el=document.getElementById(id); if (el) el.textContent = val; };
  const meter = document.getElementById('usage-meter-fill');
  if (meter) meter.style.width = `${pct}%`;
  const contractStatusLabel = getContractStatusLabel(customerMeta.contractStatus);
  const planLabel = formatPlanLabel(customerMeta.plan);
  const hasActiveContractForPlanCard = customerMeta.hasActiveContract === true;
  const hasKnownInactiveContractForPlanCard = hasKnownInactiveContract();
  const activeSuffix = hasKnownInactiveContractForPlanCard ? ' · Kein aktiver Vertrag' : '';
  set('usage-plan-sub', `${planLabel} · CHF ${(cfg.monthlyPrice||0)} / Mt · ${cfg.minutes||0} Min. inkl. · Vertragsstatus: ${contractStatusLabel}${activeSuffix}`);
  set('usage-month-minutes', `${used} Min`);
  // Dashboard KPI
  set('kpi-usage-min', used);
  const usageSub = document.getElementById('kpi-usage-sub');
  if (usageSub) usageSub.textContent = included > 0 ? `von ${included} Min. (${pct}%)` : 'Min. diesen Monat';
  set('usage-included', `${included} Min`);
  set('usage-meter-label', `${used} von ${included} Minuten genutzt`);
  set('usage-extra-label', over > 0 ? `${over} Zusatzminuten · ${extraCost} CHF geschätzt` : 'Keine Zusatzkosten');
  // Auswertung-Tab Nutzung
  var auPlanName = formatPlanLabel(customerMeta.plan);
  set('au-plan-name', auPlanName);
  set('au-used-min', used + ' Min.');
  set('au-included-min', included + ' Min.');
  set('au-usage-pct', pct + '%');
  set('au-meter-label', used + ' von ' + included + ' Minuten');
  set('au-extra-label', over > 0 ? over + ' Zusatzminuten · ' + extraCost + ' CHF' : '');
  var auBar = document.getElementById('au-usage-bar');
  if (auBar) auBar.style.width = pct + '%';
  var auChip = document.getElementById('au-plan-chip');
  if (auChip) {
    auChip.textContent = over > 0 ? 'Upgrade empfohlen' : pct >= 80 ? 'Nahe am Limit' : 'Plan aktiv';
    auChip.className = 'chip ' + (over > 0 ? 'chip-hot' : pct >= 80 ? 'chip-working' : 'chip-neu');
  }
  const chip = document.getElementById('usage-status-chip');
  if (chip) {
    if (hasKnownInactiveContractForPlanCard) {
      chip.textContent = 'Vertrag inaktiv';
      chip.className = 'chip chip-hot';
    } else {
      chip.textContent = over > 0 ? 'Upgrade empfohlen' : pct >= 80 ? 'Nahe am Limit' : 'Plan aktiv';
      chip.className = 'chip ' + (over > 0 ? 'chip-hot' : pct >= 80 ? 'chip-working' : 'chip-neu');
    }
  }
  vxDebugInfo('[plan-card-debug] render-usage-plan-card', {
    contractStatusLabel,
    hasActiveContractForPlanCard,
    hasKnownInactiveContractForPlanCard,
    activeSuffix,
    usageChipText: chip ? chip.textContent : null,
    usageChipClass: chip ? chip.className : null,
    customerContractMeta,
    customerContractState
  });
  const upTitle = document.getElementById('upgrade-primary-title');
  const upCopy = document.getElementById('upgrade-primary-copy');
  const addCopy = document.getElementById('upgrade-addon-copy');
  const currentCfg = cfg;
  const upgradeState = getUpgradeModalState();
  const nextPlan = upgradeState.upgrades[0] ? upgradeState.upgrades[0].label : null;
  if (upTitle) upTitle.textContent = nextPlan ? (over > 0 || pct >= 85 ? `Upgrade auf ${nextPlan}` : `Mehr Leistung mit ${nextPlan}`) : 'Aktuell kein höheres Paket';
  if (upCopy) upCopy.textContent = nextPlan
    ? (over > 0 ? `Sie sind über Ihrem Inklusivvolumen. Ein Upgrade auf ${nextPlan} reduziert Zusatzkosten und schafft Reserve.` : pct >= 80 ? `Sie nutzen Ihr Paket bereits stark aus. ${nextPlan} bietet mehr Minuten und Reserve für Spitzenzeiten.` : `Ihr aktueller Plan läuft stabil. Ein Upgrade auf ${nextPlan} lohnt sich bei wachsendem Anrufvolumen.`)
    : 'Für Ihren aktuellen Plan ist momentan kein höheres Paket verfügbar.';
  if (addCopy) addCopy.textContent = over > 0 ? `Aktuell ca. ${over} Zusatzminuten über dem Paket. Zusatzvolumen kann kurzfristig ergänzt werden.` : `Zusatzminuten sind ideal für Kampagnen, saisonale Peaks oder kurze Lastspitzen.`;
}

async function applyStatusTransition(recordId, fromStatus, toStatus) {
  const target = normalizeStatus(toStatus);
  await apiUpdateCallStatus(recordId, target);

  // PATCH v3.3 — Lifecycle-Zeitstempel zusätzlich speichern.
  // call-update-status bleibt die autoritative Status-Function; direkte Supabase-Updates
  // sind bewusst best-effort, damit ältere RLS/Function-Umgebungen nicht brechen.
  var lifecyclePatch = { updated_at: new Date().toISOString() };
  if (target === CALL_STATUS.CLOSED) {
    lifecyclePatch.completed_at = new Date().toISOString();
    lifecyclePatch.archived_at = null;
  } else if (target === CALL_STATUS.ARCHIVED) {
    lifecyclePatch.archived_at = new Date().toISOString();
  } else if (target === CALL_STATUS.NEW) {
    lifecyclePatch.completed_at = null;
    lifecyclePatch.archived_at = null;
  }
  await vxBestEffortPatchCallLifecycle(recordId, lifecyclePatch);
}


async function vxBestEffortPatchCallLifecycle(recordId, fields) {
  if (!recordId || !fields || typeof fields !== 'object') return null;
  try {
    var sb = (typeof getSupabaseAuthClient === 'function') ? getSupabaseAuthClient() : _sb;
    if (!sb || !CALLS_TABLE) return null;
    var res = await sb.from(CALLS_TABLE).update(fields).eq('id', recordId).select('*').maybeSingle();
    if (res && res.error) throw res.error;
    var updated = res && res.data ? res.data : null;
    var rec = (allRecords || []).find(function(r){ return String(r.id) === String(recordId); });
    if (rec && rec.fields) Object.assign(rec.fields, fields, updated || {});
    return updated;
  } catch (err) {
    console.warn('[voxera] lifecycle timestamp update skipped', err && (err.message || err));
    var local = (allRecords || []).find(function(r){ return String(r.id) === String(recordId); });
    if (local && local.fields) Object.assign(local.fields, fields);
    return null;
  }
}

async function vxBestEffortPatchTaskLifecycle(recordId, fields) {
  if (!recordId || !fields || typeof fields !== 'object') return null;
  var rec = (typeof vxResolveDprRecord === 'function') ? vxResolveDprRecord(recordId) : null;
  if (!rec || !rec.fields) return null;
  var entityId = String(rec.fields.task_entity_id || '').trim() || String(recordId || '').replace(/^case:/,'');
  if (!entityId) return null;
  try {
    var sb = (typeof getSupabaseAuthClient === 'function') ? getSupabaseAuthClient() : _sb;
    if (!sb || !CASES_TABLE) return null;
    var res = await sb.from(CASES_TABLE).update(fields).eq('id', entityId).select('*').maybeSingle();
    if (res && res.error) throw res.error;
    var updatedRaw = res && res.data ? res.data : null;
    var updatedRec = updatedRaw ? mapCaseToTaskRecord(updatedRaw) : null;
    var idx = (allManualTaskRecords || []).findIndex(function(r){
      var f = (r && r.fields) || {};
      return String(r.id) === String(recordId) || String(f.task_entity_id || '') === entityId;
    });
    if (updatedRec && idx >= 0) allManualTaskRecords[idx] = updatedRec;
    else if (idx >= 0 && allManualTaskRecords[idx] && allManualTaskRecords[idx].fields) Object.assign(allManualTaskRecords[idx].fields, fields);
    return updatedRec || updatedRaw;
  } catch (err) {
    console.warn('[voxera] task lifecycle timestamp update skipped', err && (err.message || err));
    if (rec && rec.fields) Object.assign(rec.fields, fields);
    return null;
  }
}

async function apiPatchOptional(recordId, preferredFields, fallbackFields) {
  try {
    return await apiPatch(recordId, preferredFields);
  } catch (err) {
    const msg = (err && err.message ? String(err.message) : '').toLowerCase();
    const retryable = msg.includes('422') || msg.includes('unknown field') || msg.includes('invalid') || msg.includes('cannot parse') || msg.includes('field');
    if (!fallbackFields || !retryable) throw err;
    return await apiPatch(recordId, fallbackFields);
  }
}

// Store last incoming record IDs for banner click navigation
var _incomingBannerRecordIds = [];

function flagIncomingRecords(records) {
  const ids = new Set(records.map(r => r.id));
  if (!hasLoadedInitialRecords) {
    // Baseline nur setzen wenn wirklich Daten vorhanden – sonst wird nach
    // Login alles als "neu" markiert (leere Pre-Auth-Antwort als Baseline).
    if (records.length === 0) return;
    seenRecordIds = ids;
    hasLoadedInitialRecords = true;
    return;
  }

  const newIds = records.map(r => r.id).filter(id => !seenRecordIds.has(id));
  seenRecordIds = ids;

  if (!newIds.length) return;

  // Filter out test calls from banner notifications
  const visibleNewIds = newIds.filter(function(id) {
    var rec = records.find(function(r) { return r.id === id; });
    return rec && !isActivationCallCategory((rec.fields || {}).category);
  });

  if (!visibleNewIds.length) return;

  visibleNewIds.forEach(id => recentIncomingIds.add(id));
  _incomingBannerRecordIds = visibleNewIds;

  // Collect call data for the banner
  var newRecords = visibleNewIds.map(id => records.find(r => r.id === id)).filter(Boolean);
  showIncomingBanner(visibleNewIds.length, newRecords);

  setTimeout(() => {
    visibleNewIds.forEach(id => recentIncomingIds.delete(id));
    renderDashboard(allRecords, allManualTaskRecords);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    renderArchiv(allRecords);
    vxUpdateRefreshPolling(allRecords);
  }, 18000);
}

var TEMP_CALL_POLL_INTERVAL_MS = 9000;
var CALL_DISPLAY_LIVE_STABLE_MS = 12000;
var CALL_DISPLAY_TEMP_STALE_MS = 30 * 60 * 1000;
var CALL_DISPLAY_CACHE_TTL_MS = 60 * 60 * 1000;

function vxGetDefaultCallDisplayState() {
  return {
    state: 'ready',
    label: 'Neu',
    tone: 'neutral',
    title: 'Anfrage',
    description: 'Zusammenfassung und Details sind verfügbar.',
    isTemporary: false
  };
}

function vxGetCallDisplayState(callOrFields) {
  var fallback = vxGetDefaultCallDisplayState();
  try {
    if (typeof window !== 'undefined') {
      window._vxCallDisplayStateCache = window._vxCallDisplayStateCache || {};
    }
    var cache = (typeof window !== 'undefined' && window._vxCallDisplayStateCache) ? window._vxCallDisplayStateCache : {};
    var now = Date.now();
    Object.keys(cache).forEach(function(key) {
      var entry = cache[key];
      if (!entry) return;
      if ((now - Number(entry.lastSeenAt || 0)) > CALL_DISPLAY_CACHE_TTL_MS) delete cache[key];
    });

    if (!callOrFields) return fallback;
    var src = (callOrFields && callOrFields.fields) ? callOrFields.fields : callOrFields;
    if (!src || typeof src !== 'object') return fallback;

    // Manual tasks may pass through generic render paths. They are not live-call records.
    if (typeof isManualTaskRecord === 'function' && isManualTaskRecord(callOrFields)) return fallback;
    var liveStatus = String(src.live_status || '').trim().toLowerCase();
    var summary = (typeof vxGetCallSummary === 'function') ? vxGetCallSummary(src) : String(src.call_summary || src.summary || src.call_summary_short || src.ai_summary || src.analysis_summary || '').trim();
    var transcript = (typeof vxGetCallTranscript === 'function') ? vxGetCallTranscript(src) : String(src.transcript || src.transcription || '').trim();
    var hasSummary = !!summary;
    var hasTranscript = !!transcript;
    var hasDuration = src.duration_seconds !== null && src.duration_seconds !== undefined && !Number.isNaN(Number(src.duration_seconds));
    var hasReadySignals = hasSummary || (hasTranscript && hasDuration);
    var hasCompletedLikeState = ['completed', 'archived'].includes(liveStatus);
    var isLiveStatus = ['incoming','active','live','ringing'].includes(liveStatus);
    var hasDurationReadySignal = hasDuration && !isLiveStatus;
    var hasStrongReadySignals = hasReadySignals || hasCompletedLikeState || hasDurationReadySignal;
    var tsRaw = src.updated_at || src.created_at;
    var ts = (tsRaw && typeof parseCallTimestamp === 'function') ? parseCallTimestamp(tsRaw) : (tsRaw ? new Date(tsRaw) : null);
    var ageMs = ts && !Number.isNaN(ts.getTime()) ? (Date.now() - ts.getTime()) : Number.POSITIVE_INFINITY;
    var isProcessingStatus = ['analyzing','processing'].includes(liveStatus);
    var callId = String((callOrFields && (callOrFields.id || callOrFields.call_id)) || src.id || src.call_id || '').trim();
    var cacheEntry = callId ? (cache[callId] || {}) : null;

    // Hard ready lock: completed or final duration outside explicit live states must never render as temporary.
    if (hasStrongReadySignals) {
      if (cacheEntry) {
        cacheEntry.state = 'ready';
        cacheEntry.readyLocked = true;
        cacheEntry.firstSeenAt = cacheEntry.firstSeenAt || now;
        cacheEntry.lastSeenAt = now;
        cacheEntry.stableUntil = now;
        cache[callId] = cacheEntry;
      }
      return fallback;
    }

    if (cacheEntry && cacheEntry.readyLocked) return fallback;

    if (isLiveStatus) {
      if (cacheEntry) {
        cacheEntry.state = 'live';
        cacheEntry.firstSeenAt = cacheEntry.firstSeenAt || now;
        cacheEntry.lastSeenAt = now;
        cacheEntry.stableUntil = now + CALL_DISPLAY_LIVE_STABLE_MS;
        cache[callId] = cacheEntry;
      }
      return { state:'live', label:'LIVE', tone:'danger', title:'Live-Anruf', description:'Lara nimmt den Anruf gerade entgegen. Die Zusammenfassung erscheint nach Gesprächsende.', isTemporary:true };
    }

    if (cacheEntry && cacheEntry.state === 'live' && Number(cacheEntry.stableUntil || 0) > now) {
      cacheEntry.lastSeenAt = now;
      cache[callId] = cacheEntry;
      return { state:'live', label:'LIVE', tone:'danger', title:'Live-Anruf', description:'Lara nimmt den Anruf gerade entgegen. Die Zusammenfassung erscheint nach Gesprächsende.', isTemporary:true };
    }

    if (!hasCompletedLikeState && !hasDuration && ageMs > CALL_DISPLAY_TEMP_STALE_MS && !hasSummary && !hasTranscript) {
      if (cacheEntry) {
        cacheEntry.state = 'checking';
        cacheEntry.firstSeenAt = cacheEntry.firstSeenAt || now;
        cacheEntry.lastSeenAt = now;
        cache[callId] = cacheEntry;
      }
      return { state:'checking', label:'Wird geprüft', tone:'neutral', title:'Anrufdaten werden geprüft', description:'Die Zusammenfassung erscheint, sobald die Verarbeitung abgeschlossen ist.', isTemporary:true };
    }

    if (!hasCompletedLikeState && !hasDuration && (isProcessingStatus || (cacheEntry && cacheEntry.state === 'live') || ageMs <= CALL_DISPLAY_TEMP_STALE_MS)) {
      if (cacheEntry) {
        cacheEntry.state = 'processing';
        cacheEntry.firstSeenAt = cacheEntry.firstSeenAt || now;
        cacheEntry.lastSeenAt = now;
        cache[callId] = cacheEntry;
      }
      return { state:'processing', label:'Wird verarbeitet', tone:'blue', title:'Anruf wird verarbeitet', description:'Zusammenfassung und Transkript werden vorbereitet.', isTemporary:true };
    }
    return fallback;
  } catch (e) {
    return fallback;
  }
}

function getCallDisplayState(record) {

  var f = (record && record.fields) || {};
  var liveStatus = String(f.live_status || '').toLowerCase();
  var hasCallSummary = !!String(f.call_summary || '').trim();
  var hasCallSummaryShort = !!String(f.call_summary_short || '').trim();
  var hasTranscript = !!String(f.transcript || '').trim();
  var hasDuration = f.duration_seconds !== null && f.duration_seconds !== undefined;
  var hasFinalContent = hasCallSummary || hasCallSummaryShort || hasTranscript || hasDuration;
  var isCompleted = liveStatus === 'completed' || hasFinalContent;
  var isFailed = liveStatus === 'failed';
  var durationValue = Number(f.duration_seconds);
  var isShortDuration = !hasDuration || Number.isNaN(durationValue) || durationValue < 8;
  var isShortAbandoned = isFailed && !hasCallSummary && !hasCallSummaryShort && !hasTranscript && isShortDuration;
  var isLiveStatus = liveStatus === 'incoming' || liveStatus === 'active';
  var isAnalyzing = liveStatus === 'analyzing';
  var isTerminalStatus = CALL_LIVE_TERMINAL_STATES.has(liveStatus);
  var ts = getRecordTimestamp(record);
  var parsed = ts ? parseCallTimestamp(ts) : null;
  var updatedAtMs = parsed && !Number.isNaN(parsed.getTime()) ? parsed.getTime() : 0;
  var nowMs = Date.now();
  var liveThresholdMs = liveStatus === 'incoming' ? CALL_LIVE_INCOMING_FRESHNESS_MS : CALL_LIVE_ACTIVE_FRESHNESS_MS;
  var recordId = String((record && record.id) || '').trim();
  var isFresh = updatedAtMs > 0 && (nowMs - updatedAtMs) <= liveThresholdMs;

  var uiState = (typeof vxGetCallDisplayState === 'function' ? vxGetCallDisplayState(record) : null) || vxGetDefaultCallDisplayState();
  if (uiState.state === 'checking') return 'checking';
  if (uiState.state === 'ready') return 'final';

  if (isLiveStatus && !isFresh && recordId) {
    _staleLiveRecordIds.add(recordId);
    _staleLiveRecordUpdatedAtMs.set(recordId, updatedAtMs);
  }

  if (isTerminalStatus) return 'final';
  if (isShortAbandoned) return 'short_abandoned';
  if (isCompleted) {
    if (recordId) {
      _staleLiveRecordIds.delete(recordId);
      _staleLiveRecordUpdatedAtMs.delete(recordId);
    }
    return 'final';
  }
  if (isAnalyzing) {
    if (recordId) {
      _staleLiveRecordIds.delete(recordId);
      _staleLiveRecordUpdatedAtMs.delete(recordId);
    }
    return 'processing';
  }
  if (isLiveStatus && isFresh && recordId && _staleLiveRecordIds.has(recordId)) {
    var staleTs = Number(_staleLiveRecordUpdatedAtMs.get(recordId) || 0);
    if (updatedAtMs > staleTs) {
      _staleLiveRecordIds.delete(recordId);
      _staleLiveRecordUpdatedAtMs.delete(recordId);
      return 'live';
    }
    return 'processing';
  }
  if (isLiveStatus && isFresh) return 'live';
  if (isLiveStatus && !isFresh) return 'processing';
  if (isFailed && !hasFinalContent) return 'short_abandoned';
  if (!hasFinalContent) return 'processing';
  return 'final';
}

function pickLiveCall(records) {
  if (!Array.isArray(records) || !records.length) return null;
  var live = records.filter(function(r) {
    var f = r.fields || {};
    if (isActivationCallCategory(f.category)) return false;
    return getCallDisplayState(r) === 'live';
  });
  if (!live.length) return null;
  // Newest first
  live.sort(function(a, b) {
    var pa = (typeof parseCallTimestamp === 'function') ? parseCallTimestamp((a.fields || {}).created_at) : null;
    var pb = (typeof parseCallTimestamp === 'function') ? parseCallTimestamp((b.fields || {}).created_at) : null;
    var ta = pa ? pa.getTime() : 0;
    var tb = pb ? pb.getTime() : 0;
    return tb - ta;
  });
  return live[0];
}

function buildLiveCallMatcher(liveRecord) {
  if (!liveRecord || !liveRecord.fields) return null;
  var liveFields = liveRecord.fields || {};
  var liveId = String(liveRecord.id || '').trim();
  var livePhone = String(liveFields.caller_phone || liveFields.phone_number || '').trim();
  var liveCreated = parseCallTimestamp(liveFields.created_at);
  var liveCreatedMs = liveCreated && !Number.isNaN(liveCreated.getTime()) ? liveCreated.getTime() : null;
  return function(record) {
    if (!record || !record.fields) return false;
    if (liveId && String(record.id || '').trim() === liveId) return true;
    var f = record.fields || {};
    if (getCallDisplayState(record) !== 'live') return false;
    var recordPhone = String(f.caller_phone || f.phone_number || '').trim();
    var phoneMatch = !!(livePhone && recordPhone && livePhone === recordPhone);
    if (!phoneMatch) return false;
    var recordCreated = parseCallTimestamp(f.created_at);
    var recordCreatedMs = recordCreated && !Number.isNaN(recordCreated.getTime()) ? recordCreated.getTime() : null;
    if (liveCreatedMs === null || recordCreatedMs === null) return false;
    return Math.abs(recordCreatedMs - liveCreatedMs) <= 2 * 60 * 1000;
  };
}

function updateLiveHero(records) {
  var source = Array.isArray(records) ? records : (allRecords || []);
  var rec = source
    .filter(function(r) {
      if (!r || !r.fields) return false;
      var f = r.fields || {};
      if (isActivationCallCategory(f.category)) return false;
      var liveStatus = String(f.live_status || '').toLowerCase();
      var isLiveCandidate = liveStatus === 'incoming' || liveStatus === 'active';
      return isLiveCandidate && !CALL_LIVE_TERMINAL_STATES.has(liveStatus);
    })
    .sort(function(a, b) {
      var pa = (typeof parseCallTimestamp === 'function') ? parseCallTimestamp((a.fields || {}).created_at) : null;
      var pb = (typeof parseCallTimestamp === 'function') ? parseCallTimestamp((b.fields || {}).created_at) : null;
      var ta = pa ? pa.getTime() : 0;
      var tb = pb ? pb.getTime() : 0;
      return tb - ta;
    })[0] || null;
  var priorityList = document.getElementById('dash-priority-list');

  // Alten Live-Row entfernen
  var existing = document.getElementById('live-call-row');
  if (existing) existing.remove();

  if (!rec) {
    if (_liveHeroTimer) { clearInterval(_liveHeroTimer); _liveHeroTimer = null; }
    _liveHeroCurrentId = null;
    return;
  }

  var f = rec.fields || {};
  var caller = (typeof getCallDisplayName === 'function' ? getCallDisplayName(f) : null)
               || f.caller_name || f.phone_number || f.caller_phone || 'Unbekannte Nummer';
  var company = f.company_name || '';

  // Live-Row erstellen — v3.0 Design
  var row = document.createElement('div');
  row.id = 'live-call-row';
  row.dataset.recordId = rec.id;
  row.innerHTML = [
    '<div class="vx-live-avatar-wrap">',
      '<div class="vx-live-pulse"></div>',
      '<div class="vx-live-avatar">' + (caller.split(' ').map(function(w){return w[0]||'';}).join('').slice(0,2).toUpperCase()||'?') + '</div>',
    '</div>',
    '<div class="vx-live-body">',
      '<div class="vx-live-name">' + (company ? _esc(company) : _esc(caller)) + '</div>',
      '<div class="vx-live-sub">',
        '<span class="vx-live-sub-dot"></span>',
        'Anruf läuft gerade · wird verarbeitet',
      '</div>',
    '</div>',
    '<div class="vx-live-badge">LIVE</div>'
  ].join('');
  row.onclick = function(e) {
    if (e && e.preventDefault) e.preventDefault();
    if (e && e.stopPropagation) e.stopPropagation();
    // Live calls are intentionally excluded from the Anfragen worklist until the call is finalized.
    // Keep the user on Heute and avoid navigating to an empty filtered/detail state.
    if (typeof vxToast === 'function') {
      vxToast({ type:'call', title:'Live-Anruf läuft', sub:'Details erscheinen nach Gesprächsende automatisch in den Anfragen.' });
    } else if (typeof toast === 'function') {
      toast('Live-Anruf läuft. Details erscheinen nach Gesprächsende.', 'info');
    }
  };

  if (_liveHeroTimer) { clearInterval(_liveHeroTimer); _liveHeroTimer = null; }

  var recStableId = vxGetCallStableId(rec);
  if (recStableId && !_liveHeroNotified.has(recStableId)) {
    _liveHeroNotified.add(recStableId);
    notifyLiveCall(caller, company, rec.id);

    vxBellAdd({ type: 'call', title: 'Eingehender Anruf', sub: company ? company + ' · ' + caller : caller, recId: rec.id });
    vxMaybePlayLiveCallSound(rec, 'live-hero');
  }

  // Live-Call nur auf Heute-Tab zeigen (Jetzt wichtig)
  // Anfragen-Liste zeigt nur abgeschlossene Anrufe
  if (priorityList) {
    priorityList.insertBefore(row, priorityList.firstChild);
  }

  _liveHeroCurrentId = rec.id;
}

// Browser Notification API — sendet OS-Notification wenn Tab nicht im Fokus
function ensureNotificationPermission() {
  if (!('Notification' in window)) return Promise.resolve('unsupported');
  if (Notification.permission === 'granted') return Promise.resolve('granted');
  if (Notification.permission === 'denied') return Promise.resolve('denied');
  try {
    return Notification.requestPermission().then(function(p) { return p; });
  } catch (e) {
    return new Promise(function(resolve) {
      Notification.requestPermission(function(p) { resolve(p); });
    });
  }
}

function notifyLiveCall(caller, company, recordId) {
  if (localStorage.getItem('voxera_browser_notif_off') === '1') return;
  if (!('Notification' in window)) return;
  if (Notification.permission !== 'granted') return;
  if (document.visibilityState === 'visible') return; // Tab sichtbar → Hero reicht
  try {
    var title = 'Voxera · Eingehender Anruf';
    var body = company ? (caller + ' (' + company + ')') : caller;
    var n = new Notification(title, {
      body: body,
      icon: '/brand/05-app-icon-512px.svg',
      tag: 'voxera-live-call',
      requireInteraction: false
    });
    n.onclick = function() {
      window.focus();
      try { n.close(); } catch (e) {}
      if (recordId) {
        var navEl = document.getElementById('nav-anrufe');
        if (typeof showTab === 'function') showTab('anrufe', navEl);
        setTimeout(function() {
          if (typeof vxOpenDetailV2 === 'function') vxOpenDetailV2(recordId); else if (typeof openDetail === 'function') openDetail(recordId);
        }, 120);
      }
    };
  } catch (err) {
    console.warn('[live-hero] notification failed', err);
  }
}

function injectTaskGroupHeaders(listId) {
  var el = document.getElementById(listId);
  if (!el) return;
  // Alte Header entfernen
  el.querySelectorAll('.task-section-header').forEach(function(h) { h.remove(); });
  var rows = Array.from(el.querySelectorAll('.call-row:not(.hidden):not(.done)'));
  if (rows.length < 2) return; // Kein Header nötig wenn nur eine Row

  var getBucket = function(row) {
    if (row.querySelector('.task-due-pill.overdue')) return {key:'overdue', label:'Überfällig', cls:'sgh--red'};
    if (row.querySelector('.task-due-pill.today'))   return {key:'today',   label:'Heute fällig', cls:'sgh--blue'};
    if (row.querySelector('.task-due-pill.planned')) return {key:'planned', label:'Geplant', cls:'sgh--gray'};
    return {key:'open', label:'Ohne Termin', cls:'sgh--gray'};
  };

  // Anzahl pro Bucket vorberechnen
  var counts = {};
  rows.forEach(function(row) { var k = getBucket(row).key; counts[k] = (counts[k] || 0) + 1; });

  var lastKey = null;
  rows.forEach(function(row) {
    var bucket = getBucket(row);
    if (bucket.key !== lastKey) {
      var div = document.createElement('div');
      div.className = 'task-section-header ' + bucket.cls;
      div.innerHTML = '<span class="tsh-label">' + bucket.label + '</span>'
                    + '<span class="tsh-count">' + (counts[bucket.key] || '') + '</span>';
      row.parentNode.insertBefore(div, row);
      lastKey = bucket.key;
    }
  });
}

var _auPeriod = 'week';

function auSetPeriod(period, btn) {
  _auPeriod = period;
  document.querySelectorAll('.au-period-btn').forEach(function(b){ b.classList.remove('au-period-active'); });
  if (btn) btn.classList.add('au-period-active');

function _auFilterRecords(records) {
  var now = new Date();
  var start;
  if (_auPeriod === 'today') {
    start = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  } else if (_auPeriod === 'week') {
    start = new Date(now.getTime() - 7 * 86400000);
  } else if (_auPeriod === 'month') {
    start = new Date(now.getFullYear(), now.getMonth(), 1);
  } else {
    return records;
  }
  return records.filter(function(r) {
    var ts = getRecordTimestamp(r);
    if (!ts) return false;
    var d = parseCallTimestamp(ts);
    return d && d >= start;
  });
}

function renderAuswertung() {
  var all = allRecords || [];
  var records = _auFilterRecords(all);
  var total = records.length;

  // KPIs
  var closed = records.filter(function(r){ return isClosedStatus((r.fields||{}).dashboard_status); }).length;
  var totalDur = records.reduce(function(s,r){ return s+Number((r.fields||{}).duration_seconds||0); }, 0);
  var avgDur = total ? Math.round(totalDur/total) : 0;
  var avgDurLabel = !total ? '–' : avgDur < 60 ? avgDur+'s' : Math.floor(avgDur/60)+':'+(String(avgDur%60).padStart(2,'0'));
  var hotCount = records.filter(function(r){ return qualKey((r.fields||{}).lead_quality)==='hot'; }).length;
  var warmCount = records.filter(function(r){ return qualKey((r.fields||{}).lead_quality)==='warm'; }).length;
  var kaltCount = records.filter(function(r){ return qualKey((r.fields||{}).lead_quality)==='kalt'; }).length;
  var unknownCount = total - hotCount - warmCount - kaltCount;
  var rate = total ? Math.round(closed/total*100) : 0;

  // Set KPI values
  _auEl('au2-calls', total || '–');
  _auEl('au2-dur', avgDurLabel);
  _auEl('au2-hot', hotCount || '–');
  _auEl('au2-rate', total ? rate+'%' : '–');

  // Trends (compare to previous period)
  var prev = _auPrevPeriodRecords(all);
  var prevTotal = prev.length;
  var prevHot = prev.filter(function(r){ return qualKey((r.fields||{}).lead_quality)==='hot'; }).length;
  var prevRate = prevTotal ? Math.round(prev.filter(function(r){ return isClosedStatus((r.fields||{}).dashboard_status); }).length/prevTotal*100) : 0;
  _auTrend('au2-calls-trend', total - prevTotal, prevTotal ? 'vs. Vorperiode' : '');
  _auTrend('au2-hot-trend', hotCount - prevHot, prevTotal ? 'vs. Vorperiode' : '');
  var rateDiff = rate - prevRate;
  _auTrend('au2-rate-trend', rateDiff, prevTotal ? '% vs. Vorperiode' : '');

  // Sparklines
  _auBuildSparkCalls(all);

  // AI Summary
  _auAiSummary(records, total, hotCount, closed);

  // Insights
  _auInsights(all, records);

  // Week chart
  _auWeekChart2(records);

  // Categories
  _auCategories2(records, total);

  // Call times
  _auCallTimes2(records, total);

  // Lead quality
  _auLeadQuality(total, hotCount, warmCount, kaltCount, unknownCount);
}


function _auEl(id, val) { var el = document.getElementById(id); if (el) el.textContent = val; }

function _auTrend(id, diff, suffix) {
  var el = document.getElementById(id);
  if (!el) return;
  if (!diff && diff !== 0) { el.textContent = ''; el.className = 'au2-kpi-trend'; return; }
  el.className = 'au2-kpi-trend ' + (diff > 0 ? 'up' : diff < 0 ? 'down' : '');
  el.textContent = (diff > 0 ? '↑ +' : diff < 0 ? '↓ ' : '→ ') + (diff !== 0 ? Math.abs(diff) : '') + ' ' + (suffix||'');
}

function _auPrevPeriodRecords(all) {
  var now = new Date();
  var start, end;
  if (_auPeriod === 'today') {
    end = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    start = new Date(end.getTime() - 86400000);
  } else if (_auPeriod === 'week') {
    end = new Date(now.getTime() - 7*86400000);
    start = new Date(end.getTime() - 7*86400000);
  } else if (_auPeriod === 'month') {
    end = new Date(now.getFullYear(), now.getMonth(), 1);
    start = new Date(now.getFullYear(), now.getMonth()-1, 1);
  } else { return []; }
  return all.filter(function(r) {
    var ts = getRecordTimestamp(r); if (!ts) return false;
    var d = parseCallTimestamp(ts); return d && d >= start && d < end;
  });
}

function _auBuildSparkCalls(all) {
  var el = document.getElementById('au2-spark-calls');
  if (!el) return;
  var days = 7, now = new Date();
  var counts = [];
  for (var i = days-1; i >= 0; i--) {
    var d = new Date(now.getFullYear(), now.getMonth(), now.getDate()-i);
    var nd = new Date(d.getTime()+86400000);
    counts.push(all.filter(function(r){ var ts=getRecordTimestamp(r); if(!ts) return false; var rd=parseCallTimestamp(ts); return rd&&rd>=d&&rd<nd; }).length);
  }
  var max = Math.max.apply(null, counts) || 1;
  el.innerHTML = counts.map(function(c, i) {
    var h = Math.max(3, Math.round(c/max*22));
    var isToday = (i === counts.length-1);
    return '<div class="au2-spark-bar" style="height:'+h+'px;background:'+(isToday?'#4B9FFF':'rgba(96,165,250,'+(0.2+i*0.05)+')')+';"></div>';

function injectArchivDateHeaders(listId, records) {
  var el = document.getElementById(listId);
  if (!el) return;
  el.querySelectorAll('.archiv-date-header').forEach(function(h) { h.remove(); });
  var rows = Array.from(el.querySelectorAll('.call-row:not(.hidden)'));
  if (rows.length < 2) return;

  var now = new Date();
  var todayStart     = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  var yesterdayStart = new Date(todayStart.getTime() - 86400000);
  var weekStart      = new Date(todayStart.getTime() - 6 * 86400000);
  var monthStart     = new Date(now.getFullYear(), now.getMonth(), 1);

  // Record-Lookup für Timestamps
  var recordMap = {};
  (records || _archivAllDone || allRecords || []).forEach(function(r) {
    if (r && r.id) recordMap[String(r.id)] = r;
  });

  var getBucket = function(id) {
    var rec = recordMap[String(id)];
    var ts = rec ? getRecordTimestamp(rec) : null;
    if (!ts) return {key:'older', label:'Älter'};
    var d = parseCallTimestamp(ts);
    if (!d || isNaN(d.getTime())) return {key:'older', label:'Älter'};
    if (d >= todayStart)     return {key:'today',     label:'Heute'};
    if (d >= yesterdayStart) return {key:'yesterday', label:'Gestern'};
    if (d >= weekStart)      return {key:'week',      label:'Diese Woche'};
    if (d >= monthStart)     return {key:'month',     label:'Dieser Monat'};
    return {key:'older', label:'Älter'};
  };

  var lastKey = null;
  rows.forEach(function(row) {
    var bucket = getBucket(row.dataset && row.dataset.id);
    if (bucket.key !== lastKey) {
      var div = document.createElement('div');
      div.className = 'archiv-date-header';
      div.innerHTML = '<span class="adh-label">' + bucket.label + '</span>';
      row.parentNode.insertBefore(div, row);
      lastKey = bucket.key;
    }
  });
}

function vxNormalizeTranscriptText(raw) {
  if (raw == null) return '';

  // ElevenLabs / webhook data may arrive as plain text, JSON string, array of turns,
  // or object-shaped transcript payload. Normalize all formats into readable text.
  try {
    if (Array.isArray(raw)) {
      return raw.map(function(item){
        if (item == null) return '';
        if (typeof item === 'string') return item;
        if (typeof item === 'object') {
          var speaker = item.speaker || item.role || item.type || item.author || '';
          var body = item.text || item.message || item.content || item.transcript || item.value || '';
          if (Array.isArray(body)) body = body.join(' ');
          if (body && typeof body === 'object') body = JSON.stringify(body);
          return (speaker ? String(speaker).trim() + ': ' : '') + String(body || '').trim();
        }
        return String(item || '');
      }).filter(Boolean).join('\n');
    }

    if (typeof raw === 'object') {
      if (Array.isArray(raw.turns)) return vxNormalizeTranscriptText(raw.turns);
      if (Array.isArray(raw.messages)) return vxNormalizeTranscriptText(raw.messages);
      if (Array.isArray(raw.transcript)) return vxNormalizeTranscriptText(raw.transcript);
      if (Array.isArray(raw.conversation)) return vxNormalizeTranscriptText(raw.conversation);
      if (raw.text || raw.content || raw.message) return vxNormalizeTranscriptText(raw.text || raw.content || raw.message);
      return JSON.stringify(raw);
    }

    var text = String(raw || '').trim();
    if (!text) return '';

    // Some rows store transcript JSON as string. Parse only when it is clearly JSON.
    var first = text.charAt(0);
    if (first === '[' || first === '{') {
      try {
        var parsed = JSON.parse(text);
        return vxNormalizeTranscriptText(parsed);
      } catch(parseErr) {}
    }

    return text
      .replace(/\r\n/g, '\n')
      .replace(/\r/g, '\n')
      .replace(/\\n/g, '\n')
      .replace(/\n{3,}/g, '\n\n')
      .trim();
  } catch(e) {
    try { return String(raw || '').trim(); } catch(_e) { return ''; }
  }
}

function vxTranscriptAssistantName() {
  try {
    var name = '';
    if (typeof getAssistantName === 'function') name = getAssistantName();
    if (!name && typeof customerMeta !== 'undefined' && customerMeta) name = customerMeta.assistantName;

function vxTranscriptSpeakerLabel(rawSpeaker) {
  var s = String(rawSpeaker || '').trim().toLowerCase();
  if (!s) return 'Unbekannt';
  // Assistant turns should show the configured assistant name, not the brand name.
  if (s === 'agent' || s === 'assistant' || s === 'ai' || s === 'bot' || s === 'lara' || s === 'voxera') return vxTranscriptAssistantName();
  if (s === 'user' || s === 'caller' || s === 'kunde' || s === 'kundin' || s === 'anrufer' || s === 'anruferin') return 'Anrufer';
  return s.charAt(0).toUpperCase() + s.slice(1);
}

function vxParseTranscriptTurns(raw) {
  var text = vxNormalizeTranscriptText(raw);
  if (!text) return [];

  // Provider formats seen in Voxera data:
  // 1) "agent: ...\nuser: ..."
  // 2) "agent: ... user: ... agent: ..." (speaker labels in one long line)
  // 3) line based transcript without explicit speaker labels
  var turns = [];
  var speakerRe = /\b(agent|assistant|ai|bot|voxera|lara|user|caller|kunde|kundin|anrufer|anruferin)\s*:\s*/ig;
  var matches = [];
  var m;
  while ((m = speakerRe.exec(text)) !== null) {
    matches.push({
      speaker: vxTranscriptSpeakerLabel(m[1]),
      start: m.index,
      contentStart: speakerRe.lastIndex
    });
  }

  if (matches.length) {
    var prefix = text.slice(0, matches[0].start).trim();
    if (prefix) turns.push({ speaker:'Gesprächsverlauf', text: prefix });

    for (var i = 0; i < matches.length; i++) {
      var contentEnd = (i + 1 < matches.length) ? matches[i + 1].start : text.length;
      var part = text.slice(matches[i].contentStart, contentEnd).trim();
      if (!part) continue;
      // Keep the transcript readable without destroying intentional punctuation.
      part = part.replace(/\n{3,}/g, '\n\n').trim();
      turns.push({ speaker: matches[i].speaker, text: part });
    }
    return turns.filter(function(t){ return t && String(t.text || '').trim(); });
  }

  // Fallback: one card per meaningful line. If there is only one line, keep it as one section.
  text.split('\n').map(function(line){ return line.trim(); }).filter(Boolean).forEach(function(line){
    turns.push({ speaker:'Gesprächsverlauf', text: line });
  });
  return turns.filter(function(t){ return t && String(t.text || '').trim(); });
}

function vxGetCallAudioUrl(callOrFields) {
  var src = (callOrFields && callOrFields.fields) ? callOrFields.fields : (callOrFields || {});
  var directKeys = [
    'recording_url','recordingUrl','recordingURL','call_recording_url','callRecordingUrl','callRecordingURL',
    'audio_url','audioUrl','audioURL','call_audio_url','callAudioUrl','conversation_audio_url','conversationAudioUrl',
    'elevenlabs_audio_url','elevenlabsAudioUrl','elevenlabs_recording_url','elevenlabsRecordingUrl',
    'twilio_recording_url','twilioRecordingUrl','voicemail_url','voicemailUrl','media_url','mediaUrl',
    'recording','audio','audio_file','audioFile','audio_download_url','download_url','signed_url','signedUrl'
  ];
  function validAudioUrl(text){
    text = String(text || '').trim();
    if (!text) return '';
    if (/^data:audio\//i.test(text)) return text;
    if (/^https?:\/\//i.test(text) && (/audio|record|recording|voice|voicemail|media|download|signed|mp3|wav|m4a|ogg/i.test(text))) return text;
    return '';
  }
  for (var i = 0; i < directKeys.length; i++) {
    var val = src[directKeys[i]];
    var url = validAudioUrl(val);
    if (url) return url;
  }
  var seen = [];
  function scan(obj, depth){
    if (!obj || depth > 4) return '';
    if (typeof obj === 'string') {
      var direct = validAudioUrl(obj);
      if (direct) return direct;
      try {
        var parsed = JSON.parse(obj);
        return scan(parsed, depth + 1);
      } catch(e) { return ''; }
    }
    if (typeof obj !== 'object') return '';
    if (seen.indexOf(obj) >= 0) return '';
    seen.push(obj);
    var keys = Object.keys(obj);
    for (var k = 0; k < keys.length; k++) {
      var key = keys[k];
      var value = obj[key];
      if (/audio|record|recording|voice|voicemail|media|download|signed/i.test(key)) {
        var byKey = validAudioUrl(value);
        if (byKey) return byKey;
      }
    }
    for (var j = 0; j < keys.length; j++) {
      var nested = scan(obj[keys[j]], depth + 1);
      if (nested) return nested;
    }
    return '';
  }
  return scan(src, 0);
}

async function vxTryLoadElevenLabsAudioFromDashboard(trigger) {
  var card = trigger && trigger.closest ? trigger.closest('#vx-call-audio-card') : (trigger && trigger.id === 'vx-call-audio-card' ? trigger : document.getElementById('vx-call-audio-card'));
  var status = card ? card.querySelector('.vx-audio-fetch-status') : null;
  var conversationId = card ? String(card.getAttribute('data-conversation-id') || '').trim() : '';
  if (!conversationId || !card) return false;
  if (card.getAttribute('data-vx-audio-loading') === '1') return false;
  card.setAttribute('data-vx-audio-loading','1');
  var btn = trigger && trigger.tagName === 'BUTTON' ? trigger : null;
  var endpoints = [
    '/.netlify/functions/elevenlabs-conversation-audio?conversation_id=' + encodeURIComponent(conversationId)
  ];
  if (btn) btn.disabled = true;
  if (status) status.textContent = '';
  try {
    for (var i = 0; i < endpoints.length; i++) {
      var res = await fetch(endpoints[i], { method:'GET' });
      if (!res.ok) continue;
      var ct = String(res.headers.get('content-type') || '').toLowerCase();
      var url = '';
      if (ct.indexOf('application/json') >= 0) {
        var json = await res.json();
        url = json.audio_url || json.audioUrl || json.recording_url || json.url || '';
      } else if (ct.indexOf('audio/') >= 0 || ct.indexOf('application/octet-stream') >= 0) {
        var blob = await res.blob();
        url = URL.createObjectURL(blob);
      }
      if (url && card) {
        card.outerHTML = '<div class="vx-transcript-card" id="vx-call-audio-card"><div style="padding:12px 14px;border-top:0.5px solid var(--line);">' + vxRenderModernAudioPlayerHtml(url, 'Bereit zum Abspielen') + '</div></div>';
        if (window.vxHydrateModernAudioPlayers) window.vxHydrateModernAudioPlayers(document);
        return true;
      }
    }
    card.outerHTML = vxRenderAudioStateCardHtml('error', 'Aufnahme konnte nicht geladen werden.', conversationId);
    return false;
  } catch(e) {
    console.warn('[voxera-audio] loading failed', e);
    card.outerHTML = vxRenderAudioStateCardHtml('error', 'Aufnahme konnte nicht geladen werden.', conversationId);
    return false;
  } finally {
    if (card) card.removeAttribute('data-vx-audio-loading');
    if (btn) btn.disabled = false;
  }
}

function vxMarkCallAsRead(callId) {
  if (!callId) return;
  if (typeof vxGetManualUnreadIdSet === 'function') vxGetManualUnreadIdSet().delete(String(callId));
  // Record im Cache suchen
  var rec = null;
  if (typeof allRecords !== 'undefined' && Array.isArray(allRecords)) {
    rec = allRecords.find(function(r) { return String((r && r.id) || '') === String(callId); });
  }
  if (!rec || !rec.fields) return;
  if (rec.fields.read_at) return;
  var readAtIso = new Date().toISOString();
  if (typeof vxPatchLocalRecordReadState === 'function') vxPatchLocalRecordReadState(callId, false, readAtIso);
  else rec.fields.read_at = readAtIso;
  if (typeof vxRefreshAfterReadStateChange === 'function') vxRefreshAfterReadStateChange();
  // DB async aktualisieren — non-blocking, kein Toast
  if (typeof _sb !== 'undefined' && _sb) {
    _sb.from('calls')
      .update({ read_at: readAtIso })
      .eq('id', callId)
      .is('read_at', null)
      .then(function(res) {
        if (res && res.error) {
          if (typeof vxPatchLocalRecordReadState === 'function') vxPatchLocalRecordReadState(callId, true, null);
          else rec.fields.read_at = null;
          if (typeof vxRefreshAfterReadStateChange === 'function') vxRefreshAfterReadStateChange();
        }
      })
      .catch(function() {
        if (typeof vxPatchLocalRecordReadState === 'function') vxPatchLocalRecordReadState(callId, true, null);
        else rec.fields.read_at = null;
        if (typeof vxRefreshAfterReadStateChange === 'function') vxRefreshAfterReadStateChange();
      });
  }
}

  if (typeof vxGetManualUnreadIdSet === 'function') vxGetManualUnreadIdSet().add(id);
  if (typeof vxPatchLocalRecordReadState === 'function') vxPatchLocalRecordReadState(id, true, null);
  if (typeof vxRefreshAfterReadStateChange === 'function') vxRefreshAfterReadStateChange();
  if (!(typeof _sb !== 'undefined' && _sb)) return Promise.resolve(true);
  return _sb.from('calls').update({ read_at: null }).eq('id', id).then(function(res){
    if (res && res.error) {
      console.warn('[vxRequestUnread] read_at NULL update failed.', res.error);
      return false;
    }
    return true;
  }).catch(function(err){
    console.warn('[vxRequestUnread] read_at NULL update failed.', err);
    return false;
  });
}

function vxEnsureNoOrphanDetailView(reason) {
  try {
    var page = document.getElementById('call-detail-page');
    if (!page) return false;
    var hasDetailHash = vxIsDetailHash();

    var hasCurrentDetail = !!(String(window._vxCurrentCallId || '').trim() || String(window._vxCurrentTaskId || '').trim());
    // Root-tab URLs (?tab=...) must never show a stale/empty detail shell.
    if (!hasDetailHash && (visible || hasCurrentDetail)) {
      vxForceCloseCallDetail(reason || 'orphan-detail');
      return true;
    }
    // Defensive: if CSS/inline styles expose the shell without an explicit open class, close it.
    if (!hasDetailHash && !page.classList.contains('is-open') && (page.style.display === 'block' || page.style.display === 'flex')) {

      return true;
    }
  } catch (err) {
    console.warn('[detail] orphan guard failed', { reason: reason || 'unknown', error: err && err.message ? err.message : err });
  }
  return false;

async function buildCallHistory(phoneNumber) {
  var sb = (typeof getSupabaseAuthClient === 'function') ? getSupabaseAuthClient() : null;
  if (!sb) { return []; }
  if (!phoneNumber || phoneNumber === 'undefined') { 
    console.warn('buildCallHistory: phoneNumber is empty');
    return []; 
  }
  try {
    // calls table only has 'caller_phone' field (no phone_number column)
    const { data: callsData, error: callsError } = await sb.from('calls').select('*').eq('caller_phone', phoneNumber).order('created_at', { ascending: true });
    if (callsError) throw callsError;
    const calls = callsData || [];
    
    // Tasks query - customer_tasks DB has 'phone' field (not phone_number/caller_phone)
    let tasks = [];
    try {
      const { data: tasksData, error: tasksError } = await sb.from('customer_tasks').select('*').eq('phone', phoneNumber).order('due_at', { ascending: true });
      if (tasksError) {
        console.warn('Tasks query failed (non-critical):', tasksError);
      } else {
        tasks = tasksData || [];
      }
    } catch (e) {
      console.warn('Tasks query failed (non-critical):', e);
    }
    
    const history = [];
    calls.forEach(function(call) {
      history.push({
        type: 'call_inbound',
        timestamp: call.created_at,
        duration: call.duration_seconds || 0,
        summary: vxGetCallSummary(call),
        category: call.category || '',
        lead_quality: call.lead_quality || '',
        call_id: call.id,
        transcription: vxGetCallTranscript(call),
        notes: call.notes_customer_voxera || ''
      });
    });
    tasks.forEach(function(task) {
      // customer_tasks DB schema: type, title, note, phone, due_at, status, created_at
      const taskType = task.type || 'task';
      history.push({
        type: 'followup_created',
        timestamp: task.created_at,
        action: taskType,
        title: task.title || '',
        due_at: task.due_at,
        notes: task.note || task.notes || '',
        status: task.status || 'open',
        task_id: task.id
      });
    });
    history.sort(function(a, b) {
      var ta = (typeof parseCallTimestamp === 'function' ? parseCallTimestamp(b.timestamp) : new Date(b.timestamp)) || new Date(0);
      var tb = (typeof parseCallTimestamp === 'function' ? parseCallTimestamp(a.timestamp) : new Date(a.timestamp)) || new Date(0);
      return ta.getTime() - tb.getTime();
    });
    return history;
  } catch (error) {
    console.error('Error building call history:', error);
    return [];
  }
}

async function loadAndRenderCallDetail(callId) {
  callId = String(callId || '').trim();
  if (!callId) {
    console.warn('[detail] blocked load without id', { source: 'loadAndRenderCallDetail' });
    if (typeof vxForceCloseCallDetail === 'function') vxForceCloseCallDetail('loadAndRenderCallDetail:missing-id');
    return;
  }
  var sb = (typeof getSupabaseAuthClient === 'function') ? getSupabaseAuthClient() : null;
  if (!sb) { 
    var el0 = document.getElementById('call-detail-content');
    if (el0) el0.innerHTML = '<p style="color:#dc2626;padding:16px;">⚠️ Authentifizierung erforderlich</p>';
    return;
  }
  
  // Show loading state
  var loadingEl = document.getElementById('call-detail-content');
  if (loadingEl) loadingEl.innerHTML = '<div style="padding:40px;text-align:center;color:var(--slate);">Lade…</div>';
  
  // Lokalen Record finden (schnell)
  var recordsSource = (typeof allRecords !== 'undefined' && allRecords) ? allRecords : (window.allRecords || []);
  var localRec = recordsSource.find(function(r) { return r.id === callId; });
  
  if (localRec && localRec.fields) {
    var f = localRec.fields;
    var localCall = {
      id: callId,
      caller_name: f.caller_name || '',
      caller_phone: f.caller_phone || f.phone_number || '',
      phone_number: f.phone_number || f.caller_phone || '',
      created_at: f.created_at || f.received_at || new Date().toISOString(),
      duration_seconds: f.duration_seconds || 0,
      dashboard_status: f.dashboard_status || '',
      status: f.status || '',
      follow_up_at: f.follow_up_at || '',
      call_summary: vxGetCallSummary(f),
      category: f.category || '',
      lead_quality: f.lead_quality || '',
      callback_requested: !!f.callback_requested,
      next_action: f.next_action || '',
      urgency: f.urgency || '',
      priority: f.priority || '',
      transcription: vxGetCallTranscript(f),
      notes_customer_voxera: f.notes_customer_voxera || '',
      customer_id: f.customer_id || ''
    };
    
    var phoneForHistory = localCall.caller_phone || '';

    // [v3.5 FIX] Detailseite sofort aus lokalem Record rendern.
    // Vorher wurde zuerst die Kontakt-Historie geladen. Wenn diese Abfrage langsam
    // war oder hing, blieb die Detailseite bei "Lade…" stehen. Die Detailakte
    // ist jetzt sofort sichtbar; die Historie wird danach best-effort nachgeladen.
    if (String(window._vxCurrentCallId || '') !== String(callId) || window._vxCurrentDetailType !== 'call') return;
    renderCallDetailPage(localCall, []);

    if (phoneForHistory) {
      try {
        var history = await buildCallHistory(phoneForHistory);
        if (String(window._vxCurrentCallId || '') !== String(callId) || window._vxCurrentDetailType !== 'call') return;
        renderCallDetailPage(localCall, history || []);
      } catch (err) {
        console.error('[call-detail] history error:', err);
        // Detail bleibt sichtbar; nur Historie fällt weg.
      }
    }
    return;
  }
  
  // Fallback: from Supabase
  try {
    const { data: call, error } = await sb.from('calls').select('*').eq('id', callId).single();
    if (error || !call) {
      var el = document.getElementById('call-detail-content');
      if (el) el.innerHTML = '<p style="color:#dc2626;padding:16px;">Anruf nicht gefunden.</p>';
      return;
    }
    const phoneNumber = call.caller_phone || call.phone_number || '';
    const history = phoneNumber ? await buildCallHistory(phoneNumber) : [];
    if (String(window._vxCurrentCallId || '') !== String(callId) || window._vxCurrentDetailType !== 'call') return;
    renderCallDetailPage(call, history);
  } catch (error) {
    console.error('[call-detail] load error:', error);
    var el = document.getElementById('call-detail-content');
    if (el) el.innerHTML = '<p style="color:#dc2626;padding:16px;">Fehler beim Laden.</p>';
  }
}
