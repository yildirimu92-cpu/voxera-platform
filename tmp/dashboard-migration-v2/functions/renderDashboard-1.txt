  if (kpiCellDue) kpiCellDue.style.borderTop = visibleImportantCount > 0 ? '2px solid var(--vx-gold)' : '2px solid rgba(148,163,184,.45)';
  if (kpiDueBar) kpiDueBar.style.background = visibleImportantCount > 0 ? 'linear-gradient(90deg,#FCD34D,#D97706)' : 'linear-gradient(90deg,#CBD5E1,#94A3B8)';
  if (typeof vxSetHeuteKpiCell === 'function') vxSetHeuteKpiCell(kpiCellDue, visibleImportantCount > 0, 'vxHeuteKpiScrollImportant');
  renderDashPriorityList();
  // [v2.70] Focus-Card entfernt
  var moreBtn = document.getElementById('dash-recent-more');
  if (moreBtn) {
    if (!dashPriorityExpanded && _dashPriorityAllRecords.length > 5) {
      moreBtn.style.display = 'block';
      moreBtn.textContent = 'Alle ' + _dashPriorityAllRecords.length + ' offenen Punkte anzeigen';
    } else {
      moreBtn.style.display = 'none';
    }
  }

  // Sparkline section visibility
  var sparkSection = document.getElementById('dash-sparkline-section');
  if (sparkSection && document.getElementById('dash-sparkline') && document.getElementById('dash-sparkline').children.length > 0) {
    sparkSection.style.display = '';
  }

  // ── Badges (Sidebar + Mobile Nav) ────────────────────
  var badgeA = document.getElementById('badge-anrufe');
  var badgeR = document.getElementById('badge-rueckrufe');
  var mb1    = document.getElementById('mnav-badge-anrufe');
  var mb2    = document.getElementById('mnav-badge-rueckrufe');
  // Badge = nur neue, noch nicht gesehene Anrufe
  var seenCalls = getSeenCallIdsFromStorage();
  var newCount = allOpen.filter(function(r) {
    return normalizeStatus((r.fields||{}).dashboard_status) === CALL_STATUS.NEW && !seenCalls.has(String(r.id));
  }).length;
  if (badgeA) { badgeA.textContent = newCount; badgeA.style.display = newCount > 0 ? '' : 'none'; }
  if (badgeR) { badgeR.style.display = 'none'; }
  if (mb1) { mb1.textContent = formatMobileBadgeCount(newCount); mb1.style.display = newCount > 0 ? 'block' : 'none'; }
  if (mb2) { mb2.textContent = formatMobileBadgeCount(unifiedTasks.length);   mb2.style.display = unifiedTasks.length > 0   ? 'block' : 'none'; }
  if (typeof vxUpdateAnfragenNavBadges === 'function') vxUpdateAnfragenNavBadges();

  // Callback list lives on dedicated Rückrufe tab only

  // Kompatibilität: renderUsage und updateSystemStatus bleiben unverändert
  renderUsage(records);
  updateSystemStatus(records);
}
