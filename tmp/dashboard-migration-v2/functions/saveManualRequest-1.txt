function saveManualRequest() {
  var btn = document.getElementById('manual-request-save-btn');
  if (vxIsActionButtonSaving(btn)) return;
  vxSetActionButtonLoading(btn, true, 'Erstelle...');
  try {
  var name = String(document.getElementById('manual-request-name').value || '').trim();
  if (!name) { toast('Bitte Name / Firma ausfüllen.'); return; }
  var phone = String(document.getElementById('manual-request-phone').value || '').trim();
  var category = String(document.getElementById('manual-request-category').value || 'anruf');
  var priority = String(document.getElementById('manual-request-priority').value || 'hot');
  var summary = String(document.getElementById('manual-request-summary').value || '').trim();
  var forceInbox = !!document.getElementById('manual-request-inbox').checked;
  var nowIso = new Date().toISOString();
  var rec = {
    id: 'manual_req_' + Date.now(),
    fields: {
      caller_name: name,
      company_name: name,
      phone_number: phone || null,
      category: category,
      lead_quality: priority,
      call_summary: summary || 'Manuell erfasst',
      dashboard_status: forceInbox ? 'new' : 'in_progress',
      created_at: nowIso,
      updated_at: nowIso
    }
  };
  allRecords = Array.isArray(allRecords) ? allRecords : [];
  allRecords.unshift(rec);
  closeManualRequestModal();
  renderDashboard(allRecords, allManualTaskRecords);
  renderAnrufe();
  vxUpdateAnfragenCount();
  toast('Anfrage erstellt');
  } finally {
    vxSetActionButtonLoading(btn, false);
  }
}
