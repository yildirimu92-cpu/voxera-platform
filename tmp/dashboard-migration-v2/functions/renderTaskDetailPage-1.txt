function showTaskDetail(taskId) {
  taskId = String(taskId || '').trim();
  if (!taskId) {

    if (typeof vxForceCloseCallDetail === 'function') vxForceCloseCallDetail('showTaskDetail:missing-id');
    return;
  }
  var rec = vxFindManualTaskRecord(taskId);
  if (!rec) { vxTaskDetailNotify('Aufgabe nicht gefunden', 'warning'); return; }
  var page = document.getElementById('call-detail-page');
  if (!page) return;
  if (typeof vxResetCallDetailViewState === 'function') vxResetCallDetailViewState();

  window._vxCurrentCallId = null;
  window._vxCurrentTaskId = String(taskId);

  var anrufeTabActive = document.getElementById('tab-anrufe') &&
    document.getElementById('tab-anrufe').classList.contains('active');
  var useSplitDetail = anrufeTabActive && splitRight && window.innerWidth >= 768;

  if (useSplitDetail) {
    if (!splitRight.contains(page)) splitRight.appendChild(page);
    page.classList.remove('is-fullscreen');

    page.style.position = 'absolute';
    page.style.top = '0';
    page.style.left = '0';

    page.style.bottom = '0';

    page.style.overflowY = '';

    if (_swSplit) {

      _swSplit.style.overflowX = 'hidden';

    var _swFull = document.getElementById('call-detail-scroll-wrap');
    if (_swFull) { _swFull.style.minHeight = ''; _swFull.style.height = '100%'; _swFull.style.overflowY = 'auto'; _swFull.style.boxSizing = 'border-box'; _swFull.style.width = '100%'; }

  if (!recForRouting) return;

  if (typeof showCallDetail === 'function') { showCallDetail(recordId); return; }
  rememberOverlayFocusReturn('detail-overlay');
  const rec = allRecords.find(r => r.id === recordId) || allManualTaskRecords.find(r => r.id === recordId);
  if (!rec) return;
  const f = rec.fields || {};
  const isManual = isManualTaskRecord(rec);
  const phone = (f.phone_number || f.caller_phone || '').trim();
  const tel = (f.tel_link || (phone ? ('tel:' + phone.replace(/\s/g,'')) : '')).trim();
  const detailMetaParts = [];
  if (f.company_name && !isDashLikePlaceholder(f.company_name)) detailMetaParts.push(f.company_name);
  const createdLabel = fmtDate(getRecordTimestamp(rec));
  if (createdLabel) detailMetaParts.push(createdLabel);
  const renderDetailLine = function(label, value) {

async function saveNotiz() {
  if (!notizRecordId) return;
  const btn = document.getElementById('notiz-save-btn');
  if (vxIsActionButtonSaving(btn)) return;
  const text = document.getElementById('notiz-input').value.trim();
  vxSetActionButtonLoading(btn, true, 'Speichern...');
  const recordId = notizRecordId;
  const rec = (allRecords || []).find(function(r){ return r && String(r.id) === String(recordId); });
  const currentStatus = rec && rec.fields ? normalizeStatus(rec.fields.dashboard_status) : CALL_STATUS.NEW;
  try {
    await patchCallRecord(recordId, { notes_customer_voxera: text });
    vxUpdateLocalCallRecord(recordId, { notes_customer_voxera: text });
    if (currentStatus === CALL_STATUS.NEW) {
      await applyStatusTransition(recordId, currentStatus, CALL_STATUS.IN_PROGRESS);
      vxUpdateLocalCallRecord(recordId, { dashboard_status: CALL_STATUS.IN_PROGRESS });
    }
    vxRefreshOperationalViews(allRecords, allManualTaskRecords);
    closeNotiz();
    toast('Notiz gespeichert');
    vxScheduleSingleDataRefresh(600);
  } catch(e) {
    const detail = e && e.message ? String(e.message) : '';
    toast(detail ? 'Speichern fehlgeschlagen: ' + detail : 'Speichern fehlgeschlagen');
    // Error contract: keep modal open and preserve inputs.
  } finally {
    vxSetActionButtonLoading(btn, false);
  }
}

})();

function selectNotifCard(mode, opts) {
  var options = opts || {};
  var shouldAutosave = options.autosave !== false;
  if (isContractRestrictedState() && shouldAutosave) {
    toast('Benachrichtigungen sind ohne aktiven Vertrag schreibgeschützt.', 'warning');
    return;
  }
  var source = options.source || 'ui';
  var cards = {none:'notif-card-none', callback_only:'notif-card-callback', all_calls:'notif-card-all'};
  var dots  = {none:'notif-dot-none', callback_only:'notif-dot-callback', all_calls:'notif-dot-all'};
  Object.keys(cards).forEach(function(m) {
    var card = document.getElementById(cards[m]);
    var dot  = document.getElementById(dots[m]);
    if (!card) return;
    var active = m === mode;
    card.style.borderColor = active ? 'var(--blue-mid)' : 'var(--line)';
    card.style.background  = active ? 'var(--blue-light)' : 'var(--surface)';
    var titleEl = card.querySelector('div > div:first-child');
    if (titleEl) titleEl.style.color = active ? '#1A6FE8' : 'var(--ink)';
    if (dot) {
      dot.style.background = active ? '#1A6FE8' : 'transparent';
      dot.style.border = active ? 'none' : '1.5px solid var(--line)';
      dot.innerHTML = '';
    }
  });
  var hidden = document.getElementById('notification-mode-hidden');
  if (hidden) hidden.value = mode;
  // Sync neue Radio-Buttons in Mehr → Benachrichtigungen
  document.querySelectorAll('input[name="notif-email"]').forEach(function(r) {
    r.checked = r.value === mode;
  });
  notifDebugLog('mode-selected', {
    source: source,
    mode: mode,
    autosave: shouldAutosave
  });
  if (shouldAutosave) {
    if (_notifSaveTimer) clearTimeout(_notifSaveTimer);
    _notifSaveTimer = setTimeout(function() {
      saveNotifSettings({ source: source + ':autosave' });
    }, 600);
  }
}

function previewSound(type) {
  if (localStorage.getItem('voxera_sound_off') === '1') return;
  try {
    var ctx = getAudioCtx();
    function note(freq, delay, duration, vol, waveType) {
      var osc = ctx.createOscillator();
      var gain = ctx.createGain();
      osc.connect(gain); gain.connect(ctx.destination);
      osc.type = waveType || 'sine';
      osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
      gain.gain.setValueAtTime(0, ctx.currentTime + delay);
      gain.gain.linearRampToValueAtTime(vol, ctx.currentTime + delay + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + duration);
      osc.start(ctx.currentTime + delay);
      osc.stop(ctx.currentTime + delay + duration);
    }
    if (type === 'subtle') {
      note(1200, 0, 0.06, 0.12, 'sine');
    } else if (type === 'warm') {
      note(660, 0, 0.05, 0.45, 'triangle');
      note(660, 0, 0.55, 0.32, 'sine');
      note(990, 0.08, 0.40, 0.18, 'sine');
    } else if (type === 'crisp') {
      note(900, 0, 0.1, 0.28, 'sine');
      note(1200, 0.14, 0.1, 0.28, 'sine');
    }
  } catch(e) {}
}

async function saveNotifSettings(opts) {
  if (isContractRestrictedState()) {
    toast('Benachrichtigungen können ohne aktiven Vertrag nicht gespeichert werden.', 'warning');
    return;
  }
  var options = opts || {};
  var source = options.source || 'unknown';
  var requestId = ++_notifSaveSeq;
  if (_notifSaveInFlight) {
    notifDebugLog('save-skipped-inflight', {
      source: source,
      requestId: requestId
    });
    return;
  }
  _notifSaveInFlight = requestId;
  const recordId = document.getElementById('account-overlay').dataset.customerRecordId;
  if (!recordId) {
    // Try to get record ID if not loaded yet
    try {
      const context = await resolveCustomerContext();
      if (context.customerRecord && context.customerRecord.id) {
        document.getElementById('account-overlay').dataset.customerRecordId = context.customerRecord.id;
      }
    } catch(e) {}
    var rid = document.getElementById('account-overlay').dataset.customerRecordId;
    if (!rid) {
      notifDebugLog('save-abort-no-record', {
        source: source,
        requestId: requestId
      });
      _notifSaveInFlight = null;
      toast('Fehler: Kundendaten nicht geladen');
      return;
    }
  }
  var finalRecordId = document.getElementById('account-overlay').dataset.customerRecordId;
  var notificationMode = getSelectedNotificationMode('notification-mode') || 'none';
  var payload = { notification_mode: notificationMode };
  notifDebugLog('save-request-start', {
    source: source,
    requestId: requestId,
    recordId: finalRecordId,
    notificationMode: notificationMode,
    payload: payload
  });
  try {
    await patchCustomerRecord(finalRecordId, payload);
    notifDebugLog('save-request-success', {
      source: source,
      requestId: requestId,
      recordId: finalRecordId
    });
    const s = document.getElementById('notif-success');
    s.textContent = 'Gespeichert';
    s.style.display = 'block';
    setTimeout(() => s.style.display = 'none', 2000);
  } catch(e) {
    notifDebugLog('save-request-error', {
      source: source,
      requestId: requestId,
      recordId: finalRecordId,
      error: e && e.message ? e.message : String(e)
    });
    toast(isContractInactiveError(e) ? getRestrictedContractMessage('Benachrichtigungen') : 'Fehler beim Speichern der Benachrichtigungen');
    console.error('Notif save error:', e);
  } finally {
    _notifSaveInFlight = null;
  }
}

function resolveForwardingCode(template) {
  var num = getDialableVoxeraNumber();
  if (!template || !num) return '';
  return template.replace('{voxera_number}', num);
}

function renderForwardingCodes() {
  var num = getDialableVoxeraNumber();
  var hasNumber = !!num;
  var ua = String((window.navigator && window.navigator.userAgent) || '').toLowerCase();
  var isPhoneLikeUa = /iphone|android.+mobile|windows phone|ipod/.test(ua);
  var isNarrowViewport = !!(window.matchMedia && window.matchMedia('(max-width: 820px)').matches);
  var isTouchPrimary = !!(window.matchMedia && window.matchMedia('(hover: none) and (pointer: coarse)').matches);
  var canDialDirectOnThisDevice = isPhoneLikeUa || (isTouchPrimary && isNarrowViewport);
  var fallbackEl = document.getElementById('forwarding-code-fallback');
  if (fallbackEl) fallbackEl.style.display = hasNumber ? 'none' : 'inline-flex';

  document.querySelectorAll('code[data-code-template]').forEach(function(codeEl) {
    var template = codeEl.getAttribute('data-code-template') || '';
    var resolved = resolveForwardingCode(template);
    codeEl.textContent = resolved || 'Nicht verfügbar';
  });

  document.querySelectorAll('button[data-code-template], button[data-fixed-code]').forEach(function(btn) {
    btn.disabled = !hasNumber;
    btn.style.opacity = hasNumber ? '' : '.55';
    btn.style.cursor = hasNumber ? '' : 'not-allowed';
  });

  document.querySelectorAll('.help-direct-btn').forEach(function(linkEl) {
    var fixedCode = linkEl.getAttribute('data-fixed-code') || '';
    var template = linkEl.getAttribute('data-code-template') || '';
    var resolved = fixedCode || resolveForwardingCode(template);
    var needsVoxeraNumber = !fixedCode;
    var hasRequiredData = needsVoxeraNumber ? hasNumber : true;
    var canUseDirect = !!resolved && hasRequiredData && canDialDirectOnThisDevice;
    linkEl.style.display = canDialDirectOnThisDevice ? '' : 'none';
    linkEl.setAttribute('href', canUseDirect ? ('tel:' + encodeURIComponent(resolved)) : 'javascript:void(0)');
    linkEl.setAttribute('aria-disabled', canUseDirect ? 'false' : 'true');
    linkEl.setAttribute('data-code', resolved || '');
    linkEl.setAttribute('data-direct-reason', (!hasRequiredData ? 'missing_number' : (!canDialDirectOnThisDevice ? 'device_unsupported' : 'ready')));
    var noteRoot = linkEl.closest('.help-forwarding-card') || linkEl.closest('.setup-help-card');
    var noteEl = noteRoot ? noteRoot.querySelector('.help-dialer-note') : null;
    if (noteEl) noteEl.style.display = (!canDialDirectOnThisDevice && !!resolved) ? 'block' : 'none';
  });
}

async function copyForwardingCode(event, code) {
  if (event) event.stopPropagation();
  if (!code) {
    toast('Code nicht verfügbar');
    return;
  }
  try {
    if (navigator.clipboard && typeof navigator.clipboard.writeText === 'function') {
      await navigator.clipboard.writeText(code);
    } else {
      var temp = document.createElement('textarea');
      temp.value = code;
      temp.setAttribute('readonly', '');
      temp.style.position = 'fixed';
      temp.style.opacity = '0';
      document.body.appendChild(temp);
      temp.focus();
      temp.select();
      var copied = document.execCommand('copy');
      document.body.removeChild(temp);
      if (!copied) throw new Error('Clipboard API unavailable');
    }
    announceCopyFeedback('Code kopiert', 'success');
  } catch (e) {
    announceCopyFeedback('Kopieren fehlgeschlagen', 'error');
  }
}

function announceCopyFeedback(message, type) {
  var liveEl = document.getElementById('copy-feedback-live');
  if (liveEl) liveEl.textContent = message || '';
  toast(message || 'Aktion ausgeführt', type || 'success');
}

function applyForwarding(code) {
  if (!code) return false;
  var encodedCode = encodeURIComponent(code);
  window.location.href = 'tel:' + encodedCode;
  return true;
}

function applyForwardingCode(event, linkEl) {
  if (event) event.stopPropagation();
  if (!linkEl) return;
  var code = linkEl.getAttribute('data-code') || '';
  var disabled = linkEl.getAttribute('aria-disabled') === 'true';
  var reason = linkEl.getAttribute('data-direct-reason') || '';
  if (disabled || !code) {
    if (event) event.preventDefault();
    if (reason === 'missing_number') toast('Ihre Voxera-Nummer ist noch nicht verfügbar.');
    else toast('Direktes Anwenden ist auf diesem Gerät nicht verfügbar. Bitte Code kopieren.');
    return;
  }
  applyForwarding(code);
}

async function changePassword() {
  const sb = getSupabaseAuthClient();
  if (!sb) return;
  const pw = document.getElementById('pw-new').value;
  const pw2 = document.getElementById('pw-confirm').value;
  const btn = document.getElementById('pw-save-btn');
  const errEl = document.getElementById('pw-error');
  const succEl = document.getElementById('pw-success');

  errEl.style.display = 'none';
  succEl.style.display = 'none';

  if (!pw || pw.length < 8) {
    errEl.textContent = 'Passwort muss mindestens 8 Zeichen haben.';
    errEl.style.display = 'block';
    return;
  }
  if (pw !== pw2) {
    errEl.textContent = 'Passwörter stimmen nicht überein.';
    errEl.style.display = 'block';
    return;
  }

  btn.disabled = true; btn.textContent = 'Speichern...';
  try {
    const { error } = await sb.auth.updateUser({ password: pw });
    if (error) throw error;
    succEl.textContent = 'Passwort erfolgreich geändert';
    succEl.style.display = 'block';
    document.getElementById('pw-new').value = '';
    document.getElementById('pw-confirm').value = '';
    toast('Passwort geändert');
  } catch (error) {
    errEl.textContent = error?.message || 'Fehler beim Speichern.';
    errEl.style.display = 'block';
  } finally {
    btn.disabled = false;
    btn.textContent = 'Passwort speichern';
  }
}

const MAIN_NAV_HISTORY_TABS = new Set(['dashboard','anrufe','rueckrufe','archiv','einstellungen','hilfe','assistent','auswertung','mehr']);
const MAIN_NAV_DEFAULT_TAB = 'dashboard';
const MAIN_NAV_URL_PARAM = 'tab';
const MAIN_NAV_TAB_TO_URL = {
  dashboard: 'overview',
  anrufe: 'requests',
  rueckrufe: 'tasks',
  archiv: 'history',
  einstellungen: 'settings',
  hilfe: 'help',
  assistent: 'assistant',
  auswertung: 'stats',
  mehr: 'more'
};
const MAIN_NAV_URL_TO_TAB = Object.fromEntries(
  Object.entries(MAIN_NAV_TAB_TO_URL).map(([tab,urlTab])=>[urlTab,tab])
);

function isMainNavHistoryTab(tabName){
  return MAIN_NAV_HISTORY_TABS.has(String(tabName||'').trim());
}

function getCurrentTabName(){
  const active=document.querySelector('.tab-page.active');
  if(!active||!active.id) return '';
  return active.id.replace(/^tab-/,'');
}

function getRequestedMainNavTabFromUrl(){
  const params=new URLSearchParams(location.search||'');
  if(!params.has(MAIN_NAV_URL_PARAM)) return {hasParam:false, tab:MAIN_NAV_DEFAULT_TAB, isValid:false};
  const raw=String(params.get(MAIN_NAV_URL_PARAM)||'').trim().toLowerCase();
  const mappedTab=MAIN_NAV_URL_TO_TAB[raw]||(isMainNavHistoryTab(raw)?raw:'');
  if(isMainNavHistoryTab(mappedTab)) return {hasParam:true, tab:mappedTab, isValid:true};
  return {hasParam:true, tab:MAIN_NAV_DEFAULT_TAB, isValid:false};
}

function getMainNavFromStateOrUrl(eventState){
  const stateTab=eventState&&isMainNavHistoryTab(eventState.mainTab)?eventState.mainTab:'';
  if(stateTab) return stateTab;
  return getRequestedMainNavTabFromUrl().tab;
}

function writeMainNavHistory(tabName, mode){
  if(!isMainNavHistoryTab(tabName)) return;
  const params=new URLSearchParams(location.search||'');
  params.set(MAIN_NAV_URL_PARAM,MAIN_NAV_TAB_TO_URL[tabName]||MAIN_NAV_TAB_TO_URL[MAIN_NAV_DEFAULT_TAB]);
  const query=params.toString();
  const nextUrl=`${location.pathname}${query?`?${query}`:''}${location.hash||''}`;
  const statePayload={ ...(history.state||{}), mainTab:tabName };
  if(mode==='replace') history.replaceState(statePayload,'',nextUrl);
  else history.pushState(statePayload,'',nextUrl);
}

function vxCloseTransientViewsForRootTab(nextTab) {
  if (!VX_ROOT_TABS.has(String(nextTab || ''))) return;
  try {
    vxBellClose();
    _vxBellOpen = false;
  } catch(e) {}
  try {
    if (String(nextTab) !== 'mehr' && typeof vxMehrBack === 'function') vxMehrBack();
  } catch(e) {}
  try {
    document.querySelectorAll('.overlay.open').forEach(function(el){ el.classList.remove('open'); });
    if (typeof syncOverlayScrollLock === 'function') syncOverlayScrollLock();
  } catch(e) {}
  try {
    if (typeof vxSetScrollLock === 'function') vxSetScrollLock('detail-v2', false);
    if (typeof vxSyncScrollLock === 'function') vxSyncScrollLock();
  } catch(e) {}
}

function showTab(name, el, options) {
  if (!name) return;
  const opts = (options && typeof options === 'object') ? options : {};
  vxCloseTransientViewsForRootTab(name);
  
  // ─── FIX: Call-Detail Page nur bei EXPLIZITEN Tab-Wechseln schließen ───
  // (nicht bei internen showTab-Aufrufen ohne Click-Element wie Polling/Refresh)
  var callDetailPage = document.getElementById('call-detail-page');
  if (callDetailPage && callDetailPage.style.display === 'block') {
    var isExplicitUserAction = !!el || opts.historyMode === 'push';
    if (isExplicitUserAction) {
      window._vxLastAnfragenRecordId = window._vxActiveRecordId || window._vxLastAnfragenRecordId || window._vxCurrentCallId || window._vxCurrentTaskId || '';
      if (typeof hideCallDetail === 'function') {
        hideCallDetail();
      } else {
        callDetailPage.style.display = 'none';
        vxSetScrollLock('call-detail-page', false);
        vxSyncScrollLock();
      }
    }
  }
  
  const previousTab = getCurrentTabName();
  if (previousTab === 'dashboard' && name !== 'dashboard') dashRecentExpanded = false;
  if (previousTab === 'anrufe' && name !== 'anrufe') {
    // Anfragen-Position bewusst behalten: Filter und aktiver Eintrag bleiben erhalten.
    window._vxLastAnfragenRecordId = window._vxActiveRecordId || window._vxLastAnfragenRecordId || window._vxCurrentCallId || window._vxCurrentTaskId || '';
  }
  if (name === 'assistent') { setTimeout(loadAssistentRequests, 100); }
  if (name === 'archiv') { setTimeout(function(){ if (typeof renderArchiv === 'function') renderArchiv(allRecords || []); }, 50); }
  if (previousTab === 'rueckrufe' && name !== 'rueckrufe') { _rueckrufeFilter = 'alle'; }
  closeUserMenu();
  closeMobileUserMenu();
  var targetTab = document.getElementById('tab-' + name);
  if (!targetTab) return;
  document.querySelectorAll('.tab-page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
  document.querySelectorAll('.mobile-nav-btn').forEach(n => n.classList.remove('active'));
  targetTab.classList.add('active');
  injectSortButtons();
  var navEl = el || document.getElementById('nav-' + name) || document.getElementById('mnav-' + name);
  if (navEl) navEl.classList.add('active');
  const mnav = document.getElementById('mnav-'+name);
  if (mnav) mnav.classList.add('active');
  if (name === 'vertrag') renderVertrag();
  if (name === 'einrichtung') renderEinrichtungTab();
  if (name === 'anrufe') {
    markVisibleCallsAsSeen((allRecords || []).filter(function(r) { return !isClosedStatus(((r && r.fields) || {}).dashboard_status); }));
    renderDashboard(allRecords || [], allManualTaskRecords || []);
    setTimeout(function(){
      var lastId = String(window._vxLastAnfragenRecordId || window._vxActiveRecordId || '').trim();
      if (!lastId || window.innerWidth < 768) return;
      var rec = (typeof vxResolveDprRecord === 'function') ? vxResolveDprRecord(lastId) : null;
      if (!rec) return;
      window._vxActiveRecordId = lastId;
      if (typeof renderAnrufe === 'function') renderAnrufe();
      setTimeout(function(){
        if (rec && typeof isManualTaskRecord === 'function' && isManualTaskRecord(rec)) {
          if (typeof showTaskDetail === 'function') showTaskDetail(lastId);
        } else if (typeof showCallDetail === 'function') showCallDetail(lastId, { fromBackStack: true });
      }, 60);
    }, 120);
  }
  if (isMainNavHistoryTab(name)) {
    const mode = opts.historyMode || 'push';
    const shouldWrite = mode !== 'none' && (mode === 'replace' || previousTab !== name);
    if (shouldWrite) writeMainNavHistory(name, mode === 'replace' ? 'replace' : 'push');
  }
  // Onboarding zurückbringen wenn Kunde zu Haupt-Tabs navigiert
  if (window._vxOnboardingPendingReturn && typeof customerMeta !== 'undefined' && !customerMeta.onboardingCompleted &&
      (name === 'dashboard' || name === 'mehr')) {
    window._vxOnboardingPendingReturn = false;
    setTimeout(function() {
      if (typeof vxOnboardingUpdateSteps === 'function') vxOnboardingUpdateSteps();
      const overlay = document.getElementById('vx-onboarding-overlay');
      if (overlay) overlay.style.display = 'flex';
    }, 400);
  }

  try {
    // PRODUKTREGEL: Nach First-Login darf 'einrichtung' nicht mehr persistent gespeichert werden
    const setupFinished = localStorage.getItem(getSetupStorageKey('voxera_setup_finished'));
    if (!setupFinished || name !== 'einrichtung') {
      vxSafeSessionStorageSet('voxera_tab', name);
      if (name !== 'einrichtung') {
        vxSafeSessionStorageSet('voxera_last_non_setup_tab', name);
      }
    }
  } catch(e){}
  if (name === 'einstellungen' || name === 'profil') {
    syncSystemVoxeraSettings();
    openAccount();
  }
  try {
    if (typeof vxEnsureNoOrphanDetailView === 'function') vxEnsureNoOrphanDetailView('showTab:' + name);
  } catch(e) {}
}

function consumeForceDashboardAfterLogin() {
  var shouldForce = !!window._vxForceDashboardAfterLogin || !!forceDashboardOnNextRestore;
  if (!shouldForce) return false;
  window._vxForceDashboardAfterLogin = false;
  forceDashboardOnNextRestore = false;
  window._vxActiveRecordId = null;
  window._vxCurrentCallId = null;
  window._vxCurrentTaskId = null;
  _anrufeFilter = 'offen';
  try {
    vxSafeSessionStorageSet('voxera_tab', 'dashboard');
    vxSafeSessionStorageSet('voxera_last_non_setup_tab', 'dashboard');
    vxSafeLocalStorageSet('voxera_tab', 'dashboard');
    vxSafeLocalStorageSet('currentTab', 'dashboard');
    vxSafeLocalStorageSet('activeTab', 'dashboard');
  } catch(e) {}
  if (typeof hideCallDetail === 'function') hideCallDetail();
  if (typeof closeDetail === 'function') closeDetail();
  if (typeof vxCloseDetailV2 === 'function') vxCloseDetailV2();

  try {
    history.replaceState({ ...(history.state||{}), mainTab: MAIN_NAV_DEFAULT_TAB }, '', location.pathname + location.search);
  } catch(e) {}
  showTab(MAIN_NAV_DEFAULT_TAB, findTabNavElement(MAIN_NAV_DEFAULT_TAB), { historyMode: 'replace' });
  return true;
}

function restoreTab() {
  try {
    if (consumeForceDashboardAfterLogin()) return;

    const urlState = getRequestedMainNavTabFromUrl();
    if (urlState.hasParam) {
      const navEl = findTabNavElement(urlState.tab);
      if (navEl) {
        showTab(urlState.tab, navEl, { historyMode: 'replace' });
        return;
      }
    }

    const saved = sessionStorage.getItem('voxera_tab');
    if (!saved || saved === 'dashboard') {
      showTab(MAIN_NAV_DEFAULT_TAB, findTabNavElement(MAIN_NAV_DEFAULT_TAB), { historyMode: 'replace' });
      return;
    }

    // PRODUKTREGEL: Nach First-Login darf 'einrichtung' niemals mehr automatisch restored werden
    if (saved === 'einrichtung') {
      // Neues Design: Rufumleitung ist unter Mehr → Rufumleitung
      const mehrNavEl = findTabNavElement('mehr');
      if (mehrNavEl) {
        showTab('mehr', mehrNavEl, { historyMode: 'replace' });
        setTimeout(function() { vxMehrShow('rufumleitung'); }, 100);
        return;
      }
    }

    const navEl = findTabNavElement(saved);
    if (navEl) showTab(saved, navEl, { historyMode: isMainNavHistoryTab(saved) ? 'replace' : 'none' });
  } catch(e){}
}

window.addEventListener('popstate', function(event){
  // ─── Wenn Hash #call/... oder #task/... ist, ist die Detail-Page der aktive View ───
  // → Kein showTab aufrufen (das würde die Detail-Page schließen)
  if (window.location.hash.indexOf('#call/') === 0 || window.location.hash.indexOf('#task/') === 0) {
    return;
  }
  
  // ─── Wenn vorher Detail-Hash war und jetzt nicht mehr: Detail-Page schließen ───
  var callDetailPage = document.getElementById('call-detail-page');
  if (callDetailPage && callDetailPage.style.display === 'block') {
    if (typeof hideCallDetail === 'function') hideCallDetail();
    else {
      callDetailPage.style.display = 'none';
      vxSetScrollLock('call-detail-page', false);
      vxSyncScrollLock();
    }
  }
  
  const requestedTab = getMainNavFromStateOrUrl(event && event.state);
  if (!isMainNavHistoryTab(requestedTab)) return;
  const navEl = findTabNavElement(requestedTab);
  if (!navEl) return;
  showTab(requestedTab, navEl, { historyMode: 'none' });
  try { if (typeof vxEnsureNoOrphanDetailView === 'function') vxEnsureNoOrphanDetailView('popstate'); } catch(e) {}
});

function filterList(listId, type, btn) {
  if (btn) {
    btn.closest('.filter-tabs').querySelectorAll('.ftab').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  }
  // Filter-State speichern
  if (listId === 'anrufe-list') _anrufeFilter = type;
  if (listId === 'rueckrufe-list') _rueckrufeFilter = type;
  // Gruppen-Header immer erst entfernen
  var listEl = document.getElementById(listId);
  if (listEl) listEl.querySelectorAll('.task-section-header').forEach(function(h) { h.remove(); });
  document.querySelectorAll('#'+listId+' .call-row').forEach(row => {
    if (type === 'alle') { row.classList.remove('hidden'); return; }
    if (type === 'offen') { row.classList.toggle('hidden', row.dataset.open !== '1'); return; }
    if (type === 'dringend') { row.classList.toggle('hidden', row.dataset.priority !== 'urgent'); return; }
    if (type === 'inbearbeitung') { row.classList.toggle('hidden', !['in_progress','follow_up_scheduled'].includes(row.dataset.status)); return; }
    if (type === 'neu') { row.classList.toggle('hidden', row.dataset.status !== 'new'); return; }
    row.classList.toggle('hidden', row.dataset.qual !== type);
  });
  // Gruppen-Header neu einsetzen wenn Aufgaben-Liste und Filter "alle"
  if (listId === 'rueckrufe-list' && type === 'alle' && typeof injectTaskGroupHeaders === 'function') {
    injectTaskGroupHeaders('rueckrufe-list');
  }
}

async function persistSetupDeviceType(type) {
  var normalized = type === 'mobile' ? 'mobile' : type === 'landline' ? 'landline' : null;
  if (!normalized) {
    toast('Ungültiger Aktivierungsweg.');
    return null;
  }
  try {
    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) { toast('Fehler: Kundendatensatz nicht gefunden.'); return null; }
    // Reset activation test context if currently in pending_test
    if (customerMeta.forwardingStatus === 'pending_test') {
      await resetActivationTestContext(recId, context);
    }
    await patchCustomerRecord(recId, { setup_device_type: normalized });
    customerMeta.setupDeviceType = normalized;
    if (context && context.customerRecord) context.customerRecord.setup_device_type = normalized;
    return normalized;
  } catch(e) {
    console.error('setup_device_type save failed', e);
    toast(isContractInactiveError(e) ? getRestrictedContractMessage('Aktivierung') : 'Aktivierungsweg konnte nicht gespeichert werden.');
    return null;
  }
}

}

async function activationSkipTest(event) {
  if (event) event.stopPropagation();
  var btn = event && event.currentTarget;
  if (btn) { btn.disabled = true; btn.textContent = 'Wird aktiviert…'; }
  try {
    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) { toast('Fehler: Kundendatensatz nicht gefunden.'); if (btn) { btn.disabled = false; btn.textContent = 'Ohne Test fortfahren'; } return; }
    await patchCustomerRecord(recId, {
      forwarding_status: 'active',
      forwarding_setup_completed: true,
      last_confirmed_setup_device_type: customerMeta.setupDeviceType,
      last_confirmed_forwarding_mode: customerMeta.forwardingMode,
      activation_test_mode: customerMeta.activationTestMode || 'system_call',
      activation_test_candidate_call_id: customerMeta.activationTestOutboundCallId || null
    });
    customerMeta.forwardingStatus = 'active';
    customerMeta.forwardingSetupCompleted = true;
    customerMeta.lastConfirmedSetupDeviceType = customerMeta.setupDeviceType;
    customerMeta.lastConfirmedForwardingMode = customerMeta.forwardingMode;
    if (context && context.customerRecord) {
      context.customerRecord.forwarding_status = 'active';
      context.customerRecord.forwarding_setup_completed = true;
      context.customerRecord.last_confirmed_setup_device_type = customerMeta.setupDeviceType;
      context.customerRecord.last_confirmed_forwarding_mode = customerMeta.forwardingMode;
      context.customerRecord.activation_test_mode = customerMeta.activationTestMode || 'system_call';
      context.customerRecord.activation_test_candidate_call_id = customerMeta.activationTestOutboundCallId || null;
    }
    clearActivationFreshRunTracking();
    candidateCall = null;
    activationFlowSuccessToken = '';
    stopActivationPolling();
    toast('Rufumleitung erfolgreich aktiviert.');
    renderActivationV2();
    scrollToStatusCard();
    triggerStatusHighlight('success');
  } catch(e) {
    console.error('activationSkipTest failed', e);
    toast('Aktivierung konnte nicht abgeschlossen werden.');
    if (btn) { btn.disabled = false; btn.textContent = 'Ohne Test fortfahren'; }
  }
}

async function persistForwardingStatus(status) {
  var validStatuses = ['not_started', 'pending_test', 'active', 'inactive'];
  if (!validStatuses.includes(status)) {
    console.error('persistForwardingStatus: ungültiger Status', status);
    toast('Ungültiger Weiterleitungsstatus.');
    return false;
  }
  try {
    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) { toast('Fehler: Kundendatensatz nicht gefunden.'); return false; }
    await patchCustomerRecord(recId, { forwarding_status: status });
    customerMeta.forwardingStatus = status;
    if (context && context.customerRecord) context.customerRecord.forwarding_status = status;
    return true;
  } catch(e) {
    console.error('persistForwardingStatus failed', e);
    toast('Weiterleitungsstatus konnte nicht gespeichert werden.');
    return false;
  }
}

function resolveActivationCreatedAtMs(fields) {
  var createdAtRaw = fields && fields.created_at ? String(fields.created_at) : '';
  var createdAtMs = createdAtRaw ? new Date(createdAtRaw).getTime() : NaN;
  return {
    raw: createdAtRaw,
    ms: createdAtMs,

  };
}

// Returns the first qualifying outbound test marker for the current run.
function findActivationOutboundTestCall(calls, activationStartedAt) {
  if (!activationStartedAt || !Array.isArray(calls) || !calls.length) {
    logActivationDetectionDiagnostics('outbound_missing_inputs', {
      activation_started_at: activationStartedAt || null,
      calls_count: Array.isArray(calls) ? calls.length : 0
    });
    return null;
  }
  var activationStartMs = new Date(activationStartedAt).getTime();
  if (!Number.isFinite(activationStartMs)) {
    logActivationDetectionDiagnostics('outbound_invalid_activation_started_at', {
      activation_started_at: activationStartedAt
    });
    return null;
  }
  var currentCustomerId = String((customerMeta && customerMeta.customerId) || '').trim();
  var best = null;
  var timeoutDeadlineMs = activationStartMs + ACTIVATION_TEST_TIMEOUT_MS;
  logActivationDetectionDiagnostics('outbound_scan_started', {
    activation_started_at: activationStartedAt,
    timeout_deadline: new Date(timeoutDeadlineMs).toISOString(),
    customer_id: currentCustomerId || null
  });
  for (var i = 0; i < calls.length; i++) {
    var call = calls[i];
    var fields = (call && call.fields) || {};
    var callCustomerId = String(fields.customer_id || '').trim();
    if (currentCustomerId && callCustomerId && callCustomerId !== currentCustomerId) continue;

    var direction = String(fields.direction || fields.call_direction || '').trim().toLowerCase();
    if (direction !== 'outbound') continue;

    var category = String(fields.category || '').trim().toLowerCase();
    if (category !== 'activation_test_outbound') continue;

    var createdAtRaw = fields.created_at || '';
    if (!createdAtRaw) continue;
    var createdAtMs = new Date(createdAtRaw).getTime();
    if (!Number.isFinite(createdAtMs)) continue;
    if (createdAtMs < (activationStartMs - ACTIVATION_OUTBOUND_START_GRACE_MS)) continue;
    if (createdAtMs > timeoutDeadlineMs) continue;

    if (!best || createdAtMs < new Date(best.fields.created_at || 0).getTime()) best = call;
  }
  logActivationDetectionDiagnostics('outbound_marker_found', {
    found: !!best,
    outbound_call_id: best ? String(best.fields.call_id || best.id || '') : null,
    outbound_created_at: best ? (best.fields.created_at || null) : null
  });
  return best || null;
}

function findActivationCandidateCall(calls, activationStartedAt) {
  if (!activationStartedAt || !Array.isArray(calls) || !calls.length) return null;
  var isPendingTest = customerMeta && customerMeta.forwardingStatus === 'pending_test';
  if (!isPendingTest) return null;

  var outbound = findActivationOutboundTestCall(calls, activationStartedAt);
  var outboundCreatedMs = outbound ? new Date(((outbound.fields || {}).created_at) || '').getTime() : NaN;
  var hasValidOutboundMarker = Number.isFinite(outboundCreatedMs);

  var activationStartMs = new Date(activationStartedAt).getTime();
  if (!Number.isFinite(activationStartMs)) {
    logActivationDetectionDiagnostics('candidate_scan_invalid_activation_started_at', {
      activation_started_at: activationStartedAt
    });
    return null;
  }
  var timeoutDeadlineMs = activationStartMs + ACTIVATION_TEST_TIMEOUT_MS;

  var currentCustomerId = String((customerMeta && customerMeta.customerId) || '').trim();
  var bestInbound = null;
  for (var i = 0; i < calls.length; i++) {
    var call = calls[i];
    var fields = (call && call.fields) || {};
    var candidateCallId = String(fields.call_id || call.id || '');
    var callCustomerId = String(fields.customer_id || '').trim();
    var direction = normalizeActivationDirection(fields.direction || fields.call_direction);
    var category = String(fields.category || '').trim().toLowerCase();
    var createdMeta = resolveActivationCreatedAtMs(fields);
    var createdAtMs = createdMeta.ms;

    var customerMatch = !(currentCustomerId && callCustomerId && callCustomerId !== currentCustomerId);
    // direction wird NICHT als Filter verwendet.
    // Grund: Voxera-Calls kommen via Twilio-Webhook und haben direction='outbound'
    // auch wenn sie fuer den Kunden eingehende Weiterleitungen sind.
    // Der korrekte Filter ist category: nur 'activation_test_outbound' wird ausgeschlossen.
    // directionMatch = immer true (direction-Feld ist nicht zuverlaessig).
    var directionMatch = true;
    var categoryMatch = category !== 'activation_test_outbound';
    var createdAtParsed = createdMeta.parsedSuccessfully;
    // 30s Grace vor activation_started_at für Clock-Skew zwischen Client/DB
    var afterActivationStartedAt = createdAtParsed && createdAtMs >= (activationStartMs - 30000);
    var afterOutboundMarker = createdAtParsed && (!hasValidOutboundMarker || createdAtMs >= outboundCreatedMs);
    var withinTimeout = createdAtParsed && createdAtMs <= timeoutDeadlineMs;
    var persistedActivationInbound = category === 'activation_test_inbound';
    var withinTimeoutOrPersistedDetection = withinTimeout || persistedActivationInbound;

    var rejectReason = '';
    if (!customerMatch) rejectReason = 'customer_mismatch';
    else if (!directionMatch) rejectReason = 'direction_mismatch';
    else if (!categoryMatch) rejectReason = 'category_mismatch_outbound_marker';
    else if (!createdAtParsed) rejectReason = 'created_at_parse_failed';
    else if (!afterActivationStartedAt) rejectReason = 'before_activation_started_at';
    // afterOutboundMarker nur ablehnen wenn Outbound-Marker in DB vorhanden
    else if (hasValidOutboundMarker && !afterOutboundMarker) rejectReason = 'before_outbound_marker';
    else if (!withinTimeoutOrPersistedDetection) rejectReason = 'after_timeout';

    var accepted = !rejectReason;
    logActivationDetectionDiagnostics('pending_test_candidate_evaluated', {
      candidate_call_id: candidateCallId || null,
      customer_match: customerMatch,
      direction_raw: fields.direction || fields.call_direction || null,
      direction_normalized: direction || null,
      direction_match: directionMatch,
      category_match: categoryMatch,
      created_at_parsed_successfully: createdAtParsed,
      after_activation_started_at: afterActivationStartedAt,
      after_outbound_marker: afterOutboundMarker,
      within_timeout: withinTimeout,
      persisted_activation_inbound: persistedActivationInbound,
      within_timeout_or_persisted_detection: withinTimeoutOrPersistedDetection,
      final_accepted: accepted,
      reject_reason: rejectReason || null,
      customer_id: callCustomerId || null,
      direction: direction || null,
      category: category || null,
      created_at: createdMeta.raw || null,
      called_number: fields.called_number || null,
      voxera_number: fields.voxera_number || null
    });

    if (!accepted) continue;
    if (!bestInbound || createdAtMs < new Date(bestInbound.fields.created_at || 0).getTime()) bestInbound = call;
  }

  logActivationDetectionDiagnostics('inbound_candidate_found', {
    found: !!bestInbound,
    inbound_call_id: bestInbound ? String(bestInbound.fields.call_id || bestInbound.id || '') : null,
    inbound_created_at: bestInbound ? (bestInbound.fields.created_at || null) : null,
    outbound_marker_call_id: outbound ? String(outbound.fields.call_id || outbound.id || '') : null,
    outbound_marker_present: hasValidOutboundMarker
  });

  if (bestInbound) {
    customerMeta.activationTestMode = 'system_call';
    customerMeta.activationTestOutboundCallId = outbound ? String(outbound.fields.call_id || outbound.id || '') : null;
  }
  return bestInbound || null;
}

async function activationStartSmartphone(event, code) {
  if (event) event.stopPropagation();
  if (activationStartTriggerInFlight) return;
  activationStartTriggerInFlight = true;
  var btn = event && event.currentTarget;
  var previousStatus = customerMeta.forwardingStatus;
  var previousStartedAt = customerMeta.activationStartedAt;
  try {
    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) { toast('Fehler: Kundendatensatz nicht gefunden.'); return; }
    if (btn) { btn.disabled = true; btn.textContent = 'Wird gespeichert…'; }
    var now = new Date().toISOString();
    clearActivationStateForRunStart();
    markActivationFreshRunStarted(now);
    activationRunStartLocalIso = now;
    customerMeta.forwardingStatus = 'pending_test';
    customerMeta.activationStartedAt = now;
    renderActivationV2();
    // Step 1: persist state before opening dialer
    await patchCustomerRecord(recId, { forwarding_status: 'pending_test', activation_started_at: now, activation_test_mode: 'system_call', activation_test_candidate_call_id: null });
    customerMeta.forwardingStatus = 'pending_test';
    customerMeta.activationStartedAt = now;
    customerMeta.activationTestMode = 'system_call';
    customerMeta.activationTestOutboundCallId = null;
    if (context && context.customerRecord) {
      context.customerRecord.forwarding_status = 'pending_test';
      context.customerRecord.activation_started_at = now;
      context.customerRecord.activation_test_mode = 'system_call';
      context.customerRecord.activation_test_candidate_call_id = null;
    }
    markActivationFreshRunPersisted();
    activationRunStartLocalIso = '';
    // Step 2: rerender UI (pending_test card is now shown)
    renderActivationV2();
    // Step 3: start lightweight polling to detect the candidate call
    startActivationPolling();
    // Step 4: only after state is safely persisted, open the dialer
    if (code) window.location.href = 'tel:' + code;
  } catch(e) {
    console.error('activationStartSmartphone failed', e);
    clearActivationFreshRunTracking();
    activationRunStartLocalIso = '';
    customerMeta.forwardingStatus = previousStatus;
    customerMeta.activationStartedAt = previousStartedAt;
    toast('Aktivierung konnte nicht gestartet werden.');
    if (btn) { btn.disabled = false; btn.textContent = 'Jetzt aktivieren'; }
    renderActivationV2();
  } finally {
    activationStartTriggerInFlight = false;
  }
}

    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) { toast('Fehler: Kundendatensatz nicht gefunden.'); if (btn) { btn.disabled = false; btn.textContent = 'Aktivierung abschliessen'; } return false; }
    // Aktivierung erlauben wenn: automatisch erkannt ODER User hat manuell bestätigt
    // (forwarding_status = pending_test = User hat Code eingegeben und Testanruf gemacht)
    var isPendingTest = customerMeta && customerMeta.forwardingStatus === 'pending_test';
    if (!isPendingTest) {
      toast('Aktivierung ist nur im Status "Wird geprüft" möglich.');
      if (btn) { btn.disabled = false; btn.textContent = 'Aktivierung abschliessen'; }
      return false;
    }
    var activatedAtNow = new Date().toISOString();
    await patchCustomerRecord(recId, {
      forwarding_status: 'active',
      forwarding_setup_completed: true,
      activated_at: activatedAtNow,
      last_confirmed_setup_device_type: customerMeta.setupDeviceType,
      last_confirmed_forwarding_mode: customerMeta.forwardingMode,
      activation_test_mode: customerMeta.activationTestMode || 'system_call',
      activation_test_candidate_call_id: customerMeta.activationTestOutboundCallId || null
    });
    customerMeta.forwardingStatus = 'active';
    customerMeta.forwardingSetupCompleted = true;
    customerMeta.activatedAt = activatedAtNow;
    customerMeta.lastConfirmedSetupDeviceType = customerMeta.setupDeviceType;
    customerMeta.lastConfirmedForwardingMode = customerMeta.forwardingMode;
    if (context && context.customerRecord) {
      context.customerRecord.forwarding_status = 'active';
      context.customerRecord.forwarding_setup_completed = true;
      context.customerRecord.activated_at = customerMeta.activatedAt;
      context.customerRecord.last_confirmed_setup_device_type = customerMeta.setupDeviceType;
      context.customerRecord.last_confirmed_forwarding_mode = customerMeta.forwardingMode;
      context.customerRecord.activation_test_mode = customerMeta.activationTestMode || 'system_call';
      context.customerRecord.activation_test_candidate_call_id = customerMeta.activationTestOutboundCallId || null;
    }
    clearActivationFreshRunTracking();
    candidateCall = null;
    activationFlowSuccessToken = '';
    stopActivationPolling();
    // ── Quiet Period setzen ────────────────────────────────────────────────
    // Blockiert loadData() fuer ACTIVATION_CONFIRM_QUIET_MS (2500ms) davon,
    // forwardingStatus aus der DB zu ueberschreiben. Verhindert das Zurueck-
    // fallen auf 'pending_test' wenn die DB-Propagation noch laeuft und
    // Realtime/Polling schneller feuert als der Write committed ist.
    activationConfirmQuietUntil = Date.now() + ACTIVATION_CONFIRM_QUIET_MS;
    // State explizit setzen – Active Guard in renderActivationV2 greift zwar auch,
    // aber direkte Zuweisung verhindert jeden Edge Case.
    erreichbarkeitCurrentState = 'overview';
    // Deactivation-State zurücksetzen – verhindert Flicker auf deactivation_success
    // falls der User in einer früheren Session deaktiviert hatte.
    erreichbarkeitDeactivationDoneAt = '';
    toast('Rufumleitung erfolgreich aktiviert.');
    renderActivationV2();
    // Nochmals nach kurzem Delay rendern falls ein paralleler loadData-Lauf
    // den State zwischenzeitlich überschrieben hat.
    setTimeout(function() {
      if (customerMeta && customerMeta.forwardingStatus === 'active') {
        erreichbarkeitCurrentState = 'overview';
        renderActivationV2();
        updateSidebarStatus();
      }
    }, 600);
    scrollToStatusCard();
    triggerStatusHighlight('success');
    return true;
  } catch(e) {
    console.error('confirmActivationTestCall failed', e);
    toast('Aktivierung konnte nicht abgeschlossen werden.');
    if (btn) { btn.disabled = false; btn.textContent = 'Aktivierung abschliessen'; }
    return false;
  } finally {
    activationConfirmInFlight = false;
  }
}

async function startActivationTestFromManualCta(event) {
  if (event) event.stopPropagation();
  // Wenn Polling läuft aber User drückt explizit → Flags clearen und neu starten
  if (activationSystemCallPending || activationSystemCallTriggerInFlight) {
    activationSystemCallPending = false;
    activationSystemCallTriggerInFlight = false;
    activationSystemCallError = '';
  }
  var btn = event && event.currentTarget;
  if (btn && typeof btn.disabled !== 'undefined') {
    btn.disabled = true;
    btn.textContent = 'Testanruf wird vorbereitet…';
    btn.setAttribute('aria-busy', 'true');
  }
  var sessionLockKey = getActivationSessionLockKey();
  if (sessionLockKey && activationSystemCallSessionLock === sessionLockKey) {
    activationSystemCallPending = true;
    activationSystemCallError = '';
    startActivationPolling();
    renderActivationV2();
    return;
  }
  if (customerMeta && customerMeta.activationTestOutboundCallId) {
    activationSystemCallError = '';
    if (sessionLockKey) activationSystemCallSessionLock = sessionLockKey;
    startActivationPolling();
    renderActivationV2();
    toast('Wir warten bereits auf die Erkennung des Testanrufs auf Ihrer hinterlegten Hauptnummer.');
    return;
  }
  var started = await startSystemActivationTestCall(event);
  if (!started && btn && typeof btn.disabled !== 'undefined') {
    btn.disabled = false;
    btn.removeAttribute('aria-busy');
    btn.textContent = 'Testanruf starten';
  }
}

async function activationDeactivateFast(event) {
  if (event) event.stopPropagation();
  if (activationSystemCallPending || activationSystemCallTriggerInFlight) return;
  var btn = event && event.currentTarget;
  if (btn) { btn.disabled = true; btn.textContent = 'Wird deaktiviert…'; }
  try {
    const context = await resolveCustomerContext();
    var recId = (context && context.customerRecord && context.customerRecord.id) ? context.customerRecord.id : '';
    if (!recId) throw new Error('customer record missing');
    await patchCustomerRecord(recId, {
      forwarding_status: 'inactive',
      forwarding_setup_completed: false,
      activation_started_at: null,
      activated_at: null,
      activation_test_mode: null,
      activation_test_candidate_call_id: null
    });
    customerMeta.forwardingStatus = 'inactive';
    customerMeta.forwardingSetupCompleted = false;
    customerMeta.activationStartedAt = null;
    customerMeta.activatedAt = null;
    customerMeta.activationTestMode = null;
    customerMeta.activationTestOutboundCallId = null;
    if (context && context.customerRecord) {
      context.customerRecord.forwarding_status = 'inactive';
      context.customerRecord.forwarding_setup_completed = false;
      context.customerRecord.activation_started_at = null;
      context.customerRecord.activated_at = null;
      context.customerRecord.activation_test_mode = null;
      context.customerRecord.activation_test_candidate_call_id = null;
    }
    candidateCall = null;
    clearActivationFreshRunTracking();
    stopActivationPolling();
    toast('Weiterleitung deaktiviert.');
    renderActivationV2();
    return;
  } catch (e) {
    console.error('activationDeactivateFast failed', e);
    toast('Weiterleitung konnte nicht deaktiviert werden.');
  }
  if (btn) { btn.disabled = false; btn.textContent = 'Weiterleitung deaktivieren'; }
}
