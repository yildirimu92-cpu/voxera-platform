function saveFollowUpV2(callId) {
  var sb = (typeof getSupabaseAuthClient === 'function') ? getSupabaseAuthClient() : null;
  if (!sb) { toast('Authentifizierung erforderlich', 'warning'); return; }

  var recordsSource = (typeof allRecords !== 'undefined' && allRecords) ? allRecords : (window.allRecords || []);
  var rec = recordsSource.find(function(r) { return r.id === callId; });
  if (!rec) { toast('Anruf nicht gefunden', 'warning'); return; }
  var f = rec.fields || {};

  var activePill = document.querySelector('.vx-cd-pill.is-active');
  if (!activePill) { toast('Bitte eine Aktion wählen', 'warning'); return; }

  var dueDateInput = document.getElementById('followup-date');
  var dueDate = dueDateInput ? dueDateInput.value : '';
  if (!dueDate) { toast('Bitte Datum/Uhrzeit wählen', 'warning'); return; }

  var taskType = activePill.dataset.action || 'task';
  var taskTitleMap = { callback: 'Rückruf', offerte: 'Offerte', termin: 'Termin', task: 'Sonstiges' };
  var taskTitle = taskTitleMap[taskType] || 'Aufgabe';

  // Notiz aus Textarea
  var noteInput = document.getElementById('followup-note');
  var noteText = noteInput ? (noteInput.value || '').trim() : '';

  // Button als Loading markieren
  var saveBtn = document.querySelector('[onclick*="saveFollowUpV2"]');
  if (saveBtn) { saveBtn.disabled = true; saveBtn.textContent = '…'; }

  var followUpLocalValue = normalizeFollowUpLocalValue(dueDate) || String(dueDate || '').trim();
  var followUpDueAt = followUpLocalValue.slice(0, 10);
  var followUpDueTime = normalizeManualTaskDueTime(followUpLocalValue.slice(11, 16));
  if (!followUpDueAt || !followUpDueTime) { toast(MANUAL_TASKS_DUE_TIME_REQUIRED_MESSAGE, 'warning'); return; }

  var payload = {
    title: taskTitle + (f.caller_name ? ': ' + f.caller_name : ''),
    note: noteText,
    status: 'new',
    priority: 'medium',
    type: taskType,
    due_at: followUpDueAt,
    due_time: followUpDueTime,
    phone: f.caller_phone || f.phone_number || ''
  };

  createManualTask(payload).then(function(created) {
    if (!Array.isArray(allManualTaskRecords)) allManualTaskRecords = [];
    if (created && created.id) {
      var exists = allManualTaskRecords.some(function(r) { return String(r.id) === String(created.id); });
      if (!exists) allManualTaskRecords.unshift(created);
    }

    // [PATCH 2026-05-13 v2.3 — Rückruf-Konsolidierung (Weg 2: DB-seitig)]
    // Wenn der Call callback_requested=true hatte UND die Folge-Aufgabe NICHT
    // ein Rückruf ist (sondern Offerte/Termin/Aufgabe), wird der Callback-Status
    // am Call ausgeschaltet. Damit verschwindet die Doppelzählung in KPI und Liste.
    var isCallbackFollowup = (taskType === 'callback' || taskType === 'rueckruf');
    var hadCallbackRequest = (f.callback_requested === true || f.callback_requested === 'true');
    if (!isCallbackFollowup && hadCallbackRequest) {
      sb.from('calls')
        .update({ callback_requested: false })
        .eq('id', callId)
        .then(function(rr) {
          if (rr.error) {
            console.warn('[v2.3] callback_requested-Reset fehlgeschlagen:', rr.error);
          } else {
            // Lokal sofort aktualisieren
            if (rec.fields) rec.fields.callback_requested = false;
            // Re-Render
            if (typeof renderDashboard === 'function') {
              try { renderDashboard(allRecords, allManualTaskRecords); } catch(e) {}
            }
          }
        });
    }

    renderDashboard(allRecords, allManualTaskRecords);
    vxNotifMarkReadByRecord(recordId);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    toast('Nachfassen gespeichert', 'success');
    // Zurück ins Heute-Cockpit: dort ist die neue Aufgabe unmittelbar sichtbar.
    if (typeof hideCallDetail === 'function') hideCallDetail();
    var navEl = document.getElementById('nav-dashboard');
    if (typeof showTab === 'function') showTab('dashboard', navEl);
    if (typeof loadData === 'function') setTimeout(loadData, 300);
  }).catch(function(err) {
    console.error('[followup] save error:', err);
    toast(err.message || 'Fehler beim Speichern', 'warning');
    if (saveBtn) { saveBtn.disabled = false; saveBtn.textContent = 'Speichern'; }
  });
}
