function setupDashboardRealtime(customerIdRaw) {
  if (!_sb || typeof _sb.channel !== 'function') return;
  const customerId = String(customerIdRaw || '').trim();
  if (!customerId) {
    teardownDashboardRealtime();
    return;
  }
  const nextKey = `dashboard:${customerId}`;
  if (dashboardRealtimeChannel && dashboardRealtimeKey === nextKey) return;
  teardownDashboardRealtime();
  const channelName = `customer-dashboard-${customerId}`;
  const channel = _sb
    .channel(channelName)
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'calls',
      filter: `customer_id=eq.${customerId}`
    }, function(payload) {
      var next = (payload && payload.new) || null;
      var nextId = next && next.id != null ? String(next.id) : '';
      var nextLiveStatus = String((next && next.live_status) || '').toLowerCase();
      var isTerminalLiveStatus = CALL_LIVE_TERMINAL_STATES.has(nextLiveStatus);
      if (nextId && isTerminalLiveStatus && Array.isArray(allRecords) && allRecords.length) {
        var idx = allRecords.findIndex(function(r) { return String((r && r.id) || '') === nextId; });
        if (idx >= 0 && allRecords[idx] && allRecords[idx].fields) {
          allRecords[idx].fields.live_status = next.live_status;
          if (next.updated_at) allRecords[idx].fields.updated_at = next.updated_at;
          if (next.completed_at) allRecords[idx].fields.completed_at = next.completed_at;
          if (next.duration_seconds !== undefined) allRecords[idx].fields.duration_seconds = next.duration_seconds;
        }
      }
      if (isTerminalLiveStatus) {
        // Terminal-Callbacks müssen LIVE sofort ausblenden (keine künstliche Nachlaufzeit).
        renderDashboard(allRecords, allManualTaskRecords);
        renderAnrufe();
      }
      scheduleRealtimeLoad(isTerminalLiveStatus ? 'calls-terminal-change' : 'calls-change');
    })
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'customer_tasks',
      filter: `customer_id=eq.${customerId}`
    }, function() {
      scheduleRealtimeLoad('cases-change');
    })
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'customers',
      filter: `id=eq.${customerId}`
    }, function() {
      scheduleRealtimeLoad('customer-change');
    })
    .subscribe(function(status) {
      if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
        console.warn('[realtime-sync] channel degraded, polling fallback remains active', { status: status, customerId: customerId });
      }
    });
  dashboardRealtimeChannel = channel;
  dashboardRealtimeKey = nextKey;
}
