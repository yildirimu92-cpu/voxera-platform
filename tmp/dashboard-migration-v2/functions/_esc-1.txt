function _formatRelativeTime(ts) {
  var now = new Date();
  var d = null;
  try {
    if (typeof parseCallTimestamp === 'function') d = parseCallTimestamp(ts);
  } catch(e) { d = null; }
  if (!d || Number.isNaN(d.getTime())) {
    var tsStr = String(ts||'').trim();
    if (tsStr && !tsStr.endsWith('Z') && !tsStr.includes('+') && tsStr.length >= 16) tsStr += 'Z';
    d = new Date(tsStr);
  }
  if (!d || Number.isNaN(d.getTime())) return '';
  // Zürich-Timezone für alle Zeitanzeigen
  var tz = 'Europe/Zurich';
  var formatter = new Intl.DateTimeFormat('de-CH', { timeZone: tz, hour:'2-digit', minute:'2-digit', hour12:false });
  var dateFormatter = new Intl.DateTimeFormat('de-CH', { timeZone: tz, day:'2-digit', month:'2-digit', year:'numeric' });
  var dayFormatter = new Intl.DateTimeFormat('de-CH', { timeZone: tz, year:'numeric', month:'2-digit', day:'2-digit' });
  // Heute/Gestern-Vergleich in Zürich-Zeit
  var zurichParts = new Intl.DateTimeFormat('de-CH', { timeZone: tz, year:'numeric', month:'2-digit', day:'2-digit' }).formatToParts(d);
  var zurichDay = zurichParts.find(function(p){return p.type==='day';});
  var zurichMonth = zurichParts.find(function(p){return p.type==='month';});
  var zurichYear = zurichParts.find(function(p){return p.type==='year';});
  var nowParts = new Intl.DateTimeFormat('de-CH', { timeZone: tz, year:'numeric', month:'2-digit', day:'2-digit' }).formatToParts(now);
  var nowDay = nowParts.find(function(p){return p.type==='day';});
  var nowMonth = nowParts.find(function(p){return p.type==='month';});
  var nowYear = nowParts.find(function(p){return p.type==='year';});
  var isToday = zurichDay && zurichDay.value === nowDay.value && zurichMonth.value === nowMonth.value && zurichYear.value === nowYear.value;
  var nowDate = new Date(now);
  var yesterdayNow = new Date(now.getTime() - 86400000);
  var yesterdayParts = new Intl.DateTimeFormat('de-CH', { timeZone: tz, year:'numeric', month:'2-digit', day:'2-digit' }).formatToParts(yesterdayNow);
  var ydDay = yesterdayParts.find(function(p){return p.type==='day';});
  var ydMonth = yesterdayParts.find(function(p){return p.type==='month';});
  var ydYear = yesterdayParts.find(function(p){return p.type==='year';});
  var isYesterday = zurichDay && zurichDay.value === ydDay.value && zurichMonth.value === ydMonth.value && zurichYear.value === ydYear.value;
  var hhmm = formatter.format(d);
  if (isToday) return 'Heute, ' + hhmm;
  if (isYesterday) return 'Gestern, ' + hhmm;
  return dateFormatter.format(d);
}

  if (action === 'call') {

        else if (typeof showCallDetail === 'function') showCallDetail(recId);

      else if (typeof showToast === 'function') showToast('⚠️ Keine gültige Telefonnummer vorhanden');
    }
    return;
  }
  if (action === 'done') {
    if (actionIsDone) {
      if (typeof toast === 'function') toast('Dieser Eintrag ist bereits erledigt.', 'info');
      else if (typeof showToast === 'function') showToast('Dieser Eintrag ist bereits erledigt.');

function contractRecencyScore(row) {
  if (!row || typeof row !== 'object') return 0;
  var updatedAt = new Date(row.updated_at || 0).getTime();
  var createdAt = new Date(row.created_at || 0).getTime();
  var startDate = new Date(row.start_date || 0).getTime();
  var maxTs = Math.max(
    Number.isFinite(updatedAt) ? updatedAt : 0,
    Number.isFinite(createdAt) ? createdAt : 0,
    Number.isFinite(startDate) ? startDate : 0
  );
  return Number.isFinite(maxTs) ? maxTs : 0;
}

function deriveContractState(contractRows) {
  const rows = Array.isArray(contractRows) ? [...contractRows] : [];
  rows.sort(function(a, b) {
    return contractRecencyScore(b) - contractRecencyScore(a);
  });
  const latest = rows.length ? rows[0] : null;
  const activeContract = rows.find(function(row) {
    return ACTIVE_CONTRACT_STATUSES.has(normalizeContractStatus((row && (row.status != null ? row.status : row.contract_status)) || ''));
  }) || null;
  const effectiveContract = activeContract || latest || null;
  const effectiveStatus = normalizeContractStatus((effectiveContract && (effectiveContract.status != null ? effectiveContract.status : effectiveContract.contract_status)) || '') || 'none';
  return {
    status: effectiveStatus,
    hasContract: rows.length > 0,
    hasActiveContract: !!activeContract,
    latestContract: latest,
    effectiveContract: effectiveContract
  };
}

function resolveHasActiveContractFromState(stateLike) {
  const state = stateLike && typeof stateLike === 'object' ? stateLike : {};
  if (state.hasActiveContract === true) return true;
  const status = normalizeContractStatus(state.status);
  if (ACTIVE_CONTRACT_STATUSES.has(status)) return true;
  const effective = state.effectiveContract || state.latestContract || null;
  const effectiveStatus = normalizeContractStatus((effective && (effective.status != null ? effective.status : effective.contract_status)) || '');
  return ACTIVE_CONTRACT_STATUSES.has(effectiveStatus);
}

function scheduleRealtimeLoad(reason) {
  if (realtimeSuppressUntil > 0 && Date.now() < realtimeSuppressUntil) return;
  if (realtimeLoadDebounceTimer) clearTimeout(realtimeLoadDebounceTimer);
  realtimeLoadDebounceTimer = setTimeout(function() {
    realtimeLoadDebounceTimer = null;
    loadData().catch(function(err) {
      console.error('[realtime-sync] loadData failed', { reason: reason || 'unknown', error: err && err.message ? err.message : err });
    });
  }, 250);
}

function teardownDashboardRealtime() {
  if (!_sb || !dashboardRealtimeChannel) return;
  try {
    _sb.removeChannel(dashboardRealtimeChannel);
  } catch (err) {
    console.warn('[realtime-sync] removeChannel failed', err);
  }
  dashboardRealtimeChannel = null;
  dashboardRealtimeKey = '';
}

function setupDashboardRealtime(customerIdRaw) {
  if (!_sb || typeof _sb.channel !== 'function') return;
  const customerId = String(customerIdRaw || '').trim();
  if (!customerId) {
    teardownDashboardRealtime();
    return;
  }
  const nextKey = `dashboard:${customerId}`;
  if (dashboardRealtimeChannel && dashboardRealtimeKey === nextKey) return;
  teardownDashboardRealtime();
  const channelName = `customer-dashboard-${customerId}`;
  const channel = _sb
    .channel(channelName)
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'calls',
      filter: `customer_id=eq.${customerId}`
    }, function(payload) {
      var next = (payload && payload.new) || null;
      var nextId = next && next.id != null ? String(next.id) : '';
      var nextLiveStatus = String((next && next.live_status) || '').toLowerCase();
      var isTerminalLiveStatus = CALL_LIVE_TERMINAL_STATES.has(nextLiveStatus);
      if (nextId && isTerminalLiveStatus && Array.isArray(allRecords) && allRecords.length) {
        var idx = allRecords.findIndex(function(r) { return String((r && r.id) || '') === nextId; });
        if (idx >= 0 && allRecords[idx] && allRecords[idx].fields) {
          allRecords[idx].fields.live_status = next.live_status;
          if (next.updated_at) allRecords[idx].fields.updated_at = next.updated_at;
          if (next.completed_at) allRecords[idx].fields.completed_at = next.completed_at;
          if (next.duration_seconds !== undefined) allRecords[idx].fields.duration_seconds = next.duration_seconds;
        }
      }
      if (isTerminalLiveStatus) {
        // Terminal-Callbacks müssen LIVE sofort ausblenden (keine künstliche Nachlaufzeit).
        renderDashboard(allRecords, allManualTaskRecords);
        renderAnrufe();
      }
      scheduleRealtimeLoad(isTerminalLiveStatus ? 'calls-terminal-change' : 'calls-change');
    })
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'customer_tasks',
      filter: `customer_id=eq.${customerId}`
    }, function() {
      scheduleRealtimeLoad('cases-change');
    })
    .on('postgres_changes', {
      event: '*',
      schema: 'public',
      table: 'customers',
      filter: `id=eq.${customerId}`
    }, function() {
      scheduleRealtimeLoad('customer-change');
    })
    .subscribe(function(status) {
      if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
        console.warn('[realtime-sync] channel degraded, polling fallback remains active', { status: status, customerId: customerId });
      }
    });
  dashboardRealtimeChannel = channel;
  dashboardRealtimeKey = nextKey;
}

function normalizeTaskStatus(rawStatus) {
  const raw = String(rawStatus || '').trim().toLowerCase();
  if (raw === 'closed' || raw === 'done' || raw === 'abgeschlossen' || raw === 'resolved' || raw === 'erledigt') return 'done';
  if (raw === 'waiting_external' || raw === 'waiting') return 'in_progress';
  if (raw === 'planned' || raw === 'scheduled' || raw === 'geplant' || raw === 'follow_up_scheduled') return 'planned';
  if (raw === 'in_progress' || raw === 'follow_up_scheduled' || raw === 'in bearbeitung') return 'in_progress';
  if (raw === 'archived' || raw === 'archiviert' || raw === 'deleted' || raw === 'cancelled') return 'archived';
  return 'open';
}

/* PR-M — Entry lifecycle source of truth.
   One operational status only: archived → done → planned → open.
   Important: callback_requested alone is NOT planned. */
function vxNormalizeStatusToken(value) {
  var raw = String(value == null ? '' : value).trim().toLowerCase();
  if (!raw) return '';
  raw = raw.replace(/[\u2010-\u2015]/g, '-').replace(/[_-]+/g, ' ').replace(/\s+/g, ' ').trim();
  var archivedAliases = ['archived','archive','archiv','archiviert','deleted','gelöscht','geloescht','cancelled','canceled','storniert'];
  var doneAliases = ['done','closed','resolved','erledigt','abgeschlossen','geschlossen','fertig'];
  var plannedAliases = ['planned','geplant','scheduled','followup','follow up','follow up scheduled','follow up geplant','callback planned','callback geplant','rückruf geplant','rueckruf geplant'];
  var openAliases = ['new','neu','open','offen','pending','in progress','in bearbeitung','processing','triaged','waiting external','waiting'];
  if (archivedAliases.indexOf(raw) !== -1) return 'archived';
  if (doneAliases.indexOf(raw) !== -1) return 'done';
  if (plannedAliases.indexOf(raw) !== -1) return 'planned';
  if (openAliases.indexOf(raw) !== -1) return 'open';
  return raw;
}

function vxHasFilledField(record, keys) {
  for (var i = 0; i < keys.length; i++) {

  }

}

function vxGetEntryLifecycleStatus(record) {
  if (!record) return 'open';

  var archivedDateFields = ['archived_at','deleted_at'];
  var doneDateFields = ['completed_at','closed_at','done_at','resolved_at'];
  var plannedDateFields = ['callback_at','follow_up_at','followup_at','followUpAt','due_at','due_date','reminder_at','planned_at','task_due_at'];
  var statusFields = ['dashboard_status','status','state','case_status','call_status','live_status'];

  // 1. Archived wins over everything.
  if (vxHasFilledField(record, archivedDateFields)) return 'archived';
  for (var a = 0; a < statusFields.length; a++) {
    if (vxNormalizeStatusToken(vxRecordField(record, statusFields[a])) === 'archived') return 'archived';
  }

  // 2. Done wins over planned/open, even if legacy status still says new.
  if (vxHasFilledField(record, doneDateFields)) return 'done';
  for (var d = 0; d < statusFields.length; d++) {
    if (vxNormalizeStatusToken(vxRecordField(record, statusFields[d])) === 'done') return 'done';
  }

  // 3. Planned only means a real due/follow-up timestamp or explicit planned status.
  if (vxHasFilledField(record, plannedDateFields)) return 'planned';
  for (var p = 0; p < statusFields.length; p++) {
    if (vxNormalizeStatusToken(vxRecordField(record, statusFields[p])) === 'planned') return 'planned';
  }

  // callback_requested alone remains an open request, not a planned one.
  return 'open';
}

function vxIsDoneRecord(record) {

}

function vxHasPlannedNextAction(record) {

function vxIsUnreadEligibleRecord(record) {
  var f = (record && record.fields) || {};
  try { if (typeof isManualTaskRecord === 'function' && isManualTaskRecord(record)) return false; } catch(e) {}
  try { if (typeof vxIsDoneRecord === 'function' && vxIsDoneRecord(record)) return false; } catch(e) {}
  try { if (typeof vxHasPlannedNextAction === 'function' && vxHasPlannedNextAction(record)) return false; } catch(e) {}
  var statusCandidates = [
    record && record.dashboard_status, record && record.status, record && record.live_status,
    f.dashboard_status, f.status, f.live_status
  ];
  for (var i = 0; i < statusCandidates.length; i++) {
    var normalized = (typeof vxNormalizeDashboardStatus === 'function') ? vxNormalizeDashboardStatus(statusCandidates[i]) : String(statusCandidates[i] || '').toLowerCase();
    if (normalized === 'done' || normalized === 'planned') return false;
  }
  return true;
}

function vxIsUnreadRecord(record) {
  // [PR-I 2026-05-23] Apple-like unread logic.
  // Ungelesen ist ein eigener Inbox-State, kein Ableitung aus "Offen", "new" oder fehlenden Legacy-Feldern.
  if (!vxIsUnreadEligibleRecord(record)) return false;
  var f = (record && record.fields) || {};
  var id = String((record && record.id) || f.id || record && record.call_id || f.call_id || '').trim();

  function hasOwn(obj, key) {
    return !!obj && Object.prototype.hasOwnProperty.call(obj, key);
  }
  function parseBoolFlag(value) {
    if (value === true || value === 1) return true;
    if (value === false || value === 0) return false;
    if (value == null) return null;
    var s = String(value).trim().toLowerCase();
    if (['true','1','yes','ja','y'].indexOf(s) !== -1) return true;
    if (['false','0','no','nein','n',''].indexOf(s) !== -1) return false;
    return null;
  }

  // Manual override aus der aktuellen Session hat Priorität.
  try {
    if (id && typeof vxGetManualUnreadIdSet === 'function' && vxGetManualUnreadIdSet().has(id)) return true;
  } catch(e) {}

  // Explizite Flags korrekt parsen. Wichtig: String "false" darf NICHT truthy als ungelesen zählen.
  var explicitCandidates = [];
  if (hasOwn(record, 'is_unread')) explicitCandidates.push(record.is_unread);
  if (hasOwn(f, 'is_unread')) explicitCandidates.push(f.is_unread);
  if (hasOwn(record, 'unread')) explicitCandidates.push(record.unread);
  if (hasOwn(f, 'unread')) explicitCandidates.push(f.unread);
  for (var i = 0; i < explicitCandidates.length; i++) {
    var parsed = parseBoolFlag(explicitCandidates[i]);
    if (parsed === true) return true;
    if (parsed === false) return false;
  }

  // read_at ist nur aussagekräftig, wenn das Feld wirklich vorhanden ist.
  // Fehlendes read_at bei Legacy-Daten bedeutet NICHT automatisch ungelesen.
  var hasReadAt = hasOwn(record, 'read_at') || hasOwn(f, 'read_at');
  if (hasReadAt) {
    var readAt = hasOwn(record, 'read_at') ? record.read_at : f.read_at;
    if (readAt != null && String(readAt).trim() !== '') return false;
    // Wenn read_at explizit null ist, gilt der Eintrag als ungelesen – ausser er ist lokal bereits gesehen.
    try {
      if (id && typeof getSeenCallIdsFromStorage === 'function' && getSeenCallIdsFromStorage().has(id)) return false;
    } catch(e) {}
    return true;
  }

  return false;
}

function vxClassifyRequestRecord(record) {
  var f = (record && record.fields) || {};
  var isDone = vxIsDoneRecord(record);
  var isPlanned = !isDone && vxHasPlannedNextAction(record);
  var isUnread = vxIsUnreadRecord(record);
  var bucket = isDone ? 'done' : (isPlanned ? 'planned' : 'inbox');
  var statusLabel = bucket === 'done' ? 'Erledigt' : (bucket === 'planned' ? 'Geplant' : 'Inbox');
  var statusTone = bucket === 'done' ? 'done' : (bucket === 'planned' ? 'planned' : (isUnread ? 'new' : 'neutral'));
  var reason = isDone
    ? 'status_done_or_closed'
    : (isPlanned ? 'planned_next_action_present' : 'active_without_plan');
  return { bucket: bucket, isUnread: isUnread, statusLabel: statusLabel, statusTone: statusTone, reason: reason };
}

function vxGetRequestCounts(records) {
  var list = Array.isArray(records) ? records.filter(Boolean) : [];
  var counts = { inbox: 0, planned: 0, done: 0, doneToday: 0, unread: 0, read: 0, all: list.length };
  var todayYmd = new Date().toISOString().slice(0, 10);
  list.forEach(function(rec) {
    var c = vxClassifyRequestRecord(rec);
    counts[c.bucket] += 1;
    if (c.isUnread) counts.unread += 1;
    else counts.read += 1;
    if (c.bucket === 'done') {
      var f = (rec && rec.fields) || {};
      var doneTs = String(
        (rec && (rec.completed_at || rec.updated_at)) ||
        f.completed_at || f.updated_at || ''
      ).trim();
      if (doneTs && doneTs.slice(0, 10) === todayYmd) counts.doneToday += 1;
    }
  });
  // Fallback: If no reliable done timestamp exists in data, doneToday mirrors done for KPI continuity.
  if (counts.done > 0 && counts.doneToday === 0) counts.doneToday = counts.done;
  return counts;
}

function resolveTaskDueAt(fields) {
  if (!fields) return '';
  if (fields.due_at) {
    const dueDate = String(fields.due_at).trim();
    const dueTime = String(fields.due_time || '').trim();
    if (/^\d{4}-\d{2}-\d{2}$/.test(dueDate) && /^\d{2}:\d{2}$/.test(dueTime)) {
      return `${dueDate}T${dueTime}`;
    }
    return dueDate;
  }
  return resolveFollowUpAt(fields) || '';
}

function getTaskDueRaw(taskOrFields) {
  var fields = (taskOrFields && taskOrFields.fields) ? taskOrFields.fields : (taskOrFields || {});
  var dueAt = String(fields.due_at || '').trim();
  if (dueAt) {
    var dueTime = String(fields.due_time || '').trim();
    if (/^\d{4}-\d{2}-\d{2}$/.test(dueAt) && /^\d{2}:\d{2}$/.test(dueTime)) return dueAt + 'T' + dueTime;
    return dueAt;
  }
  return String(fields.follow_up_at || fields.followup_at || fields.followUpAt || fields.follow_up || '').trim();
}

function parseTaskDueAt(taskOrFields) {

function vxGetDueSortMeta(recordOrFields) {
  var f = (recordOrFields && recordOrFields.fields) ? recordOrFields.fields : (recordOrFields || {});
  var dueAt = String(f.due_at || '').trim();
  var dueTime = String(f.due_time || '').trim();
  var validDueTime = normalizeManualTaskDueTime(dueTime);

  if (dueAt) {
    if (/^\d{4}-\d{2}-\d{2}$/.test(dueAt)) {
      var dateOnly = vxParseDueDateOnly(dueAt);
      if (!dateOnly || Number.isNaN(dateOnly.getTime())) {
        return { hasDue:false, hasTime:false, missingTime:false, raw:'', date:null, sortMs:Number.MAX_SAFE_INTEGER, source:'manual_due_invalid' };
      }
      if (validDueTime) {
        var timedRaw = dueAt + 'T' + validDueTime;
        var timedDate = parseFollowUpDate(timedRaw);
        return {
          hasDue: !!(timedDate && !Number.isNaN(timedDate.getTime())),
          hasTime: true,
          missingTime: false,
          raw: timedRaw,
          date: timedDate,
          sortMs: timedDate && !Number.isNaN(timedDate.getTime()) ? timedDate.getTime() : Number.MAX_SAFE_INTEGER,
          source: 'manual_due'
        };
      }
      var endOfDay = new Date(dateOnly.getFullYear(), dateOnly.getMonth(), dateOnly.getDate(), 23, 59, 59, 999);
      return {
        hasDue: true,
        hasTime: false,
        missingTime: true,
        raw: dueAt,
        date: dateOnly,
        sortMs: endOfDay.getTime(),
        source: 'manual_due_missing_time'
      };
    }
    var parsedDueAt = parseFollowUpDate(dueAt);
    if (parsedDueAt && !Number.isNaN(parsedDueAt.getTime())) {
      return {
        hasDue: true,
        hasTime: hasMeaningfulDueTime({ due_at: dueAt }),
        missingTime: !hasMeaningfulDueTime({ due_at: dueAt }),
        raw: dueAt,
        date: parsedDueAt,
        sortMs: parsedDueAt.getTime(),
        source: 'manual_due_legacy'
      };
    }
  }

  var followUp = resolveFollowUpAt(f);
  if (followUp) {
    var followDate = parseFollowUpDate(followUp);
    if (followDate && !Number.isNaN(followDate.getTime())) {
      return {
        hasDue: true,
        hasTime: hasMeaningfulDueTime({ follow_up_at: followUp }),
        missingTime: !hasMeaningfulDueTime({ follow_up_at: followUp }),
        raw: followUp,
        date: followDate,
        sortMs: followDate.getTime(),
        source: 'call_follow_up'
      };
    }
  }

  return { hasDue:false, hasTime:false, missingTime:false, raw:'', date:null, sortMs:Number.MAX_SAFE_INTEGER, source:'none' };
}

function vxDashboardDueRank(record, now) {
  var meta = vxGetDueSortMeta(record);
  if (!meta.hasDue) return 3;
  var currentNow = now instanceof Date ? now : new Date();
  var todayStart = new Date(currentNow.getFullYear(), currentNow.getMonth(), currentNow.getDate());
  var tomorrowStart = new Date(todayStart.getTime() + 86400000);
  var nowMs = currentNow.getTime();
  if (meta.sortMs < nowMs) return 0;
  if (meta.sortMs < tomorrowStart.getTime()) return 1;
  return 2;
}

function vxCompareDashboardPriorityRecords(a, b, now) {
  var ar = vxDashboardDueRank(a, now);
  var br = vxDashboardDueRank(b, now);
  if (ar !== br) return ar - br;

  var am = vxGetDueSortMeta(a);
  var bm = vxGetDueSortMeta(b);
  if (am.hasDue || bm.hasDue) {
    if (am.sortMs !== bm.sortMs) return am.sortMs - bm.sortMs;
    if (am.hasTime !== bm.hasTime) return am.hasTime ? -1 : 1;
    var aCreatedAsc = new Date(getRecordTimestamp(a) || 0).getTime();
    var bCreatedAsc = new Date(getRecordTimestamp(b) || 0).getTime();
    if (aCreatedAsc !== bCreatedAsc) return aCreatedAsc - bCreatedAsc;
  }

  var aPriority = getPriority(a).key === 'urgent' ? 0 : 1;
  var bPriority = getPriority(b).key === 'urgent' ? 0 : 1;
  if (aPriority !== bPriority) return aPriority - bPriority;

  var aCallback = (((a && a.fields) || {}).callback_requested === true || ((a && a.fields) || {}).callback_requested === 'true') ? 0 : 1;
  var bCallback = (((b && b.fields) || {}).callback_requested === true || ((b && b.fields) || {}).callback_requested === 'true') ? 0 : 1;
  if (aCallback !== bCallback) return aCallback - bCallback;

  return new Date(getRecordTimestamp(b) || 0).getTime() - new Date(getRecordTimestamp(a) || 0).getTime();
}

// [PATCH 2026-05-13 v2.3 — 00:00-Bug Fix]
// Erkennt ob das Datum eine "bedeutsame" Uhrzeit hat (vom Nutzer gesetzt),
// im Unterschied zu reinem Datum ohne Zeit (das als 00:00 geparsed wird).
// Beispiele:
//   "2026-05-15"              → false (nur Datum, keine Uhrzeit gesetzt)
//   "2026-05-15T00:00"        → false (Zeit war "00:00", vermutlich Default)
//   "2026-05-15T14:30"        → true (echte Uhrzeit)
//   "2026-05-15T14:30:00+02"  → true (timestamptz mit Zeit)
//   ""                        → false (leer)
function hasMeaningfulDueTime(taskOrFields) {
  if (!taskOrFields) return false;
  var raw = getTaskDueRaw(taskOrFields);
  if (!raw) return false;
  raw = String(raw).trim();
  // Format-Check: enthält ein "T" oder eine Leerstelle gefolgt von Uhrzeit-Muster?
  // Wenn nur "YYYY-MM-DD" → keine Zeit-Info
  if (/^\d{4}-\d{2}-\d{2}$/.test(raw)) return false;
  // Wenn die Zeit explizit "00:00" oder "00:00:00" ist → behandeln wir als nicht bedeutsam
  // (kommt von Default-Werten bei Datumseingabe ohne Uhrzeit)
  if (/T00:00(?::00)?(\.\d+)?(Z|[+-]\d{2}:?\d{2})?$/.test(raw)) return false;
  if (/[ T]00:00(:00)?\s*$/.test(raw)) return false;
  // Sonst: vermutlich gesetzte Uhrzeit
  return /T\d{2}:\d{2}/.test(raw) || /\d{2}:\d{2}/.test(raw);
}

function formatTaskDueLabel(taskOrFields, options) {
  var opts = options || {};
  var due = parseTaskDueAt(taskOrFields);
  if (!due) return opts.emptyLabel || 'Ohne Fälligkeit';
  var now = opts.now instanceof Date ? opts.now : new Date();
  var todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate());
  var tomorrowStart = new Date(todayStart.getTime() + 86400000);
  var dayAfterTomorrowStart = new Date(todayStart.getTime() + 172800000);
  // [PATCH 2026-05-13 v2.3 — 00:00-Bug Fix]
  // Nur Uhrzeit anzeigen, wenn sie vom Nutzer gesetzt wurde (nicht Default-00:00)
  var withTime = hasMeaningfulDueTime(taskOrFields);
  var timeLabel = withTime ? formatTimeCH(due) : '';
  var timeSuffix = withTime ? (', ' + timeLabel) : '';
  if (due.getTime() < now.getTime()) {
    if (due >= todayStart && due < tomorrowStart) {
      return withTime ? ('Überfällig seit ' + timeLabel) : 'Heute überfällig';
    }
    return 'Überfällig seit ' + formatDateShortCH(due) + timeSuffix;
  }
  if (due >= todayStart && due < tomorrowStart) {
    return withTime ? ('Heute, ' + timeLabel) : 'Heute';
  }
  if (due >= tomorrowStart && due < dayAfterTomorrowStart) {
    return withTime ? ('Morgen, ' + timeLabel) : 'Morgen';
  }
  return formatDateShortCH(due) + timeSuffix;
}

function normalizePhoneForMatch(phone) {
  var raw = String(phone || '').trim();
  if (!raw) return '';
  var digits = raw.replace(/[^\d]/g, '');
  if (!digits) return '';
  if (digits.startsWith('0041')) digits = '41' + digits.slice(4);
  if (digits.startsWith('41') && digits.length > 9) digits = '0' + digits.slice(2);
  return digits;
}

function sortWorkItemsByOperationalPriority(items) {
  var now = new Date();
  return [...(items || [])].sort(function(a, b) {
    function rank(it) {
      var d = parseFollowUpDate(it.dueAt || '');
      if (!d || Number.isNaN(d.getTime())) return 4;
      var ts = d.getTime();
      var dayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
      var nextDay = dayStart + 86400000;
      if (ts < now.getTime()) return 0;
      if (ts < nextDay) return 1;
      return 2;
    }
    var ar = rank(a), br = rank(b);
    if (ar !== br) return ar - br;
    var ad = parseFollowUpDate(a.dueAt || ''); var bd = parseFollowUpDate(b.dueAt || '');
    var at = ad && !Number.isNaN(ad.getTime()) ? ad.getTime() : Number.MAX_SAFE_INTEGER;
    var bt = bd && !Number.isNaN(bd.getTime()) ? bd.getTime() : Number.MAX_SAFE_INTEGER;
    if (at !== bt) return at - bt;
    return new Date(b.createdAt || 0).getTime() - new Date(a.createdAt || 0).getTime();
  });
}

  if (['sofort','heute','hoch','mittel','urgent','high','medium'].includes(urgency)) reasons.push('urgency');
  if (lead === 'hot' || lead === 'warm') reasons.push('lead_quality');
  if (['offertenanfrage','terminanfrage','ruckrufanfrage','rueckrufanfrage','notfall','termin','appointment'].includes(category)) reasons.push('category');
  if (vxTodayHasMeaningfulNextAction(nextAction)) reasons.push('next_action');
  return reasons;
}
function vxTodayIsRelevantCallForAttention(record) {
  if (!vxTodayIsOpenCallForAttention(record)) return false;
  if (vxTodayIsIgnorableCall(record)) return false;
  return vxTodayCallRelevanceReasons(record).length > 0;
}
function vxTodayIsOpenManualTaskForAttention(record) {

  var f = (record && record.fields) || {};
  var status = vxTodayNormText(f.status || f.dashboard_status || 'new');

}

function vxBuildOperationalRowActions(record, context) {
  context = context || {};
  var id = vxRowActionLabelEscape(String((record && record.id) || ''));
  var safePhone = vxRowActionLabelEscape(vxRowActionPhone(record, context.phone));
  var isManual = typeof isManualTaskRecord === 'function' && isManualTaskRecord(record);
  var lifecycle = vxRowActionLifecycle(record);
  var isDone = lifecycle === 'done';
  var isArchived = lifecycle === 'archived';
  var isPlanned = lifecycle === 'planned';
  var hasPhone = !!vxRowActionPhone(record, context.phone);
  var primaryAction = isDone || isArchived
    ? { action:'fulldetail', title:'Details', icon:'sidebar-simple' }
    : { action:'done', title:'Erledigen', icon:'check' };
  var menuActions = [];

  if (hasPhone) menuActions.push({ action:'call', title:'Anrufen', attrs:' data-phone="'+safePhone+'"' });

  if (isManual) {
    if (isDone || isArchived) {
      menuActions.push({ action:'schedule-callback', title:'Rückruf planen' });
      if (!isArchived) menuActions.push({ action:'archive', title:'Archivieren' });
    } else {
      menuActions.push({ action:'done', title:'Erledigen' });
      menuActions.push({ action:'followup', title:'Fälligkeit ändern' });
      menuActions.push({ action:'edit', title:'Bearbeiten' });
      menuActions.push({ action:'archive', title:'Archivieren' });
    }
    return { primaryAction: primaryAction, menuActions: menuActions };
  }

  if (isDone || isArchived) {
    menuActions.push({ action:'followup', title:'Rückruf planen' });
    if (!isArchived) menuActions.push({ action:'archive', title:'Archivieren' });
    return { primaryAction: primaryAction, menuActions: menuActions };
  }

  menuActions.push({ action:'followup', title:'Rückruf planen' });
  if (isPlanned) menuActions.push({ action:'followup', title:'Fälligkeit ändern' });
  menuActions.push({ action:'fulldetail', title:'Details öffnen' });
  menuActions.push({ action:'archive', title:'Archivieren' });
  return { primaryAction: primaryAction, menuActions: menuActions };
}

function vxTodayGetPrimaryAction(record, typeMeta, phone) {
  var actions = vxBuildOperationalRowActions(record, { phone: phone, surface: 'today', typeMeta: typeMeta });
  return actions.primaryAction;
}
function vxRowActionLifecycle(record) {
  if (record && typeof vxGetEntryLifecycleStatus === 'function') return vxGetEntryLifecycleStatus(record);
  var f = (record && record.fields) || {};
  var status = String(f.dashboard_status || f.status || '').toLowerCase();
  if (f.archived_at || ['archived','archive','archiviert','deleted','cancelled','canceled'].includes(status)) return 'archived';
  if (f.completed_at || f.done_at || f.closed_at || ['closed','done','completed','resolved','erledigt','abgeschlossen'].includes(status)) return 'done';
  if (f.follow_up_at || f.due_at || ['planned','scheduled','geplant','follow_up_scheduled'].includes(status)) return 'planned';
  return 'open';
}
function vxRowActionIsDone(record) {
  var lifecycle = vxRowActionLifecycle(record);
  return lifecycle === 'done' || lifecycle === 'archived';
}
function vxRowActionIsArchived(record) {
  return vxRowActionLifecycle(record) === 'archived';
}
function vxRowActionLabelEscape(v) {
  return (typeof _esc === 'function') ? _esc(v) : String(v || '').replace(/[&<>'"]/g, function(c){ return {'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c]; });
}
