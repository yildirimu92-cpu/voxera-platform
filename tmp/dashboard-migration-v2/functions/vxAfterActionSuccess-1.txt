function vxAfterActionSuccess(options) {
  var opts = options || {};
  var targetId = String(opts.createdRecordId || opts.recordId || '').trim();
  if (targetId) {
    window._vxActiveRecordId = targetId;
    window._vxPendingHighlightRecordId = targetId;
  }
  if (opts.closeModal !== false) {
    if (opts.modal === 'manual-task' && typeof closeManualTaskModal === 'function') closeManualTaskModal();
    else if (opts.modal === 'followup' && typeof closeFollowup === 'function') closeFollowup();
  }
  if (typeof vxRefreshOperationalViews === 'function') vxRefreshOperationalViews(allRecords, allManualTaskRecords);
  else {
    if (typeof renderDashboard === 'function') renderDashboard(allRecords, allManualTaskRecords);
    if (typeof renderAnrufe === 'function') renderAnrufe();
    if (typeof renderRueckrufe === 'function') renderRueckrufe(allRecords, allManualTaskRecords);
  }
  try { vxRefreshActiveDetailAfterAction({ recordId: opts.recordId || '', createdRecordId: opts.createdRecordId || '' }); } catch(e) {}
  if (targetId && typeof vxSetActiveRequestRow === 'function') {
    setTimeout(function(){ try { vxSetActiveRequestRow(targetId); } catch(e) {} }, 0);
  }
}
