function renderDashPriorityList() {
  var el = document.getElementById('dash-priority-list');
  if (!el) return;

  var records = (_dashPriorityAllRecords || []).filter(function(rec){ return isManualTaskRecord(rec) || isCustomerVisibleCall(rec); });
  el.classList.add('dpr-command-center');
  // Launch guard: "Laufende Anrufe" darf ausschliesslich echte Live-Calls zeigen.
  // Completed-/duration-Calls mit fehlender Summary sind normale Anfragen, keine temporären Live-Zeilen.
  var tempRecords = [];

  if (records.length === 0 && tempRecords.length === 0) {
    vxSetHtmlIfChanged(el, '<div style="padding:24px 20px;text-align:center;color:var(--slate2);font-size:13px;background:#fff;">' +
      '<i class="ph-bold ph-check-circle" style="color:#059669;margin-bottom:8px;"></i><br>' +
      'Alles erledigt &mdash; ' + getAssistantName() + ' nimmt neue Anrufe entgegen.</div>');
    refreshLucideIcons();
    return;
  }

  var limit = dashPriorityExpanded ? 999 : 5;
  var visible = records.slice(0, limit);
  var tempKeys = new Set();
  tempRecords.forEach(function(rec){
    var f = (rec && rec.fields) || {};
    var recIdKey = String((rec && rec.id) || '').trim();
    var callIdKey = String(f.call_id || '').trim();
    if (recIdKey) tempKeys.add(recIdKey);
    if (callIdKey) tempKeys.add(callIdKey);
  });
  var groups = { overdue: [], today: [], planned: [], calls: [] };
  var groupOrder = ['overdue', 'today', 'planned', 'calls'];

  visible.forEach(function(rec) {
    var keyF = (rec && rec.fields) || {};
    var recIdKey = String(rec.id || '').trim();
    var callIdKey = String(keyF.call_id || '').trim();
    if ((recIdKey && tempKeys.has(recIdKey)) || (callIdKey && tempKeys.has(callIdKey))) return;
    var typeMeta = vxDashGetCommandTypeMeta(rec);
    var dueMeta = vxDashGetDueMeta(rec, typeMeta);
    var key = groups[dueMeta.group] ? dueMeta.group : 'today';
    groups[key].push({ record: rec, typeMeta: typeMeta, dueMeta: dueMeta });
  });

  var html = '';

  if (tempRecords.length) {
    var maxTemp = tempRecords.slice(0, 3);
    html += '<div class="dpr-group dpr-group-calls">';
    html += '<div class="vx-group-head"><span class="vx-group-label"><i class="ph-bold ph-phone-incoming" aria-hidden="true"></i>Laufende Anrufe</span><span class="vx-group-count">' + tempRecords.length + '</span></div>';
    maxTemp.forEach(function(rec){
      var f = rec.fields || {};
      var st = (typeof vxGetCallDisplayState === 'function' ? vxGetCallDisplayState(rec) : null) || (typeof vxGetDefaultCallDisplayState === 'function' ? vxGetDefaultCallDisplayState() : { state:'ready', label:'Neu', isTemporary:false });
      var name = vxDashResolveWorkTitle(rec);
      var phone = String(f.caller_phone || f.phone_number || f.phone || '').trim();
      var summaryText = st.state === 'live' ? 'Lara nimmt den Anruf gerade entgegen.' : 'Zusammenfassung wird erstellt.';
      html += '<div class="dpr-card" data-id="'+_esc(rec.id)+'" data-phone="'+_esc(phone)+'" data-kind="incoming" data-rail="new">';
      html += '<div class="vx-heute-row rail-today" onclick="dprToggleClick(event,this)" ontouchstart="dprToggleTouchStart(event,this)" ontouchmove="dprToggleTouchMove(event)" ontouchend="dprToggleTouchEnd(event,this)" data-id="'+_esc(rec.id)+'" data-record-id="'+_esc(rec.id)+'">';
      html += '<div class="vx-row-icon is-call"><i class="ph-bold ph-phone-incoming"></i></div><div class="vx-row-body">';
      html += '<div class="vx-row-name">'+_esc(name)+'<span class="vx-row-pill">'+_esc(st.label)+'</span></div>';
      html += '<div class="vx-row-summary">'+_esc(summaryText)+'</div>';
      html += '</div></div></div>';
    });
    if (tempRecords.length > 3) {
      html += '<div style="padding:8px 14px;font-size:12px;color:var(--slate2);background:#fff;">+' + (tempRecords.length - 3) + ' weitere laufende Anrufe</div>';
    }
    html += '</div>';
  }

  var vxHeuteVisibleActionIndex = 0;
  groupOrder.forEach(function(groupKey) {
    var items = groups[groupKey] || [];
    if (!items.length) return;
    html += '<div class="dpr-group dpr-group-' + groupKey + '">';
    html += '<div class="vx-group-head"><span class="vx-group-label"><i class="ph-bold ph-' + (groupKey === 'overdue' ? 'alarm' : groupKey === 'today' ? 'sun' : groupKey === 'planned' ? 'calendar-dots' : 'phone-incoming') + '" aria-hidden="true"></i>' + vxDashGroupLabel(groupKey) + '</span><span class="vx-group-count">' + items.length + '</span></div>';

    items.forEach(function(item) {
      var rec = item.record;
      var f = rec.fields || {};
      var typeMeta = item.typeMeta;
      var dueMeta = item.dueMeta;
      var phone = String(f.caller_phone || f.phone_number || f.phone || '').trim();
      var name = vxDashResolveWorkTitle(rec);
      var company = (f.company_name && f.caller_name && f.company_name !== f.caller_name) ? f.company_name : '';
      var description = vxDashGetTaskDescription(rec, typeMeta, name);
      var summary = f.summary || f.call_summary || f.note || f.notes || '';
      var category = '';
      try { category = getCallCategoryDisplayLabel(rec) || ''; } catch(e) { category = ''; }
      var duration = f.duration_seconds ? _formatDuration(f.duration_seconds) : '';
      // [v2.67] Meta-Zeile: nur Telefon
      var metaLine = (phone && name !== phone) ? _esc(phone) : '';
      var action = typeMeta.action;
      var actionLabel = typeMeta.actionLabel;
      var actionIcon = typeMeta.actionIcon;
      var actionClass = typeMeta.actionClass;
      if (action === 'call' && !phone) {
        action = 'edit'; actionLabel = 'Bearbeiten'; actionIcon = 'pencil'; actionClass = 'is-soft-blue';
      }

      // [PATCH v2.53] Rail + Preview
      var _railVal = typeMeta.kind === 'callback' ? 'callback'
        : dueMeta.state === 'overdue' ? 'urgent'
        : dueMeta.state === 'planned' ? 'planned'
        : 'new';
      // [v3.0] Rail-Klasse direkt auf Row
      var _railClass = 'vx-heute-row';
      if (_railVal === 'urgent') _railClass += ' rail-overdue';
      else if (_railVal === 'planned') _railClass += ' rail-planned';
      else _railClass += ' rail-today';
      html += '<div class="dpr-card" data-id="'+_esc(rec.id)+'" data-phone="'+_esc(phone)+'" data-kind="'+_esc(typeMeta.kind)+'" data-rail="'+_railVal+'">';
      html += '<div class="'+_railClass+'" onclick="dprToggleClick(event,this)" ontouchstart="dprToggleTouchStart(event,this)" ontouchmove="dprToggleTouchMove(event)" ontouchend="dprToggleTouchEnd(event,this)" data-id="'+_esc(rec.id)+'" data-record-id="'+_esc(rec.id)+'">';
      // [v2.72] Zusammenfassung: KI-Summary für Calls, Notiz für Tasks (max 60 Zeichen)
      var _rowSummary = '';
      var _rawSum = String(f.call_summary || f.summary || f.note || f.notes || '').trim();
      if (_rawSum && !vxDashIsGenericLabel(_rawSum)) {
        _rowSummary = _rawSum.length > 60 ? _rawSum.slice(0, 57).trim() + '…' : _rawSum;
      }

      // [v3.0] Saubere vx-* Klassen
      var _isCallType = typeMeta.kind === 'callback' || typeMeta.kind === 'incoming';
      var _iconClass = _isCallType ? 'vx-row-icon is-call' : 'vx-row-icon';
      html += '<div class="'+_iconClass+'"><i class="ph-bold ph-'+typeMeta.icon+'"></i></div>';
      html += '<div class="vx-row-body">';
      var _isUnreadCall = (typeof vxIsUnreadRecord === 'function') ? vxIsUnreadRecord(rec) : (!isManualTaskRecord(rec) && !f.read_at);
      html += '<div class="vx-row-name" style="'+(_isUnreadCall?'font-weight:700;':'')+'">'+(_isUnreadCall?'<span style="display:inline-block;width:4px;height:4px;border-radius:50%;background:#1A6FE8;margin-right:6px;vertical-align:middle;"></span>':'')+_esc(name)+'<span class="vx-row-pill">'+_esc(typeMeta.label)+'</span>'+(company ? '<span class="dpr-co">'+_esc(company)+'</span>' : '')+'</div>';
      if (dueMeta && dueMeta.value) html += '<div class="vx-row-time">'+_esc(dueMeta.value)+'</div>';
      if (_rowSummary) html += '<div class="vx-row-summary">'+_esc(_rowSummary)+'</div>';
      html += '</div>';
      // [v3.1] Heute-Aktionslogik: eine sichtbare Hauptaktion + Overflow-Menü.
      // Hauptaktion:
      // - Rückruf/Anruf mit Telefonnummer → Anrufen
      // - sonst → Öffnen
      // Sekundäre Aktionen bleiben im Menü, damit die Heute-Seite ruhig bleibt.
      var _isTopAttention = vxHeuteVisibleActionIndex === 0;
      vxHeuteVisibleActionIndex += 1;
      var _primary = (typeof vxTodayGetPrimaryAction === 'function')
        ? vxTodayGetPrimaryAction(rec, typeMeta, phone)
        : { action:(phone ? 'call' : 'open'), title:(phone ? 'Anrufen' : 'Öffnen'), icon:(phone ? 'phone-call' : 'arrow-square-out') };
      var _primaryAction = _primary.action;
      var _primaryTitle = _primary.title;
      var _primaryIcon = _primary.icon;
      var _primaryClass = 'vx-ibtn vx-ibtn-main ' + (_primaryAction === 'call' ? (_isTopAttention ? 'is-call' : 'is-ghost-call') : 'is-open');
      html += '<div class="vx-row-btns vx-row-btns--single">';
      html += '<button type="button" class="'+_primaryClass+'" data-action="'+_primaryAction+'" data-rec-id="'+_esc(rec.id)+'" data-phone="'+_esc(phone)+'" title="'+_esc(_primaryTitle)+'"><i class="ph-bold ph-'+_primaryIcon+'"></i><span>'+_esc(_primaryTitle)+'</span></button>';
      html += '<button type="button" class="vx-ibtn is-overflow" data-action="overflow" data-rec-id="'+_esc(rec.id)+'" data-phone="'+_esc(phone)+'" title="Mehr"><i class="ph-bold ph-dots-three"></i></button>';
      html += '<div class="vx-row-menu" id="vx-row-menu-'+_esc(rec.id)+'">';
      html += (typeof vxTodayBuildRowMenuHtml === 'function')
        ? vxTodayBuildRowMenuHtml(rec, phone, typeMeta)
        : '<button type="button" data-action="fulldetail" data-rec-id="'+_esc(rec.id)+'">Details öffnen</button>';
      html += '</div>';
      html += '</div>';
      html += '</div>';

      // [PATCH v2.38] Gemeinsames expanded Panel — identisch mit Anfragen-Liste
      html += vxBuildExpPanel({
        id: rec.id,
        phone: phone,
        summary: (f.call_summary || f.summary || f.note || f.notes || ''),
        summaryLabel: isManualTaskRecord(rec) ? 'Notiz' : 'KI-Zusammenfassung',
        contextText: description || typeMeta.label || '',
        contextLabel: isManualTaskRecord(rec) ? 'Aufgabe' : 'Kontext',
        urgency: '',  // Heute-Liste: kein urgency-Feld
        isCallback: typeMeta.kind === 'callback',
        isDone: false,
        kind: typeMeta.kind,  // 'callback' | 'incoming' | 'offer' | 'appointment' | 'task'
        showArchive: false
      });
      html += '</div>'; // /dpr-card
    });
    html += '</div>';
  });

  vxSetHtmlIfChanged(el, html);

  el.querySelectorAll('.dpr-card').forEach(function(card) {
    card.addEventListener('click', function(e) {
      if (e.target.closest('.dpr-btn, .va-exp-panel, button')) return;
      var recId = card.dataset.id;
      if (!recId) return;
      // [v2.72] Tap → Detail-Seite auf allen Gerätegrössen
      var rec = typeof vxResolveDprRecord === 'function' ? vxResolveDprRecord(recId) : null;
      if (rec && isManualTaskRecord(rec)) {
        if (typeof showTaskDetail === 'function') { e.stopPropagation(); showTaskDetail(recId); }
      } else {
        if (typeof showCallDetail === 'function') { e.stopPropagation(); showCallDetail(recId); }
      }
    });
  });

  vxUpdateAnfragenCount();

  // [PATCH v2.31 — expanded Panels isolieren]
  el.querySelectorAll('.va-exp-panel').forEach(function(panel) {
    panel.addEventListener('click', function(e) { e.stopPropagation(); }, true);
    panel.addEventListener('touchend', function(e) { e.stopPropagation(); }, true);
  });
  dprRestoreOpen('dash-priority-list');

  // Live-Row als erstes einfügen wenn vorhanden
  var existingLive = document.getElementById('live-call-row');
  if (existingLive && existingLive.parentNode === el) el.insertBefore(existingLive, el.firstChild);

  refreshLucideIcons();
  dprRestoreOpen();
}
