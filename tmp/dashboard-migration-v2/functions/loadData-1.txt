async function loadData() {
  // ── Concurrency Guard ─────────────────────────────────────────────────────
  // Verhindert parallele loadData-Laeufe (Realtime + Polling + focus + confirm
  // koennen gleichzeitig feuern). Ohne Guard ueberschreibt der langsamere Lauf
  // customerMeta / candidateCall des schnelleren -> State Drift.
  // Wichtig: nicht hart verwerfen sondern "pending" merken und nach Abschluss
  // einmal wiederholen. Sonst werden Polling-Aufrufe waehrend pending_test
  // komplett verschluckt -> candidateCall wird nie gesetzt -> Auto-Confirm
  // laeuft nie durch.
  if (loadDataInFlight) {
    loadDataPending = true;
    return;
  }
  loadDataInFlight = true;
  loadDataPending = false;

  const btn = document.getElementById('refresh-btn');
  if (btn) { btn.textContent = '↻ Lädt...'; btn.style.opacity = '.6'; }

  try {
    await loadCustomerMeta().catch(()=>{});
    applyContractAccessStateToUI();
    await loadPlanConfigFromSupabase().catch(()=>{});
    const context = await resolveCustomerContext(true);
    setupDashboardRealtime((context && context.customerId) || '');

    // ─── ACCESS GUARD ───────────────────────────────────────────────────
    if (!(await ensureActiveCustomerOrLogout(context))) return;
    // ─── END ACCESS GUARD ────────────────────────────────────────────────

    const records = await loadCallRecords(context.customerId || '');
    const manualTasks = await loadManualTaskRecords(context.customerId || '');

    flagIncomingRecords(records);
    // Activation-Test-Calls aus der Dashboard-Anzeige ausschliessen.
    // activation_test_outbound und activation_test_inbound sind interne
    // System-Calls und sollen nicht in Anfragen/Verlauf/Dashboard erscheinen.
    // allRecordsRaw includes test calls for polling detection
    allRecordsRaw = records;
    allRecords = records.filter(function(r) {
      return !isActivationCallCategory((r.fields || {}).category);
    });
    allManualTaskRecords = manualTasks;
    window.allRecords = allRecords;
    window.allManualTaskRecords = allManualTaskRecords;
    _vxNotifDataReady = true;
    if (_vxNotifLoaded) {
      vxBellUpdateBadge();
      if (_vxBellOpen) { vxBellRender(); vxBellPageRender(); }
    }

    const titleEl = document.getElementById('greeting-main');
    if (titleEl) {
      var firstName = customerMeta.contactFirstName || (customerMeta.customerName ? customerMeta.customerName.split(' ')[0] : '') || '';
      titleEl.textContent = getTimeBasedGreeting() + (firstName ? ', ' + firstName : '');
      titleEl.dataset.custom = '1';
    }

    renderDashboard(allRecords, allManualTaskRecords);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    renderArchiv(allRecords);
    if (typeof updateLiveHero === 'function') updateLiveHero(allRecords);
    if (typeof renderAuswertung === 'function') renderAuswertung();
    applyContractAccessStateToUI();
    updateLastUpdatedLabel();

    // [legacy] Candidate-call activation polling removed — checkTestCall handles testing now

    // ── Atomarer Loader→Content Übergang ─────────────────────────────────────
    // dash-loading und dash-content werden in einer einzigen synchronen
    // Operation getauscht. Vorher: zwei separate Operationen mit einem Frame
    // Lücke dazwischen → leerer Screen / Flicker.
    // Zusätzlich: Content nur zeigen wenn customerMeta vollständig geladen
    // (customerId vorhanden) – verhindert '–' Flicker bei Voxera-Nummer/Tel.
    routeDashboardFirstSessionExperience();
    safeMountActivationV2({ reason: 'loadData', softGate: false });
    updateSidebarStatus();
    // Loader erst nach allen Render-Operationen verstecken – atomar
    var loadingEl = document.getElementById('dash-loading');
    if (loadingEl) loadingEl.style.display = 'none';

  } catch(e) {
    const rawMsg = String((e && e.message) || 'Unbekannter Fehler');
    const lower = rawMsg.toLowerCase();
    let uiMsg = rawMsg;

    if (lower.includes('airtable')) {
      uiMsg = 'Alte Version erkannt (Legacy Airtable Fehler). Bitte in Netlify "Clear cache and deploy site" ausführen und danach Hard-Reload (Ctrl/Cmd+Shift+R).';
    }
    // Legacy compatibility: older cached deployments can still return stale backend error texts.
    else if (lower.includes('401') || lower.includes('unauthorized')) {
      uiMsg = 'Daten konnten nicht geladen werden. Bitte neu deployen und Seite mit Hard-Reload öffnen (Ctrl/Cmd+Shift+R).';
    } else if (lower.includes('permission') || lower.includes('rls')) {
      uiMsg = 'Keine Berechtigung für diese Daten (RLS). Bitte Benutzer-Zuordnung in Supabase prüfen.';
    }

    const _loadingEl = document.getElementById('dash-loading');
    if (_loadingEl) _loadingEl.innerHTML =
      `<div class="error-banner">Fehler beim Laden: ${uiMsg}<br><small>Bitte Seite neu laden.</small></div>`;
  } finally {
    // Concurrency Guard: freigeben und pending-Run nachholen falls noetig
    loadDataInFlight = false;
    if (btn) { btn.textContent = '↻ Aktualisieren'; btn.style.opacity = '1'; }
    if (loadDataPending) {
      loadDataPending = false;
      // Minimalverzoegerung damit Browser-Event-Loop nicht blockiert wird
      setTimeout(function() { loadData().catch(function(){}); }, 50);
    }
  }
}
