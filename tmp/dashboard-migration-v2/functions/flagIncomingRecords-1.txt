function flagIncomingRecords(records) {
  const ids = new Set(records.map(r => r.id));
  if (!hasLoadedInitialRecords) {
    // Baseline nur setzen wenn wirklich Daten vorhanden – sonst wird nach
    // Login alles als "neu" markiert (leere Pre-Auth-Antwort als Baseline).
    if (records.length === 0) return;
    seenRecordIds = ids;
    hasLoadedInitialRecords = true;
    return;
  }

  const newIds = records.map(r => r.id).filter(id => !seenRecordIds.has(id));
  seenRecordIds = ids;

  if (!newIds.length) return;

  // Filter out test calls from banner notifications
  const visibleNewIds = newIds.filter(function(id) {
    var rec = records.find(function(r) { return r.id === id; });
    return rec && !isActivationCallCategory((rec.fields || {}).category);
  });

  if (!visibleNewIds.length) return;

  visibleNewIds.forEach(id => recentIncomingIds.add(id));
  _incomingBannerRecordIds = visibleNewIds;

  // Collect call data for the banner
  var newRecords = visibleNewIds.map(id => records.find(r => r.id === id)).filter(Boolean);
  showIncomingBanner(visibleNewIds.length, newRecords);

  setTimeout(() => {
    visibleNewIds.forEach(id => recentIncomingIds.delete(id));
    renderDashboard(allRecords, allManualTaskRecords);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    renderArchiv(allRecords);
    vxUpdateRefreshPolling(allRecords);
  }, 18000);
}
