function vxSetActiveRequestRow(recordId) {
  var activeId = recordId == null ? '' : String(recordId).trim();
  window._vxActiveRecordId = activeId || null;
  var selectors = [
    '#anrufe-list .dpr-card',
    '#anrufe-list .vx-heute-row',
    '#anrufe-list .vx-anf-row',
    '#anrufe-split-left .dpr-card',
    '#anrufe-split-left .vx-heute-row',
    '#anrufe-split-left .vx-anf-row',
    '#dash-priority-list .dpr-card'
  ].join(',');
  document.querySelectorAll(selectors).forEach(function(el) {
    var id = '';
    if (el && el.dataset) {
      id = String(el.dataset.id || el.dataset.recordId || '');
    }
    if (!id && el && el.closest) {
      var card = el.closest('.dpr-card');
      id = card && card.dataset ? String(card.dataset.id || card.dataset.recordId || '') : '';
    }
    el.classList.toggle('sp-active', !!activeId && id === activeId);
  });
}
