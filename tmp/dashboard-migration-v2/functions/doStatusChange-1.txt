async function doStatusChange(recordId, newStatus, rowElement) {
  if (hasKnownInactiveContract()) {
    toast(getRestrictedContractMessage('Aufgaben'), 'warning');
    logManualTaskDoneFlow('guard.blocked.contract-inactive', {
      recordId: String(recordId || ''),
      contractState: customerContractState
    });
    return false;
  }
  const normalizedStatus = normalizeStatus(newStatus);
  const isManualTask = (typeof vxIsManualTaskRecordId === 'function') ? vxIsManualTaskRecordId(recordId) : String(recordId || '').startsWith('case:');
  if (isManualTask) {
    logManualTaskDoneFlow('status-change.start', {
      recordId: String(recordId || ''),
      requestedStatus: String(newStatus || ''),
      normalizedStatus,
      isManualTask: true
    });
  }
  const row = rowElement || null;
  const sourceRecords = isManualTask ? (allManualTaskRecords || []) : (allRecords || []);
  const cached = (typeof vxResolveDprRecord === 'function')
    ? vxResolveDprRecord(recordId)
    : sourceRecords.find(r => String(r.id) === String(recordId));
  const currentStatus = normalizeStatus(cached && cached.fields ? (cached.fields.dashboard_status || cached.fields.status) : CALL_STATUS.NEW);

  if (currentStatus === normalizedStatus) {
    if (cached && cached.fields) cached.fields.dashboard_status = normalizedStatus;
    renderDashboard(allRecords, allManualTaskRecords);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    renderArchiv(allRecords);
    toast('Status bereits gesetzt');
    return true;
  }

  if (row) {
    row.style.transition = 'all .3s ease';
    if (isClosedStatus(normalizedStatus)) {
      row.style.opacity = '.5';
      row.style.background = '#F0FDF4';
      row.style.borderColor = '#BBF7D0';
    } else if ([CALL_STATUS.IN_PROGRESS, CALL_STATUS.FOLLOW_UP_SCHEDULED].includes(normalizedStatus)) {
      row.style.background = '#FFFBEB';
      row.style.borderColor = '#FDE68A';
    } else {
      row.style.opacity = '1';
      row.style.background = '';
      row.style.borderColor = '';
    }
    const secRow = row.querySelector('.call-secondary');
    if (secRow) {
      secRow.querySelectorAll('.chip').forEach(c => c.remove());
      const original = sourceRecords.find(r => String(r.id) === String(recordId));
      if (original && original.fields) {
        original.fields.dashboard_status = normalizedStatus;
        secRow.innerHTML = callCardBadgeHtml(original);
      }
    }
  }

  const icons = {
    [CALL_STATUS.NEW]:'Neu',
    [CALL_STATUS.IN_PROGRESS]:'In Bearbeitung',
    [CALL_STATUS.FOLLOW_UP_SCHEDULED]:'In Bearbeitung',
    [CALL_STATUS.CLOSED]:'Erledigt',
    [CALL_STATUS.ARCHIVED]:'Archiviert'
  };
  try {
    realtimeSuppressUntil = Date.now() + 2000; // Realtime unterdrücken VOR API-Call
    if (isManualTask) {
      const caseStatus = mapDashboardStatusToManualTaskDbStatus(normalizedStatus);
      logManualTaskDoneFlow('status-change.manual-mapping', {
        recordId: String(recordId || ''),
        normalizedStatus,
        mappedCaseStatus: caseStatus
      });
      const updated = await updateManualTaskStatus(recordId, caseStatus);
      const updatedRawId = String((updated && updated.fields && updated.fields.task_entity_id) || '').trim();
      const idx = (allManualTaskRecords || []).findIndex(function(r) {
        var f = (r && r.fields) || {};
        return String(r.id) === String(recordId) || String(f.task_entity_id || '') === String(recordId).replace(/^case:/,'') || (updatedRawId && String(f.task_entity_id || '') === updatedRawId);
      });
      if (idx >= 0) allManualTaskRecords[idx] = updated;
      var taskLifecyclePatch = { updated_at: new Date().toISOString() };
      if (caseStatus === 'closed') {
        taskLifecyclePatch.completed_at = new Date().toISOString();
        taskLifecyclePatch.archived_at = null;
      } else if (caseStatus === 'archived') {
        taskLifecyclePatch.archived_at = new Date().toISOString();
      } else if (caseStatus === 'new') {
        taskLifecyclePatch.completed_at = null;
        taskLifecyclePatch.archived_at = null;
      }
      await vxBestEffortPatchTaskLifecycle(recordId, taskLifecyclePatch);
    } else {
      await applyStatusTransition(recordId, currentStatus, normalizedStatus);
      if (cached && cached.fields) {
        cached.fields.dashboard_status = normalizedStatus;
        if (isClosedStatus(normalizedStatus)) cached.fields.follow_up_at = null;
      }
    }
    renderDashboard(allRecords, allManualTaskRecords);
    renderAnrufe();
    renderRueckrufe(allRecords, allManualTaskRecords);
    renderArchiv(allRecords);
    toast(icons[normalizedStatus] || 'Status aktualisiert', isClosedStatus(normalizedStatus) ? 'success' : 'info');
    if (isClosedStatus(normalizedStatus) || normalizedStatus === CALL_STATUS.ARCHIVED) {
      if (typeof vxCloseAfterSuccessfulCompletion === 'function') setTimeout(vxCloseAfterSuccessfulCompletion, 120);
    }
    return true;
  } catch(e) {
    // [PATCH v2.21 — illegal call transition explizit abfangen]
    // Das Backend lehnt bestimmte Status-Übergänge mit 422 / "illegal call transition" ab.
    // Statt generischem Fehlertext: klare Erklärung + Reload-Option anbieten.
    var errMsg = String(e && e.message ? e.message : '').toLowerCase();
    var isIllegalTransition = errMsg.includes('illegal') || errMsg.includes('transition') ||
                              errMsg.includes('422') || errMsg.includes('cannot archive') ||
                              errMsg.includes('invalid status');
    if (isIllegalTransition) {
      toast('Status kann nicht geändert werden — bitte Seite neu laden.', 'warning');
      // Lokalen Status zurücksetzen damit UI nicht inkonsistent bleibt
      if (cached && cached.fields) {
        cached.fields.dashboard_status = currentStatus;
      }
    } else {
      toast('⚠ ' + toManualTaskUserMessage(e, 'Fehler beim Status-Update'));
    }
    if (row) { row.style.opacity='1'; row.style.background=''; row.style.borderColor=''; }
    return false;
  }
}
